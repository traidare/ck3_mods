	# Law of the Land Title Succession
	# Applied in on_title_gain when a county title's capital province has required heir government types defined and the new holder does not meet the requirement. 
	law_of_the_land_succession_law = {
		can_title_have = {
			tier = tier_county
			title_province = {
				holding_type = {
					has_required_heir_governments = yes
					NOT = {
						any_required_heir_government_type = {
							this = root.holder.government_type
						}
					}
				}
			}
		}
		# Never show this in the UI, just apply it through script
		should_show_for_title = { always = no }
		can_remove_from_title = { always = no }
		succession = {
			order_of_succession = generate_from_template
		}
	}
	#Confederation elective
	confederation_elective_succession_law = {
		can_title_have = {
			tier >= tier_kingdom
		}
		should_show_for_title = {
			has_variable = confederation_culture
		}
		succession = {
			order_of_succession = election
			election_type = confederation_elective
		}
		flag = elective_succession_law
		title_allegiance_opinion = 10
		revoke_cost = {
			prestige = {
				value = change_title_succession_law_prestige_cost
				multiply = 8
				if = {
					limit = {
						NOR = {
							government_has_flag = government_is_tribal
							government_has_flag = government_is_nomadic
						}
					}
					multiply = 0.1
				}
			}
		}
		pass_cost = {
			prestige = change_title_succession_law_prestige_cost
		}
	}

	#AGOT Added
	magisterial_attached_titles_law = {
		can_have = {
			OR = {
				has_realm_law = free_city_election_succession_law
				has_character_flag = transferring_title
				liege ?= { has_realm_law = free_city_election_succession_law }
			}
		}
		can_keep = {
			OR = {
				has_realm_law = free_city_election_succession_law
				has_character_flag = transferring_title
				liege ?= { has_realm_law = free_city_election_succession_law }
			}
		}
		can_pass = {
			always = no
		}
		can_title_have = {
			always = yes
		}
		should_show_for_title = { always = no }
		succession = {
			order_of_succession = player_heir
		}
		flag = elective_succession_law
		revoke_cost = {
			prestige = 150000
		}
		pass_cost = {
			prestige = change_title_succession_law_prestige_cost
		}
		ai_will_do = {
			value = 0
		}
	}

	#AGOT Added
	dictatorship_succession_law = {
		can_have = {
			government_has_flag = government_is_free_city
			is_independent_ruler = yes
			highest_held_title_tier >= tier_kingdom
		}
		can_pass = {
			dread >= 100
		}
		can_title_have = {
			holder = {
				government_has_flag = government_is_free_city
				is_independent_ruler = yes
				highest_held_title_tier >= tier_kingdom
			}
		}
		on_pass = {
			if = {
				limit = {
					has_character_flag = election_delayed_till_war_ends
				}
				remove_character_flag = election_delayed_till_war_ends
			}
		}
		should_start_with = {
			always = no
		}
		succession = {
			order_of_succession = player_heir
		}
		flag = elective_succession_law
		modifier = {
			vassal_opinion = -50
		}
		pass_cost = {
			prestige = 15000
		}
		ai_will_do = {
			value = 0
		}
	}
