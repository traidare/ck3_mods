agot_wom_start_bond_search_effect = {
	add_character_flag = agot_wom_searching_for_bond
	remove_character_flag = agot_wom_refused_call
	if = {
		limit = {
			exists = var:agot_wom_bond_count
			var:agot_wom_bond_count = 1
		}
		add_character_flag = agot_wom_second_bond_search
	}
	trigger_event = {
		id = agot_wom_bonding.0010
		days = { 30 180 }
	}
}

agot_wom_accept_candidate_effect = {
	if = {
		limit = { has_character_flag = agot_wom_candidate_dog }
		set_variable = { name = agot_wom_affinity value = 1 }
	}
	else_if = {
		limit = {
			OR = {
				has_character_flag = agot_wom_candidate_wolf
				has_character_flag = agot_wom_candidate_cat
			}
		}
		set_variable = { name = agot_wom_affinity value = 2 }
	}
	else = {
		set_variable = { name = agot_wom_affinity value = 1 }
	}
	if = {
		limit = {
			has_character_flag = agot_wom_candidate_wolf
			has_character_flag = agot_wom_second_bond_search
		}
		trigger_event = { id = agot_wom_bonding.0200 months = { 6 12 } }
	}
	else_if = {
		limit = { has_character_flag = agot_wom_candidate_wolf }
		trigger_event = { id = agot_wom_bonding.0200 months = { 12 24 } }
	}
	else_if = {
		limit = { has_character_flag = agot_wom_second_bond_search }
		trigger_event = { id = agot_wom_bonding.0200 months = { 3 6 } }
	}
	else = {
		trigger_event = { id = agot_wom_bonding.0200 months = { 6 12 } }
	}
}

agot_wom_resolve_bond_effect = {
	if = {
		limit = {
			has_character_flag = agot_wom_candidate_dog
			var:agot_wom_affinity >= 2
		}
		trigger_event = agot_wom_bonding.0300
	}
	else_if = {
		limit = {
			OR = {
				has_character_flag = agot_wom_candidate_wolf
				has_character_flag = agot_wom_candidate_shadowcat
				has_character_flag = agot_wom_candidate_cat
			}
			var:agot_wom_affinity >= 4
		}
		trigger_event = agot_wom_bonding.0300
	}
	else_if = {
		limit = {
			NOR = {
				has_character_flag = agot_wom_candidate_dog
				has_character_flag = agot_wom_candidate_wolf
				has_character_flag = agot_wom_candidate_shadowcat
				has_character_flag = agot_wom_candidate_cat
			}
			var:agot_wom_affinity >= 3
		}
		trigger_event = agot_wom_bonding.0300
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					has_character_flag = agot_wom_candidate_dog
					var:agot_wom_affinity = 1
				}
				AND = {
					OR = {
						has_character_flag = agot_wom_candidate_wolf
						has_character_flag = agot_wom_candidate_shadowcat
						has_character_flag = agot_wom_candidate_cat
					}
					var:agot_wom_affinity = 3
				}
				AND = {
					NOR = {
						has_character_flag = agot_wom_candidate_dog
						has_character_flag = agot_wom_candidate_wolf
						has_character_flag = agot_wom_candidate_shadowcat
						has_character_flag = agot_wom_candidate_cat
					}
					var:agot_wom_affinity = 2
				}
			}
		}
		random_list = {
			50 = { trigger_event = agot_wom_bonding.0300 }
			50 = { trigger_event = agot_wom_bonding.0301 }
		}
	}
	else = {
		trigger_event = agot_wom_bonding.0301
	}
}

agot_wom_add_bond_effect = {
	add_character_flag = agot_wom_bond_$BEAST$
	add_character_modifier = { modifier = agot_wom_bond_$BEAST$_modifier }
	if = {
		limit = { NOT = { exists = var:agot_wom_bond_count } }
		set_variable = { name = agot_wom_bond_count value = 0 }
	}
	change_variable = { name = agot_wom_bond_count add = 1 }
	if = {
		limit = { var:agot_wom_bond_count = 1 }
		add_character_flag = agot_wom_primary_bond_$BEAST$
	}
	add_trait_xp = { trait = agot_wom_skinchanger value = 3 }
	add_character_flag = { flag = agot_wom_bond_maturing years = 10 }
	remove_character_flag = agot_wom_refused_call
	remove_character_flag = agot_wom_searching_for_bond
}

agot_wom_influence_roll_effect = {
	random = {
		chance = $CHANCE$
		if = {
			limit = {
				NOT = { has_character_modifier = agot_wom_influence_echoes_modifier }
				NOT = { has_character_modifier = agot_wom_influence_$BEAST$_modifier }
			}
			add_character_modifier = { modifier = agot_wom_influence_echoes_modifier }
			trigger_event = agot_wom_warging.0200
		}
		else_if = {
			limit = {
				has_character_modifier = agot_wom_influence_echoes_modifier
				NOT = { has_character_modifier = agot_wom_influence_$BEAST$_modifier }
			}
			remove_character_modifier = agot_wom_influence_echoes_modifier
			add_character_modifier = { modifier = agot_wom_influence_$BEAST$_modifier }
			trigger_event = agot_wom_warging.0201
		}
		else = {
			add_character_flag = { flag = agot_wom_confronting_$BEAST$ days = 5 }
			trigger_event = agot_wom_warging.0210
		}
	}
}
