agot_wom_bonding.0200 = {
	type = character_event
	title = agot_wom_bonding.0200.t
	desc = agot_wom_bonding.0200.desc
	theme = realm

	option = {
		name = agot_wom_bonding.0200.a
		trigger = { NOT = { has_character_flag = agot_wom_candidate_cat } }
		if = {
			limit = {
				OR = {
					AND = {
						has_character_flag = agot_wom_candidate_dog
						OR = { diplomacy >= 10 learning >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_wolf
						OR = { martial >= 10 prowess >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_raven
						OR = { learning >= 10 intrigue >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_eagle
						OR = { learning >= 10 martial >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_bear
						OR = { learning >= 10 prowess >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_shadowcat
						OR = { intrigue >= 10 prowess >= 10 }
					}
					AND = {
						has_character_flag = agot_wom_candidate_deer
						OR = {
							diplomacy >= 10
							learning >= 10
							has_trait = patient
							has_trait = calm
							has_trait = compassionate
						}
					}
				}
			}
			change_variable = { name = agot_wom_affinity add = 2 }
		}
		else = {
			change_variable = { name = agot_wom_affinity add = 1 }
		}
		agot_wom_resolve_bond_effect = yes
	}

	option = {
		name = agot_wom_bonding.0200.b
		trigger = {
			OR = {
				has_character_flag = agot_wom_candidate_bear
				has_character_flag = agot_wom_candidate_shadowcat
				has_character_flag = agot_wom_candidate_cat
			}
		}
		if = {
			limit = { has_character_flag = agot_wom_candidate_bear }
			add_stress = 20
			change_variable = { name = agot_wom_affinity add = 2 }
			agot_wom_influence_roll_effect = { BEAST = bear CHANCE = 30 }
		}
		else_if = {
			limit = { has_character_flag = agot_wom_candidate_shadowcat }
			add_stress = 30
			change_variable = { name = agot_wom_affinity add = 3 }
			agot_wom_influence_roll_effect = { BEAST = shadowcat CHANCE = 35 }
		}
		else = {
			add_stress = 20
			change_variable = { name = agot_wom_affinity add = 2 }
			agot_wom_influence_roll_effect = { BEAST = cat CHANCE = 35 }
		}
		agot_wom_resolve_bond_effect = yes
	}

	option = {
		name = agot_wom_bonding.0200.c
		agot_wom_reject_candidate_effect = yes
	}
}
