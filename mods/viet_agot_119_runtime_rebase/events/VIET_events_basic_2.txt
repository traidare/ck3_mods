# Local woman accuses another of throwing strawberries at her while she was pregnant
VIETmisc.0400 = {
	type = character_event
	title = VIETmisc.0400.t
	desc = VIETmisc.0400.desc
	theme = VIET_theme_small_town


	right_portrait = {
		character = root
		animation = boredom
	}

	trigger = {	
		has_game_rule = VIET_normal_universe_events
		capital_province = {
			VIET_is_in_wild_strawberry_region_trigger = yes
		}	
		# This appears to be a medieval (Western?) European superstition
		culture_has_european_heritage_pillar_trigger = yes
	}
	
	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.2
			has_trait = compassionate
		}
		modifier = {
			add = -0.2
			has_trait = callous
		}
		modifier = {
			add = 0.2
			has_trait = just
		}
		modifier = {
			add = -0.2
			has_trait = arbitrary
		}
		modifier = {
			add = 0.2
			has_trait = patient
		}
		modifier = {
			add = -0.2
			has_trait = impatient
		}
	}
	
	option = { # Support the mother
		name = VIETmisc.0400.a
		add_character_modifier = {
			modifier = VIET_sympathetic_to_mothers
			years = 5
		}
	}	
	
	option = { # Support the other woman
		name = VIETmisc.0400.b
		add_character_modifier = {
			modifier = VIET_suspicious_of_hystical_women
			years = 5
		}
	}	
	
	option = { # GTFO
		name = VIETmisc.0400.c
		VIET_huge_prestige_gain_effect = yes
		stress_impact = {
			just = miniscule_stress_impact_gain
			patient = miniscule_stress_impact_gain
			calm = miniscule_stress_impact_gain
		}
	}	


	after = {		
		if = {
			limit = {
				has_game_rule = VIET_historical_context_on
			}
			custom_tooltip = VIETmisc.0400.info
		}
	}

}

# Female courtier enjoys bathing in white milk / milk bather
VIETmisc.0410 = {
	type = character_event
	title = VIETmisc.0410.t
	desc = VIETmisc.0410.desc
	theme = seduction
	override_background = { reference = relaxing_room }


	left_portrait = {
		character = root
		animation = personality_cynical
	}

	right_portrait = {
		character = scope:milk_bather
		animation = eyeroll
	}

	trigger = {	
		OR = {
			any_courtier_or_guest = {
				is_available_ai_adult = yes
				is_female = yes
				VIET_is_cheese_culture_trigger = yes # Won't happen to cultures where milk isn't as important
				NOR = {
					has_trait = temperate
					has_trait = humble
					has_trait = reclusive
					probably_unintelligent_trigger = yes
					diplomacy <= low_skill_rating
					learning <= low_skill_rating
				}
			}
			any_vassal = {
				is_available_ai_adult = yes
				is_female = yes
				VIET_is_cheese_culture_trigger = yes # Won't happen to cultures where milk isn't as important
				NOR = {
					has_trait = temperate
					has_trait = humble
					has_trait = reclusive
					probably_unintelligent_trigger = yes
					diplomacy <= low_skill_rating
					learning <= low_skill_rating
				}
			}
		}
	}
	
	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.2
			has_trait = zealous
		}
		modifier = {
			add = -0.2
			has_trait = cynical
		}
		modifier = {
			add = 0.2
			has_trait = honest
		}
		modifier = {
			add = -0.2
			has_trait = deceitful
		}
		modifier = {
			add = 0.2
			has_trait = fickle
		}
		modifier = {
			add = -0.2
			has_trait = stubborn
		}
	}
	
	immediate = {
		every_courtier_or_guest = {
			limit = {
				is_available_ai_adult = yes
				is_female = yes
				VIET_is_cheese_culture_trigger = yes # Won't happen to cultures where milk isn't as important
				NOR = {
					has_trait = temperate
					has_trait = humble
					has_trait = reclusive
					probably_unintelligent_trigger = yes
					diplomacy <= low_skill_rating
					learning <= low_skill_rating
				}
			}
			add_to_list = candidate_characters
		}
		every_vassal = {
			limit = {
				is_available_ai_adult = yes
				is_female = yes
				VIET_is_cheese_culture_trigger = yes # Won't happen to cultures where milk isn't as important
				NOR = {
					has_trait = temperate
					has_trait = humble
					has_trait = reclusive
					probably_unintelligent_trigger = yes
					diplomacy <= low_skill_rating
					learning <= low_skill_rating
				}
			}
			add_to_list = candidate_characters
		}
		random_in_list = {
			list = candidate_characters
			save_scope_as = milk_bather
			add_character_modifier = {
				modifier = VIET_milk_bather
				years = 10
			}
		}
	}

	option = { # Let her be
		name = VIETmisc.0410.a
		remove_short_term_gold = 25
		stress_impact = {
			temperate = minor_stress_impact_gain
			humble = miniscule_stress_impact_gain
		}
	}	

	option = { # Convince her to stop
		name = VIETmisc.0410.b
		VIET_huge_piety_gain_effect = yes
		duel = {
			skill = diplomacy
			target = scope:milk_bather
			50 = { # Failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -1.1
					min = -49
				}
				desc = VIETmisc.0410.b.toast_bad
				send_interface_toast = {
					type = event_toast_effect_bad
					left_icon = scope:milk_bather
					title = VIETmisc.0410.b.toast_bad
					remove_short_term_gold = 25
					scope:milk_bather = {
						add_opinion = {
							target = root
							modifier = VIET_opinion_chastised_me_in_private
						}				
						reverse_add_opinion = {
							modifier = disappointed_opinion
							target = root
							opinion = -10
						}
					}
				}
			}
			50 = { # Success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1.1
					min = -49
				}
				desc = VIETmisc.0410.b.toast_good
				send_interface_toast = {
					type = event_toast_effect_good
					left_icon = scope:milk_bather
					title = VIETmisc.0410.b.toast_good
					scope:milk_bather = {
						add_opinion = {
							target = root
							modifier = VIET_opinion_good_advice
						}
						reverse_add_opinion = {
							target = root
							modifier = VIET_opinion_listened_to_my_advice
						}
					}
				}
			}
		}
		stress_impact = {
			content = miniscule_stress_impact_gain
			shy = miniscule_stress_impact_gain
		}
	}	

	option = { # Start doing it yourself
		name = VIETmisc.0410.c
		trigger = {
			OR = {
				gold >= 50
				is_ai = no
			}
		}
		remove_short_term_gold = 50
		add_character_modifier = {
			modifier = VIET_milk_bather
			years = 10
		}
		stress_impact = {
			greedy = minor_stress_impact_gain
			temperate = minor_stress_impact_gain
			content = miniscule_stress_impact_gain
			humble = miniscule_stress_impact_gain
		}
	}	


}

		animation = personality_cynical
	}

			}
		}
	}
	
	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.2
			has_trait = trusting
		}
		modifier = {
			add = -0.2
			has_trait = paranoid
		}
		modifier = {
			add = 0.2
			has_trait = humble
		}
		modifier = {
			add = -0.2
			has_trait = arrogant
		}
		modifier = {
			add = 0.2
			has_trait = fickle
		}
		modifier = {
			add = -0.2
			has_trait = stubborn
		}
	}
	
	immediate = {
		capital_county = {
			save_scope_as = target_county
		}
		create_character = {
			location = root.location
			template = VIET_generic_middle_aged_local_template
			culture = scope:target_county.culture
			faith = scope:target_county.faith
			save_scope_as = peasant
		}		
	}

	option = { # Ok
		name = VIETmisc.0415.a
		trigger = {
			root.faith.religion = scope:peasant.faith.religion
		}
		VIET_large_piety_gain_effect = yes
		stress_impact = {
			cynical = minor_stress_impact_gain
			arrogant = miniscule_stress_impact_gain
			paranoid = miniscule_stress_impact_gain
		}
	}	

	option = { # Ok
		name = VIETmisc.0415.b
		trigger = {
			NOT = { root.faith.religion = scope:peasant.faith.religion }
		}
		VIET_large_piety_gain_effect = yes
		stress_impact = {
			cynical = minor_stress_impact_gain
			arrogant = miniscule_stress_impact_gain
			paranoid = miniscule_stress_impact_gain
		}
	}	

	option = { # Nah
		name = VIETmisc.0415.c
		add_character_modifier = {
			modifier = VIET_skeptical_ruler
			days = 800
		}
		stress_impact = {
			cynical = minor_stress_impact_loss
			zealous = minor_stress_impact_gain
		}
	}	

	after = {
		scope:peasant ?= { silent_disappearance_effect = yes }
	}

}

			}
		}

			arrogant = minor_stress_impact_gain
			lazy = miniscule_stress_impact_gain

		}
	}	


}
