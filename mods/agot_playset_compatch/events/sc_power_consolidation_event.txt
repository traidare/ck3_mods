# ============================================================
# EVENT: PICK VASSAL TO VISIT
# ============================================================
sc_power_consolidation.0001 = {
	type = character_event
	title = sc_power_consolidation.0001.t
	theme = diplomacy

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { root = scope:crisis_heir }
				desc = sc_power_consolidation.0001.desc.heir
			}
			desc = sc_power_consolidation.0001.desc.claimant
		}
	}

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:consolidation_vassal_1
		animation = chancellor
	}
	lower_left_portrait = scope:consolidation_vassal_2
	lower_right_portrait = scope:consolidation_vassal_3

	trigger = {
		OR = {
			exists = scope:current_ruler
			root = {
				any_claim = {
					exists = holder
					tier >= tier_kingdom
				}
			}
			root = {
				exists = father
				father = {
					primary_title = {
						exists = holder
					}
				}
			}
		}
		NOT = { has_character_flag = already_asked }
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:current_ruler } }
			root = {
				random_claim = {
					limit = {
						exists = holder
						tier >= tier_kingdom
					}
					holder = { save_scope_as = current_ruler }
				}
			}
		}

		# Fallback to father's primary title
		if = {
			limit = { NOT = { exists = scope:current_ruler } }
			root = {
				if = {
					limit = {
						exists = father.primary_title
						father.primary_title.tier >= tier_kingdom
					}
					father = { save_scope_as = current_ruler }
				}
			}
		}

		scope:current_ruler = {
			every_vassal = {
				limit = {
					highest_held_title_tier > tier_county
					NOT = { has_character_flag = pledged_to_heir }
					NOT = { has_character_flag = pledged_to_claimant }
					NOT = { this = scope:crisis_heir }
					NOT = { scope:crisis_special_character ?= this }
					NOT = { this = root }
					NOT = { has_character_flag = having_visitor }
					NOT = { has_character_flag = visited_recently }
					is_travelling = no
				}
				add_to_list = unpledged_vassals
			}
			random_in_list = {
				list = unpledged_vassals
				save_scope_as = consolidation_vassal_1
			}
			random_in_list = {
				list = unpledged_vassals
				limit = { NOT = { this = scope:consolidation_vassal_1 } }
				save_scope_as = consolidation_vassal_2
			}
			random_in_list = {
				list = unpledged_vassals
				limit = {
					NOT = { this = scope:consolidation_vassal_1 }
					NOT = { this = scope:consolidation_vassal_2 }
				}
				save_scope_as = consolidation_vassal_3
			}
		}
	}

	option = {
		name = sc_power_consolidation.0001.a
		trigger = { exists = scope:consolidation_vassal_1 }
		ai_chance = {
			base = 33
		}
		add_character_flag = visiting_vassal
		scope:consolidation_vassal_1 = {
			add_character_flag = having_visitor
			add_character_flag = {
				flag = visited_recently
				years = 1
			}
			save_scope_as = selected_consolidation_vassal
			location = { save_scope_as = target_destination }
		}
		trigger_event = { id = succession_crisis_travel.0003 }
	}

	option = {
		name = sc_power_consolidation.0001.b
		trigger = { exists = scope:consolidation_vassal_2 }
		ai_chance = {
			base = 33
		}
		add_character_flag = visiting_vassal
		scope:consolidation_vassal_2 = {
			add_character_flag = having_visitor
			add_character_flag = {
				flag = visited_recently
				years = 1
			}
			save_scope_as = selected_consolidation_vassal
			location = { save_scope_as = target_destination }
		}
		trigger_event = { id = succession_crisis_travel.0003 }
	}

	option = {
		name = sc_power_consolidation.0001.c
		trigger = { exists = scope:consolidation_vassal_3 }
		ai_chance = {
			base = 33
		}
		add_character_flag = visiting_vassal
		scope:consolidation_vassal_3 = {
			add_character_flag = having_visitor
			add_character_flag = {
				flag = visited_recently
				years = 1
			}
			save_scope_as = selected_consolidation_vassal
			location = { save_scope_as = target_destination }
		}
		trigger_event = { id = succession_crisis_travel.0003 }
	}

	option = {
		name = sc_power_consolidation.0001.d
		trigger = { exists = scope:consolidation_vassal_1 }
		ai_chance = {
			base = 0
		}
		custom_tooltip = sc_power_consolidation.0001.d.tt
		add_character_flag = player_not_interested
	}

	option = {
		name = sc_power_consolidation.0001.e
		trigger = { exists = scope:consolidation_vassal_1 }
		ai_chance = {
			base = 0
		}
		custom_tooltip = sc_power_consolidation.0001.e.tt
		add_character_flag = {
			flag = already_asked
			months = 11
		}
		trigger_event = {
			id = sc_power_consolidation.0001
			months = 12
		}
	}

	option = {
		name = sc_power_consolidation.0001.f
		trigger = {
			NOT = { exists = scope:consolidation_vassal_1 }
		}
		custom_tooltip = sc_power_consolidation.0001.f.tt
		ai_chance = {
			base = 100
		}
	}
}
