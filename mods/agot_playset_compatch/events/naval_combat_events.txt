naval_combat.0110 = {
	type = character_event
	hidden = yes

	immediate = {
		remove_variable = naval_combat_monthly_maintenance_scheduled
		if = {
			limit = {
				exists = var:naval_combat_initialized
				naval_combat_can_manage_navy_trigger = yes
			}
			if = {
				limit = {
					current_day = 1
				}
				naval_combat_collect_monthly_maintenance_effect = yes
				naval_combat_monthly_drill_decay_effect = yes
				if = {
					limit = { is_ai = yes }
					naval_combat_ai_manage_flagship_effect = yes
				}
				set_variable = {
					name = naval_combat_monthly_maintenance_scheduled
					value = yes
				}
				# Twenty-eight days is safe in every month; daily checks resume near month end.
				trigger_event = { id = naval_combat.0110 days = 28 }
			}
			else = {
				naval_combat_schedule_monthly_maintenance_effect = yes
			}
		}
	}
}

	type = character_event
	window = agot_dragon_character_event

	desc = naval_combat.0210.desc
	theme = dragon
	override_background = { reference = fp1_ocean }

		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = var:si_selected_character
		animation = personality_bold
	}

	widget = {
		is_shown = {
			any_in_list = {
				variable = possible_characters
				count > 1
			}
		}
		gui = "agot_character_selection_three_options"
		container = "custom_widgets_container"
	}

	immediate = {
		naval_combat_battle_build_dragonrider_list_effect = yes
		ordered_in_list = {
			variable = possible_characters
			position = 0
			order_by = { value = var:current_dragon.dragon_combat_effectiveness }
			save_scope_as = naval_combat_default_dragonrider
			root = { set_variable = { name = si_selected_character value = scope:naval_combat_default_dragonrider } }
		}
	}

	option = {
		name = naval_combat.0210.a
		trigger = { exists = var:si_selected_character }
		custom_tooltip = naval_combat.0210.a.tt
			hidden_effect = {
				save_scope_as = naval_combat_dragon_duel_root
				var:si_selected_character = { save_scope_as = naval_combat_root_dragonrider }
			naval_combat_battle_assign_dragonrider_effect = {
				RIDER = scope:naval_combat_root_dragonrider
			}
			set_variable = { name = naval_combat_battle_dragon_support_requested value = 1 }
			clear_variable_list = possible_characters
			remove_variable = si_selected_character

			var:naval_combat_battle_opponent = {
				save_scope_as = naval_combat_battle_opponent
				naval_combat_battle_choose_ai_dragonrider_effect = yes
				if = {
					limit = { exists = var:naval_combat_battle_dragonrider }
					var:naval_combat_battle_dragonrider = {
						save_scope_as = naval_combat_enemy_dragonrider
					}
				}
			}

			if = {
				limit = { exists = scope:naval_combat_enemy_dragonrider }
				scope:naval_combat_battle_opponent = {
					save_scope_as = naval_combat_dragon_duel_opponent
				}
				set_variable = { name = naval_combat_battle_dragon_duel_occurred value = 1 }
				scope:naval_combat_battle_opponent = {
					set_variable = { name = naval_combat_battle_dragon_duel_occurred value = 1 }
				}
				var:naval_combat_battle_dragonrider = {
					save_scope_as = dragonrider_in_battle
					var:current_dragon = { save_scope_as = dragon_in_battle }
				}
				scope:naval_combat_enemy_dragonrider = {
					save_scope_as = enemy_dragonrider
