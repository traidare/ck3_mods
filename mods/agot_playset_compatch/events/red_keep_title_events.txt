				save_scope_as = rk_envoy_crack

					trigger_event = {
						id = rk_crackclaw.0003
						months = 2
					}
				}

						id = rk_crackclaw.0013
						months = 2
					}
				}

				name = rk_sent_envoy_1
				years = 1
			}
		}
	}
	option = {  #Send X, the Clawman
		name = rk_crackclaw.0002.b
		custom_tooltip = rk_crackclaw.0002.b_tt
		hidden_effect = {
			scope:rk_envoy_2 = {
				save_scope_as = rk_envoy_crack
			}

				60 = {
					trigger_event = {
						id = rk_crackclaw.0003
						months = 2
					}
