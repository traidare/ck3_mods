# Recalculates combat power and terror whenever body size changes.
kraken_refresh_derived_statistics_effect = {
	if = {
		limit = { age < 30 }
		set_variable = {
			name = kraken_base_size
			value = {
				value = age
				multiply = 5
				add = 20
			}
		}
	}
	else = {
		set_variable = {
			name = kraken_base_size
			value = {
				value = age
				add = 140
			}
		}
	}
	if = {
		limit = { has_variable = kraken_base_size }
		set_variable = {
			name = kraken_size
			value = var:kraken_base_size
		}
		if = {
			limit = { has_trait = kraken_physique_good_1 }
			change_variable = {
				name = kraken_size
				multiply = 1.1
			}
		}
		else_if = {
			limit = { has_trait = kraken_physique_good_2 }
			change_variable = {
				name = kraken_size
				multiply = 1.2
			}
		}
		else_if = {
			limit = { has_trait = kraken_physique_good_3 }
			change_variable = {
				name = kraken_size
				multiply = 1.3
			}
		}
		set_variable = {
			name = kraken_combat_effectiveness
			value = {
				value = var:kraken_size
				multiply = 0.75
				round = yes
			}
		}
		set_variable = {
			name = kraken_dread
			value = {
				value = var:kraken_size
				multiply = 0.5
				round = yes
			}
		}

		# Temporary wounds are mutually exclusive and always take precedence.
		if = {
			limit = { has_character_modifier = kraken_wound_light }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.9 }
		}
		else_if = {
			limit = { has_character_modifier = kraken_wound_injured }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.7 }
		}
		else_if = {
			limit = { has_character_modifier = kraken_wound_severe }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.4 }
			change_variable = { name = kraken_dread multiply = 0.8 }
		}
		else_if = {
			limit = { has_character_modifier = kraken_wound_near_death }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.2 }
			change_variable = { name = kraken_dread multiply = 0.6 }
		}

		# Permanent injuries stack because each records a different struck part.
		if = {
			limit = { has_character_modifier = kraken_injury_one_eye }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.9 }
			change_variable = { name = kraken_dread multiply = 0.9 }
		}
		if = {
			limit = { has_character_modifier = kraken_injury_severed_tentacle }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.85 }
		}
		if = {
			limit = { has_character_modifier = kraken_injury_broken_beak }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.9 }
		}
		if = {
			limit = { has_character_modifier = kraken_injury_torn_mantle }
			change_variable = { name = kraken_combat_effectiveness multiply = 0.9 }
		}
		set_variable = {
			name = kraken_combat_effectiveness
			value = {
				value = var:kraken_combat_effectiveness
				round = yes
			}
		}
		set_variable = {
			name = kraken_dread
			value = {
				value = var:kraken_dread
				round = yes
			}
		}

		set_variable = {
			name = kraken_target_prowess
			value = {
				value = var:kraken_size
				multiply = 0.45
				round = yes
			}
		}
		add_prowess_skill = {
			value = var:kraken_target_prowess
			subtract = prowess
		}
		remove_variable = kraken_target_prowess
	}
}

# Gives an existing kraken a valid sea lair in the Sunset Sea and anchors it
# there. AGOT's helper returns its chosen province as scope:oceanprovince.
kraken_assign_sunset_sea_lair_effect = {
	agot_random_province_in_is_sunset_sea = yes
	if = {
		limit = { exists = scope:oceanprovince }
		set_variable = {
			name = kraken_lair
			value = scope:oceanprovince
		}
		set_location = {
			location = scope:oceanprovince
			stick_to_location = yes
		}
	}
}

# Moves the creature exactly one province through connected sea zones. The
# saved lair remains its emergency return point rather than its current home.
kraken_move_to_neighboring_sea_effect = {
	if = {
		limit = {
			location = {
				kraken_is_traversable_sea_trigger = yes
				any_neighboring_province = {
					kraken_is_traversable_sea_trigger = yes
				}
			}
		}
		location = {
			random_neighboring_province = {
				limit = { kraken_is_traversable_sea_trigger = yes }
				save_scope_as = kraken_migration_destination
			}
		}
	}
	if = {
		limit = { exists = scope:kraken_migration_destination }
		set_location = {
			location = scope:kraken_migration_destination
			stick_to_location = yes
		}
		set_variable = {
			name = kraken_last_sea_location
			value = scope:kraken_migration_destination
		}
	}
}

# Return a displaced kraken to its most recent sea cell. The original lair is
# only a fallback for old saves. Court-pool changes are deliberately avoided
# here because they can recursively fire character lifecycle on-actions.
kraken_return_to_last_sea_effect = {
	if = {
		limit = {
			has_variable = kraken_last_sea_location
			var:kraken_last_sea_location = {
				kraken_is_traversable_sea_trigger = yes
			}
		}
		set_location = {
			location = var:kraken_last_sea_location
			stick_to_location = yes
		}
	}
	else_if = {
		limit = {
			has_variable = kraken_lair
			var:kraken_lair = {
				kraken_is_traversable_sea_trigger = yes
			}
		}
		set_location = {
			location = var:kraken_lair
			stick_to_location = yes
		}
		set_variable = {
			name = kraken_last_sea_location
			value = var:kraken_lair
		}
	}
}

# Keeps exactly one travel-danger modifier attached to the creature's current
# sea province. The saved province also lets us clean up after external moves.
kraken_refresh_presence_danger_effect = {
	save_scope_as = kraken_presence_character
	if = {
		limit = { has_variable = kraken_danger_province }
		var:kraken_danger_province = {
			remove_province_modifier = kraken_presence_travel_danger
			remove_province_modifier = hungry_kraken_presence_travel_danger
		}
		remove_variable = kraken_danger_province
	}
	if = {
		limit = {
			is_alive = yes
			location = { is_sea_province = yes }
		}
		location = {
			remove_province_modifier = kraken_presence_travel_danger
			remove_province_modifier = hungry_kraken_presence_travel_danger
			if = {
				limit = {
					scope:kraken_presence_character = {
						has_variable = kraken_hunger
						var:kraken_hunger < 30
					}
				}
				add_province_modifier = hungry_kraken_presence_travel_danger
			}
			else = {
				add_province_modifier = kraken_presence_travel_danger
			}
			save_scope_as = kraken_presence_province
		}
		set_variable = {
			name = kraken_danger_province
			value = scope:kraken_presence_province
		}
	}
}

# Iron and Salt compatibility. This only reads its public ship variables, so
# the Kraken experiment remains loadable when the naval mod is disabled.
kraken_lose_random_naval_ships_effect = {
	set_variable = { name = kraken_ship_loss value = 0 }
	set_variable = { name = kraken_requested_ship_loss value = 1 }
	random_list = {
		1 = { set_variable = { name = kraken_requested_ship_loss value = 1 } }
		1 = { set_variable = { name = kraken_requested_ship_loss value = 2 } }
		1 = { set_variable = { name = kraken_requested_ship_loss value = 3 } }
		1 = { set_variable = { name = kraken_requested_ship_loss value = 4 } }
		1 = { set_variable = { name = kraken_requested_ship_loss value = 5 } }
	}
	while = {
		limit = {
			var:kraken_requested_ship_loss > 0
			OR = {
				var:naval_combat_transport_ships > 0
				var:naval_combat_oared_warships > 0
				var:naval_combat_high_sided_warships > 0
				var:naval_combat_longships > 0
				var:naval_combat_greek_fire_ships > 0
				var:naval_combat_tower_ships > 0
				var:naval_combat_slave_galleys > 0
			}
		}
		random_list = {
			100 = {
				trigger = { var:naval_combat_transport_ships > 0 }
				change_variable = { name = naval_combat_transport_ships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_oared_warships > 0 }
				change_variable = { name = naval_combat_oared_warships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_high_sided_warships > 0 }
				change_variable = { name = naval_combat_high_sided_warships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_longships > 0 }
				change_variable = { name = naval_combat_longships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_greek_fire_ships > 0 }
				change_variable = { name = naval_combat_greek_fire_ships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_tower_ships > 0 }
				change_variable = { name = naval_combat_tower_ships subtract = 1 }
			}
			100 = {
				trigger = { var:naval_combat_slave_galleys > 0 }
				change_variable = { name = naval_combat_slave_galleys subtract = 1 }
			}
		}
		change_variable = { name = kraken_requested_ship_loss subtract = 1 }
		change_variable = { name = kraken_ship_loss add = 1 }
	}
	remove_variable = kraken_requested_ship_loss
}

# Selects a local coastal holder first, then weights that ruler and successive
# lieges at 60/25/10/5. Merchant shipping is common; naval attacks are rarer.
kraken_attack_shipping_effect = {
	save_scope_as = kraken
	location = {
		every_neighboring_province = {
			limit = {
				is_sea_province = no
				exists = county
				county = { exists = holder }
			}
			county.holder = {
				if = {
					limit = {
						is_alive = yes
						is_landed = yes
						NOT = { is_in_list = kraken_local_shipping_rulers }
					}
					add_to_temporary_list = kraken_local_shipping_rulers
				}
			}
		}
		if = {
			limit = {
				any_in_list = {
					list = kraken_local_shipping_rulers
					count > 0
				}
			}
			random_in_list = {
				list = kraken_local_shipping_rulers
				save_scope_as = kraken_local_shipping_ruler
			}
		}
	}
	if = {
		limit = { exists = scope:kraken_local_shipping_ruler }
		scope:kraken_local_shipping_ruler = {
			if = {
				limit = { exists = liege }
				liege = { save_scope_as = kraken_shipping_liege_1 }
			}
			if = {
				limit = { exists = liege.liege }
				liege.liege = { save_scope_as = kraken_shipping_liege_2 }
			}
			if = {
				limit = { exists = liege.liege.liege }
				liege.liege.liege = { save_scope_as = kraken_shipping_liege_3 }
			}
		}
		random_list = {
			60 = {
				scope:kraken_local_shipping_ruler = {
					save_scope_as = kraken_shipping_victim
				}
			}
			25 = {
				trigger = { exists = scope:kraken_shipping_liege_1 }
				scope:kraken_shipping_liege_1 = {
					save_scope_as = kraken_shipping_victim
				}
			}
			10 = {
				trigger = { exists = scope:kraken_shipping_liege_2 }
				scope:kraken_shipping_liege_2 = {
					save_scope_as = kraken_shipping_victim
				}
			}
			5 = {
				trigger = { exists = scope:kraken_shipping_liege_3 }
				scope:kraken_shipping_liege_3 = {
					save_scope_as = kraken_shipping_victim
				}
			}
		}
	}
	if = {
		limit = { exists = scope:kraken_shipping_victim }
		scope:kraken_shipping_victim = {
			random_list = {
				80 = {
					remove_short_term_gold = 100
					add_prestige = -50
					set_variable = { name = kraken_ship_loss value = 0 }
					trigger_event = kraken.0200
				}
				20 = {
					trigger = {
						has_variable = naval_combat_initialized
						OR = {
							var:naval_combat_transport_ships > 0
							var:naval_combat_oared_warships > 0
							var:naval_combat_high_sided_warships > 0
							var:naval_combat_longships > 0
							var:naval_combat_greek_fire_ships > 0
							var:naval_combat_tower_ships > 0
							var:naval_combat_slave_galleys > 0
						}
					}
					kraken_lose_random_naval_ships_effect = yes
					trigger_event = kraken.0201
				}
			}
		}
		set_variable = { name = kraken_hunger value = 100 }
		kraken_refresh_presence_danger_effect = yes
	}
}

kraken_monthly_hunger_and_hunt_effect = {
	if = {
		limit = { NOT = { has_variable = kraken_hunger } }
		set_variable = { name = kraken_hunger value = 100 }
	}
	else_if = {
		limit = { var:kraken_hunger >= 10 }
		change_variable = { name = kraken_hunger subtract = 10 }
	}
	else = {
		set_variable = { name = kraken_hunger value = 0 }
	}
	kraken_refresh_presence_danger_effect = yes
	if = {
		limit = { has_character_modifier = enraged_kraken }
		if = {
			limit = { has_character_modifier = kraken_wound_near_death }
			remove_character_modifier = enraged_kraken
		}
		else = {
			kraken_attack_shipping_effect = yes
		}
	}
	else_if = {
		limit = { var:kraken_hunger < 30 }
		random = {
			chance = 25
			kraken_attack_shipping_effect = yes
		}
	}
}

}

# Creates one fully initialized creature at scope:kraken_spawn_location.
kraken_create_at_saved_location_effect = {
	create_character = {
		location = scope:kraken_spawn_location
		template = kraken_creature_template
		random_traits = no
		save_scope_as = kraken
	}

	scope:kraken ?= {
		random_list = {
			1 = { change_first_name = kraken_name_the_deep_one }
			1 = { change_first_name = kraken_name_drowned_gods_wrath }
			1 = { change_first_name = kraken_name_black_tide }
			1 = { change_first_name = kraken_name_shipbreaker }
			1 = { change_first_name = kraken_name_grip_of_the_abyss }
			1 = { change_first_name = kraken_name_sailsplitter }
		}

		# Exceptional physiques are rarer; every generated creature gets one tier.
		random_list = {
			30 = { add_trait = kraken_physique_good_1 }
			50 = { add_trait = kraken_physique_good_2 }
			20 = { add_trait = kraken_physique_good_3 }
		}

		set_variable = {
			name = kraken_base_size
			value = 80
		}
		set_variable = {
			name = kraken_hunger
			value = 100
		}
		set_variable = {
			name = kraken_lair
			value = scope:kraken_spawn_location
		}
		set_variable = {
			name = kraken_last_sea_location
			value = scope:kraken_spawn_location
		}
		set_location = {
			location = scope:kraken_spawn_location
			stick_to_location = yes
		}
		kraken_refresh_derived_statistics_effect = yes
		kraken_refresh_presence_danger_effect = yes

		create_story = story_kraken_alive
	}

	add_to_global_variable_list = {
		name = living_krakens
		target = scope:kraken
	}
	kraken_recount_living_population_effect = yes

	if = {
		limit = { NOT = { exists = global_var:kraken_test_creature } }
		set_global_variable = {
			name = kraken_test_creature
			value = scope:kraken
		}
	}
	else_if = {
		limit = { global_var:kraken_test_creature = { is_alive = no } }
		set_global_variable = {
			name = kraken_test_creature
			value = scope:kraken
		}
	}
}

# Creates the debug creature beside the acting character's capital, falling
# back to a random Sunset Sea province for inland capitals.
kraken_spawn_test_creature_effect = {
	save_scope_as = kraken_witness
	capital_province = {
		if = {
			limit = {
				any_neighboring_province = {
					kraken_is_traversable_sea_trigger = yes
				}
			}
			random_neighboring_province = {
				limit = { kraken_is_traversable_sea_trigger = yes }
				save_scope_as = kraken_spawn_location
			}
		}
	}
	if = {
		limit = { NOT = { exists = scope:kraken_spawn_location } }
		agot_random_province_in_is_sunset_sea = yes
		scope:oceanprovince = {
			save_scope_as = kraken_spawn_location
		}
	}

	kraken_create_at_saved_location_effect = yes

	trigger_event = kraken.0001
}

# Rebuilds the count from the authoritative living-creature list. The list is
# maintained by each creature's story cycle and also supports existing saves.
kraken_recount_living_population_effect = {
	set_global_variable = {
		name = kraken_living_count
		value = 0
	}
	if = {
		limit = { has_global_variable_list = living_krakens }
		every_in_global_list = {
			variable = living_krakens
			limit = {
				is_alive = yes
				has_trait = kraken
			}
			change_global_variable = {
				name = kraken_living_count
				add = 1
			}
		}
	}
}

# Draws a creature to the summoner's nearby waters. This starts from the
# current location so landless artifact holders work without owning a realm.
kraken_spawn_enraged_near_character_effect = {

		limit = {
			location = { kraken_is_traversable_sea_trigger = yes }
		}
		location = { save_scope_as = kraken_spawn_location }
	}
	else_if = {
		limit = {
			location = {
				any_neighboring_province = {
					kraken_is_traversable_sea_trigger = yes
				}
			}
		}
		location = {
			random_neighboring_province = {
				limit = { kraken_is_traversable_sea_trigger = yes }
				save_scope_as = kraken_spawn_location
			}
		}

			location = {
				exists = county

					any_county_province = {
						any_neighboring_province = {
							kraken_is_traversable_sea_trigger = yes
						}

				}
			}
		}

		limit = {
			is_landed = yes
			capital_province = {
				any_neighboring_province = {
					kraken_is_traversable_sea_trigger = yes
				}
			}
		}

				limit = { kraken_is_traversable_sea_trigger = yes }
				save_scope_as = kraken_spawn_location
			}
		}
	}

		limit = {
			is_landed = yes
			any_held_title = {
				tier = tier_county
				any_county_province = {
					any_neighboring_province = {
						kraken_is_traversable_sea_trigger = yes
					}
				}
			}
		}
		random_held_title = {
			limit = {
				tier = tier_county
				any_county_province = {
					any_neighboring_province = {
						kraken_is_traversable_sea_trigger = yes
					}
				}
			}
			random_county_province = {
				limit = {
					any_neighboring_province = {
						kraken_is_traversable_sea_trigger = yes
					}
				}
				random_neighboring_province = {
					limit = { kraken_is_traversable_sea_trigger = yes }

				}
			}
		}
	}
	else_if = {
		limit = {
			is_landed = yes
			any_realm_county = {

					any_neighboring_province = {
						kraken_is_traversable_sea_trigger = yes

				}
