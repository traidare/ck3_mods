naval_combat_initialize_navy_effect = {
	if = {
		limit = {
			NOT = { exists = var:naval_combat_initialized }
		}
		remove_variable = naval_combat_navy_handed_off
		set_variable = {
			name = naval_combat_initialized
			value = yes
		}
		set_variable = {
			name = naval_combat_transport_ships
			value = 0
		}
		set_variable = {
			name = naval_combat_oared_warships
			value = 0
		}
		set_variable = {
			name = naval_combat_high_sided_warships
			value = 0
		}
		set_variable = {
			name = naval_combat_longships
			value = 0
		}
		set_variable = {
			name = naval_combat_greek_fire_ships
			value = 0
		}
		set_variable = {
			name = naval_combat_tower_ships
			value = 0
		}
		set_variable = {
			name = naval_combat_readiness
			value = 0
		}
		set_variable = {
			name = naval_combat_drill
			value = 0
		}
		set_variable = {
			name = naval_combat_transport_active
			value = 0
		}
		set_variable = {
			name = naval_combat_sea_control_active
			value = 0
		}
		naval_combat_shipwright_grant_cultural_fittings_effect = yes
	}

	if = {
		limit = {
			is_ai = no
		}
		set_global_variable = {
			name = naval_combat_ai_system_unlocked
			value = yes
		}
	}

	if = {
		limit = {
			naval_combat_has_direct_coastal_county_trigger = yes
		}
		set_variable = {
			name = naval_combat_has_royal_naval_base
			value = yes
		}
	}
	else = {
		remove_variable = naval_combat_has_royal_naval_base
	}

	naval_combat_refresh_transport_modifier_effect = yes
	naval_combat_schedule_weekly_readiness_effect = yes
	naval_combat_schedule_monthly_maintenance_effect = yes
}

naval_combat_ai_update_avoidance_effect = {
	if = {
		limit = {
			is_at_war = yes
			naval_combat_total_combat_power_value > 0
			any_war_enemy = {
				is_alive = yes
				exists = var:naval_combat_initialized
				naval_combat_total_ship_count_value > 0
				var:naval_combat_sea_control_active ?= 1
				var:naval_combat_readiness > 0
				naval_combat_total_combat_power_value >= root.naval_combat_ai_avoidance_threshold_value
			}
		}
		set_variable = { name = naval_combat_ai_avoiding_superior_fleet value = 1 }
	}
	else = {
		set_variable = { name = naval_combat_ai_avoiding_superior_fleet value = 0 }
	}
}

naval_combat_ai_update_orders_effect = {
	if = {
		limit = {
			is_ai = yes
			exists = var:naval_combat_initialized
			NOT = { exists = var:naval_combat_ai_focus }
		}
		naval_combat_ai_update_focus_effect = yes
	}
	naval_combat_ai_update_avoidance_effect = yes
	if = {
		limit = {
			is_ai = yes
			exists = var:naval_combat_initialized
			NOT = { exists = var:naval_combat_battle_opponent }
		}
		if = {
			limit = { naval_combat_total_ship_count_value <= 0 }
			set_variable = { name = naval_combat_transport_active value = 0 }
			set_variable = { name = naval_combat_sea_control_active value = 0 }
		}
		else_if = {
			limit = { var:naval_combat_ai_avoiding_superior_fleet ?= 1 }
			set_variable = { name = naval_combat_transport_active value = 0 }
			set_variable = { name = naval_combat_sea_control_active value = 0 }
			naval_combat_clear_contact_state_effect = yes
		}
		else_if = {
			limit = { is_at_war = yes }
			set_variable = { name = naval_combat_transport_active value = 1 }
			if = {
				limit = {
					var:naval_combat_ai_focus >= 50
					short_term_treasury_or_gold >= 0
				}
				set_variable = { name = naval_combat_sea_control_active value = 1 }
			}
			else = {
				set_variable = { name = naval_combat_sea_control_active value = 0 }
			}
		}
		else_if = {
			limit = { short_term_treasury_or_gold < 0 }

naval_combat_refresh_blockade_effect = {
	if = {
		limit = {
			naval_combat_can_manage_navy_trigger = yes
			is_at_war = yes
			OR = {
				naval_combat_total_ship_count_value <= 0
				AND = {
					NOT = { var:naval_combat_transport_active ?= 1 }
					NOT = { var:naval_combat_sea_control_active ?= 1 }
				}
			}
			any_war_enemy = {
				is_alive = yes
				naval_combat_total_ship_count_value > 0
				var:naval_combat_sea_control_active ?= 1
				var:naval_combat_readiness > 0
			}
		}

		if = {
			limit = {
				exists = var:naval_combat_blockading_enemy
				var:naval_combat_blockading_enemy = {
					is_alive = yes
					naval_combat_total_ship_count_value > 0
					var:naval_combat_sea_control_active ?= 1
					var:naval_combat_readiness > 0
					any_war_enemy = { this = root }
				}
			}
			var:naval_combat_blockading_enemy = {
				save_scope_as = naval_combat_blockading_enemy
			}
		}
		else = {
			random_war_enemy = {
				limit = {
					is_alive = yes
					naval_combat_total_ship_count_value > 0
					var:naval_combat_sea_control_active ?= 1
					var:naval_combat_readiness > 0
				}
				save_scope_as = naval_combat_blockading_enemy
			}
		}
		set_variable = {
			name = naval_combat_blockading_enemy
			value = scope:naval_combat_blockading_enemy
		}

		if = {
			limit = {
				NOT = { has_character_modifier = naval_combat_blockaded_modifier }
			}
			add_character_modifier = { modifier = naval_combat_blockaded_modifier }
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = naval_combat_blockade_event_enemy
					value = scope:naval_combat_blockading_enemy
				}
				trigger_event = naval_combat.0300
			}
		}
		naval_combat_refresh_defeat_penalty_effect = yes
	}
	else = {
		remove_character_modifier = naval_combat_blockaded_modifier
		remove_variable = naval_combat_blockading_enemy
		naval_combat_refresh_defeat_penalty_effect = yes
	}
}

naval_combat_update_contact_effect = {
	if = {
		limit = {
			OR = {
				NOT = { is_at_war = yes }
				NOT = { naval_combat_can_enter_automatic_battle_trigger = yes }
			}
		}
		naval_combat_clear_contact_state_effect = yes
	}
	else = {
		# Drop a stale target before looking for a new opposing fleet.
		if = {
			limit = {
				exists = var:naval_combat_contact_target
				NOT = {
					var:naval_combat_contact_target = {
						is_alive = yes
						naval_combat_can_enter_automatic_battle_trigger = yes
						any_war_enemy = { this = root }
					}
				}
			}
			naval_combat_clear_contact_state_effect = yes
		}

		# One fleet pair is tracked at a time. Human opponents are always processed
		# from the human side so the interactive event cannot resolve off-screen.
		if = {
			limit = { NOT = { exists = var:naval_combat_contact_target } }
			random_war_enemy = {
				limit = {
					is_alive = yes
					naval_combat_can_enter_automatic_battle_trigger = yes
					OR = {
						NOT = { exists = var:naval_combat_contact_target }
						var:naval_combat_contact_target ?= root
					}
					trigger_if = {
						limit = { root = { is_ai = yes } }
						is_ai = yes
					}
				}
				save_scope_as = naval_combat_contact_enemy
			}
			if = {
				limit = { exists = scope:naval_combat_contact_enemy }
				set_variable = {
					name = naval_combat_contact_target
					value = scope:naval_combat_contact_enemy
				}
				set_variable = { name = naval_combat_contact_value value = 0 }
				# The saved iterator scope is weak. Re-enter through the strong
				# character variable before writing the reciprocal contact state.
				var:naval_combat_contact_target = {
					set_variable = { name = naval_combat_contact_target value = root }
					set_variable = { name = naval_combat_contact_value value = 0 }
				}
			}
		}

		if = {
			limit = {
				exists = var:naval_combat_contact_target
				NOT = { has_character_flag = naval_combat_automatic_contact_checked }
				var:naval_combat_contact_target = {
					is_alive = yes
					NOT = { has_character_flag = naval_combat_automatic_contact_checked }
				}
				# A human-vs-AI pair is advanced only from the human side.
				trigger_if = {
					limit = { is_ai = yes }
					var:naval_combat_contact_target = { is_ai = yes }
				}
			}
			var:naval_combat_contact_target = {
				save_scope_as = naval_combat_contact_enemy
			}
			add_character_flag = {
				flag = naval_combat_automatic_contact_checked
				days = 6
			}
			scope:naval_combat_contact_enemy = {
				add_character_flag = {
					flag = naval_combat_automatic_contact_checked
					days = 6
				}
			}

			# Both sides contesting sea control: rapid search and a high encounter chance.
			if = {
				limit = {
					var:naval_combat_sea_control_active ?= 1
					scope:naval_combat_contact_enemy = { var:naval_combat_sea_control_active ?= 1 }
				}
				change_variable = { name = naval_combat_contact_value add = 15 }
				set_variable = { name = naval_combat_contact_type value = 1 }
				set_variable = {
					name = naval_combat_contact_chance
					value = {
						value = var:naval_combat_contact_value
						multiply = 0.30
						add = 5
					}
				}
				if = {
					limit = { var:naval_combat_contact_chance > 40 }
					set_variable = { name = naval_combat_contact_chance value = 40 }
				}
			}
			# A patrol searching for a transport fleet: lower, but persistent interception odds.
			else_if = {
				limit = {
					OR = {
						var:naval_combat_sea_control_active ?= 1
						scope:naval_combat_contact_enemy = { var:naval_combat_sea_control_active ?= 1 }
					}
				}
				change_variable = { name = naval_combat_contact_value add = 6 }
				set_variable = { name = naval_combat_contact_type value = 2 }
				set_variable = {
					name = naval_combat_contact_chance
					value = {
						value = var:naval_combat_contact_value
						multiply = 0.15
						add = 2
					}
				}
				if = {
					limit = { var:naval_combat_contact_chance > 18 }
					set_variable = { name = naval_combat_contact_chance value = 18 }
				}
			}
			# Two transport fleets are avoiding battle, but a rare chance encounter remains.
			else = {
				change_variable = { name = naval_combat_contact_value add = 2 }
				set_variable = { name = naval_combat_contact_type value = 3 }
				set_variable = {
					name = naval_combat_contact_chance
					value = {
						value = var:naval_combat_contact_value
						multiply = 0.05
						add = 0.5
					}
				}
				if = {
					limit = { var:naval_combat_contact_chance > 6 }
					set_variable = { name = naval_combat_contact_chance value = 6 }
				}
			}

			if = {
				limit = { var:naval_combat_contact_value > 100 }
				set_variable = { name = naval_combat_contact_value value = 100 }
			}
			scope:naval_combat_contact_enemy = {
				set_variable = {
					name = naval_combat_contact_value
					value = root.var:naval_combat_contact_value
				}
				set_variable = {
					name = naval_combat_contact_chance
					value = root.var:naval_combat_contact_chance
				}
				set_variable = {
					name = naval_combat_contact_type
					value = root.var:naval_combat_contact_type
				}
			}

			random = {
				chance = var:naval_combat_contact_chance
				if = {
					limit = {
						naval_combat_has_active_fleet_orders_trigger = yes
						scope:naval_combat_contact_enemy = {
							naval_combat_has_active_fleet_orders_trigger = yes
						}
					}
					set_variable = {
						name = naval_combat_battle_opponent
						value = scope:naval_combat_contact_enemy
					}
					set_variable = {
						name = naval_combat_battle_encounter_type
						value = var:naval_combat_contact_type
					}
					add_character_flag = {
						flag = naval_combat_automatic_battle_cooldown
						months = 1
					}
					scope:naval_combat_contact_enemy = {
						add_character_flag = {
							flag = naval_combat_automatic_battle_cooldown
							months = 1
						}
					}
					naval_combat_clear_contact_state_effect = yes
					trigger_event = naval_combat.0200
				}
				else = {
					naval_combat_clear_contact_state_effect = yes
				}
			}
		}
	}
}

naval_combat_ai_manage_flagship_effect = {
	if = {
		limit = {
			is_ai = yes
			var:naval_combat_flagship_constructed ?= 1
			exists = var:naval_combat_flagship_hull
			NOT = { var:naval_combat_flagship_destroyed ?= 1 }
			var:naval_combat_flagship_current_durability > 0
			exists = var:naval_combat_ai_flagship_rebuild_delay_started
		}
		remove_variable = naval_combat_ai_flagship_rebuild_delay_started
	}

	if = {
		limit = {
			is_ai = yes
			var:naval_combat_flagship_constructed ?= 1
			exists = var:naval_combat_flagship_hull
			OR = {
				var:naval_combat_flagship_destroyed ?= 1
				var:naval_combat_flagship_current_durability <= 0
			}
		}
		if = {
			limit = { NOT = { exists = var:naval_combat_ai_flagship_rebuild_delay_started } }
			set_variable = { name = naval_combat_ai_flagship_rebuild_delay_started value = 1 }
			add_character_flag = {
				flag = naval_combat_ai_flagship_rebuild_cooldown
				months = 24
			}
		}
		else_if = {
			limit = {
				NOT = { has_character_flag = naval_combat_ai_flagship_rebuild_cooldown }
				NOT = { is_at_war = yes }
				NOT = { exists = var:naval_combat_battle_opponent }
				short_term_treasury_or_gold >= naval_combat_ai_flagship_rebuild_treasury_threshold_value
			}
			remove_treasury_or_gold = naval_combat_ai_flagship_rebuild_cost_value
			set_variable = {
				name = naval_combat_flagship_current_durability
				value = naval_combat_flagship_active_max_durability_value
			}
			remove_variable = naval_combat_flagship_destroyed
			remove_variable = naval_combat_ai_flagship_rebuild_delay_started
			add_character_flag = {
				flag = naval_combat_ai_flagship_repair_cooldown
				months = 12
			}
		}
	}
	else_if = {
		limit = {
			is_ai = yes
			var:naval_combat_flagship_constructed ?= 1
			exists = var:naval_combat_flagship_hull
			NOT = { var:naval_combat_flagship_destroyed ?= 1 }
			var:naval_combat_flagship_current_durability > 0
			naval_combat_flagship_missing_durability_value >= naval_combat_ai_flagship_repair_minimum_damage_value
			NOT = { has_character_flag = naval_combat_ai_flagship_repair_cooldown }
			NOT = { is_at_war = yes }
			NOT = { exists = var:naval_combat_battle_opponent }
			short_term_treasury_or_gold >= naval_combat_ai_flagship_repair_treasury_threshold_value
		}
		remove_treasury_or_gold = naval_combat_ai_flagship_repair_cost_value
		set_variable = {
			name = naval_combat_flagship_current_durability
			value = naval_combat_flagship_active_max_durability_value
		}
		add_character_flag = {
			flag = naval_combat_ai_flagship_repair_cooldown
			months = 12
		}
	}
}

naval_combat_resolve_training_effect = {
	if = {
		limit = { NOT = { exists = var:naval_combat_training_pre_readiness } }
		set_variable = {
			name = naval_combat_training_pre_readiness
			value = 0
		}
		if = {
			limit = { exists = var:naval_combat_readiness }
			set_variable = {
				name = naval_combat_training_pre_readiness
				value = var:naval_combat_readiness
			}
		}
	}
	set_variable = {
		name = naval_combat_training_pre_resolution_drill
		value = naval_combat_drill_value
	}
	if = {
		limit = { var:naval_combat_training_progress <= 4 }
		set_variable = {
			name = naval_combat_training_result
			value = 15
		}
		naval_combat_add_drill_effect = { AMOUNT = 15 }
	}
	else_if = {
		limit = { var:naval_combat_training_progress <= 7 }
		set_variable = {
			name = naval_combat_training_result
			value = 20
		}
		naval_combat_add_drill_effect = { AMOUNT = 20 }
	}
	else = {
		set_variable = {
			name = naval_combat_training_result
			value = 25
		}
		naval_combat_add_drill_effect = { AMOUNT = 25 }
	}
	set_variable = {
		name = naval_combat_training_actual_gain
		value = naval_combat_drill_value
	}
	change_variable = {
		name = naval_combat_training_actual_gain
		subtract = var:naval_combat_training_pre_resolution_drill
	}
	if = {
		limit = { naval_combat_training_pre_readiness_value >= 30 }
		set_variable = {
			name = naval_combat_readiness
			value = naval_combat_training_pre_readiness_value
		}
		change_variable = {
			name = naval_combat_readiness
			subtract = 30
		}
		set_variable = {
			name = naval_combat_training_actual_readiness_loss
			value = 30
		}
	}
	else = {
		set_variable = {
			name = naval_combat_training_actual_readiness_loss
			value = naval_combat_training_pre_readiness_value
		}
		set_variable = {
			name = naval_combat_readiness
			value = 0
		}
	}
}

			set_variable = {
				name = naval_combat_readiness
				value = 0
			}
		}
		else = {
			change_variable = {
				name = naval_combat_readiness
				subtract = 10
			}
		}
	}
}
