MFA_pulse_apa_joked_together = {
	effect = {
		scope:activity = {
			random_attending_character = {
				limit = {
					location = scope:activity.activity_location
					NOT = { has_trait = callous }
				}
				save_scope_as = first
			}

			random_attending_character = {
				limit = {
					location = scope:activity.activity_location
					NOT = { has_trait = callous }
					this != scope:first
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = apa_joked_together
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_opinion = {
						target = scope:second
						opinion = 10
						modifier = friendliness_opinion
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_friendship
					title = apa_joked_together_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_opinion = {
									target = scope:second
									opinion = 10
									modifier = friendliness_opinion
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_apa_sang_nice_song = {
	effect = {
		scope:activity = {
			if = {
				limit = {
					has_activity_type = activity_playdate
				}
				random_attending_character = {
					limit = {
						location = scope:activity.activity_location
						has_trait = charming
					}
					save_scope_as = first
				}
			}
			else_if = {
				limit = {
					has_activity_type = activity_coronation
					has_current_phase = coronation_phase_feast
				}
				random_attending_character = {
					limit = {
						location = scope:activity.activity_location
						OR = {
							has_trait = lifestyle_poet
							has_trait = violet_poet
							AND = {
								has_trait = tourney_participant
								has_trait_xp = {
									trait = tourney_participant
									track = wit
									value >= 25
								}
							}
						}
					}
					save_scope_as = first
				}
			}

			add_activity_log_entry = {
				key = apa_sang_nice_song
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_prestige = minor_prestige_gain
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_diplomacy
					title = apa_sang_nice_song_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_prestige = minor_prestige_gain
							}
						}
					}
				}
			}
		}
	}
}
