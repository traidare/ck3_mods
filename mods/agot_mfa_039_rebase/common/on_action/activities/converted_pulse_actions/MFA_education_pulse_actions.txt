MFA_pulse_apa_leveraging_language_skills = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
			}
			scope:activity.activity_location = {
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = apa_leveraging_language_skills
				tags = { pulse_action }
				character = scope:first

				scope:activity = { activity_special_type_progression_tiny = yes }

				scope:first = {
					custom_tooltip = adult_education_tt_positive_tiny
					add_prestige = minor_prestige_gain
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = apa_leveraging_language_skills_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:activity = {
								activity_special_type_progression_tiny = yes
							}
							scope:first = {
								custom_tooltip = adult_education_tt_positive_tiny
								add_prestige = minor_prestige_gain
							}
						}
					}
				}
			}
		}
	}
}

						has_character_flag = teacher_1

				save_scope_as = second
			}

				key = apa_university_discussions
