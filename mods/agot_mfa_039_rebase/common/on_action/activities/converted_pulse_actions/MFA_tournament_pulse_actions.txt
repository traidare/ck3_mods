MFA_pulse_tournament_prayed_temple = {
	effect = {
		scope:activity = {
			random_attending_character = {
				limit = {
					location = scope:activity.activity_location
				}
				save_scope_as = first
			}

			add_activity_log_entry = {
				key = tournament_prayed_temple
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_stress = miniscule_stress_impact_loss
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = tournament_prayed_temple_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_stress = miniscule_stress_impact_loss
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_tournament_host_gifted = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
			}
			random_guest_subset = {
				name = contestant
				limit = {
					is_playable_character = yes
					highest_held_title_tier > tier_duchy
					location = scope:activity.activity_location
					opinion = {
						target = scope:host
						value >= low_positive_opinion
					}
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = tournament_host_gifted
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:second = {
					generate_trinket_effect = {
						TRINKET_RECEIVER = scope:first
						TRINKET_GIVER = scope:second
						GRAB_ALL_TRINKETS = yes
						HISTORY_TYPE = given
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = tournament_host_gifted_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:second = {
								generate_trinket_effect = {
									TRINKET_RECEIVER = scope:first
									TRINKET_GIVER = scope:second
									GRAB_ALL_TRINKETS = yes
									HISTORY_TYPE = given
								}
							}

			}
		}
