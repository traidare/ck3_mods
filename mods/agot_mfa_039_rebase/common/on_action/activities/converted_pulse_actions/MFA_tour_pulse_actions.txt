MFA_pulse_courtier_improved_skill = {
	effect = {
		scope:activity = {
			scope:host = {
				random_courtier_or_guest = {
					limit = {
						is_physically_able_ai = yes
						is_participant_in_activity = scope:activity
					}
					weight = {
						base = 1
						modifier = {
							add = 1
							is_player_heir_of = scope:host
						}
						modifier = {
							add = 1
							is_knight_of = scope:host
						}
						modifier = {
							add = 1
							is_councillor_of = scope:host
						}
					}
					save_scope_as = first
				}
			}
			scope:activity = {
				add_activity_log_entry = {
					key = courtier_improved_skill
					tags = { good }
					score = 5
					character = scope:first
					# Effects
					scope:first = {
						random_list = {
							1 = { add_diplomacy_skill = 1 }
							1 = { add_intrigue_skill = 1 }
							1 = { add_learning_skill = 1 }
							1 = { add_martial_skill = 1 }
							1 = { add_prowess_skill = 1 }
							1 = { add_stewardship_skill = 1 }
						}
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = courtier_improved_skill_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								random_list = {
									1 = { add_diplomacy_skill = 1 }
									1 = { add_intrigue_skill = 1 }
									1 = { add_learning_skill = 1 }
									1 = { add_martial_skill = 1 }
									1 = { add_prowess_skill = 1 }
									1 = { add_stewardship_skill = 1 }
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_courtier_learned_language = {
	effect = {
		scope:activity = {
			scope:activity.activity_location.county.holder = {
				save_scope_as = second
			}
			scope:host = {
				random_courtier_or_guest = {
					limit = {
						is_physically_able_ai = yes
						is_participant_in_activity = scope:activity
						NOT = { knows_language_of_culture = scope:activity.activity_location.county.holder.culture }
					}
					weight = {
						base = 1
						modifier = {
							add = learning
						}
					}
					save_scope_as = first
				}
			}
			scope:activity = {
				add_activity_log_entry = {
					key = courtier_learned_language
					tags = { bad }
					score = 5
					character = scope:first
					target = scope:second
					# Effects
					scope:first = {
						learn_language_of_culture = scope:second.culture
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_language
					title = courtier_learned_language_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								learn_language_of_culture = scope:second.culture
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_liege_visited_benevolence = {
	effect = {
		scope:activity = {
			scope:host = { save_scope_as = first }
			scope:host.location.county = {
				save_scope_as = second
				set_variable = { name = recent_dev value = yes days = 60 }
			}
			scope:activity = {
				add_activity_log_entry = {
					key = liege_visited_benevolence
					tags = { good }
					score = 5
					character = scope:first
					# Effects
					if = {
						limit = {
							scope:host.location.county = { uses_county_fertility = yes }
						}
						scope:host.location.county = {
							change_county_fertility = 5
						}
					}
					else = {
						scope:host.location.county = {
							change_development_progress = 10
						}
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_development
					title = liege_visited_benevolence_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							if = {
								limit = {
									scope:host.location.county = { uses_county_fertility = yes }
								}
								scope:host.location.county = {

MFA_pulse_courtiers_became_lovers = {
	effect = {
		scope:activity = {
			scope:host = {
				random_courtier_or_guest = {
					limit = {
						might_cheat_on_every_partner_trigger = yes
						num_of_relation_lover < 2
						is_physically_able_ai_adult = yes
						is_participant_in_activity = scope:activity
						save_temporary_scope_as = first_lover_temp
						scope:activity.activity_location.county.holder = {
							any_courtier_or_guest = {
								this != scope:first_lover_temp
								might_cheat_on_every_partner_trigger = yes
								NOT = { has_relation_lover = scope:first_lover_temp }
								num_of_relation_lover < 2
								is_physically_able_ai_adult = yes
								NOT = { has_relation_lover = scope:first_lover_temp }
								save_temporary_scope_as = second_lover_temp
								matching_gender_and_sexuality_trigger = {
									CHARACTER_1 = scope:second_lover_temp
									CHARACTER_2 = scope:first_lover_temp
								}
							}
						}
					}
					save_scope_as = first
				}
			}
			scope:activity.activity_location.county.holder = {
				random_courtier_or_guest = {
					limit = {
						this != scope:first
						might_cheat_on_every_partner_trigger = yes
						NOT = { has_relation_lover = scope:first }
						num_of_relation_lover < 2
						is_physically_able_ai_adult = yes
						NOT = { has_relation_lover = scope:first }
						save_temporary_scope_as = second_lover_temp

							CHARACTER_1 = scope:second_lover_temp

						base = 1

							factor = 2

								target = scope:first

					save_scope_as = second

					character = scope:first
					target = scope:second

						set_relation_lover = {
							target = scope:first
							reason = lover_courtiers_bonded_on_tour
						}
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_lover

					right_icon = scope:second

MFA_pulse_courtiers_got_into_fight = {
	effect = {
		scope:activity = {
			scope:host = {
				random_courtier = {
					limit = {
						OR = { # Likely to start fights
							has_trait = vengeful
							has_trait = wrathful
							has_trait = arrogant
							has_trait = callous
							has_trait = sadistic
							has_trait = drunkard
						}
						NOT = {
							is_player_heir_of = scope:host
						}
						OR = {
							is_physically_able_ai_adult = yes
							child_not_infant_trigger = yes
						}
						save_temporary_scope_as = first_fighter
						scope:activity.activity_location.county.holder = {
							any_courtier = {
								count >= 2
								OR = {
									is_physically_able_ai_adult = yes
									child_not_infant_trigger = yes
								}
								OR = {
									is_physically_able_ai_adult = yes
									age > scope:first_fighter.age_minus_3
									scope:first_fighter.age > age_minus_3
								}
							}
						}
					}
					save_scope_as = first
				}
			}
			scope:activity.activity_location.county.holder = {
				random_courtier = {
					limit = {
						OR = {
							is_physically_able_ai_adult = yes
							child_not_infant_trigger = yes
						}
						OR = {
							is_physically_able_ai_adult = yes
							age > scope:first.age_minus_3
							scope:first.age > age_minus_3
						}
					}
					weight = {
						base = 1
						modifier = {
							add = ai_vengefulness
						}
						modifier = {
							factor = 2
							trait_compatibility = {
								target = scope:first
								value <= low_negative_trait_compatibility
							}
						}
						modifier = {
							factor = 10

								compare_modifier = {
									 value = scope:duel_value
									 multiplier = 3.5
									 min = -49
								}
								scope:first = {
									increase_wounds_no_death_effect = { REASON = duel }
								}
							}
							50 = {
								compare_modifier = {
									value = scope:duel_value

									min = -49
								}
								increase_wounds_no_death_effect = { REASON = duel }
							}
						}
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_wound
