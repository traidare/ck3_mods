MFA_pulse_contact_network = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			random_attending_character = {
				limit = {
					is_physically_able_adult = yes
					OR = {
						is_ai = no
						this = scope:host
					}
					location = scope:activity.activity_location
					has_lifestyle = diplomacy_lifestyle
					trigger_if = {
						limit = {
							scope:activity = { has_activity_type = activity_wedding }
						}
						OR = {
							this = scope:host
							this = scope:activity.special_guest:spouse_1
							this = scope:activity.special_guest:spouse_2
						}
					}
					trigger_if = {
						limit = {
							scope:activity = { has_activity_type = activity_coronation }
						}
						OR = {
							this = scope:host
							scope:activity.var:officiator ?= this
							is_in_guest_subset = { name = supporter }
							is_in_guest_subset = { name = detractor }
						}
					}
				}
				alternative_limit = {
					is_physically_able_adult = yes
					OR = {
						is_ai = no
						this = scope:host
					}
					location = scope:activity.activity_location
					has_lifestyle = diplomacy_lifestyle
				}
				save_scope_as = first
			}

			add_activity_log_entry = {
				key = contact_network
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_diplomacy_lifestyle_xp = miniscule_lifestyle_xp
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_courtly
					title = contact_network_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_diplomacy_lifestyle_xp = miniscule_lifestyle_xp
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_dynasty_blessing = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			if = {
				limit = {
					scope:activity = {
						has_activity_type = activity_wedding
					}
				}
				random_attending_character = {
					limit = {
						NOR = {
							this = scope:host
							this = scope:activity.special_guest:spouse_1
							this = scope:activity.special_guest:spouse_2
						}
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
						dynasty ?= { dynasty_prestige_level > scope:host.dynasty.dynasty_prestige_level }
					}
					save_scope_as = first
				}
			}

			if = {
				limit = {
					scope:activity = {
						has_activity_type = activity_coronation
					}
				}
				random_attending_character = {
					limit = {
						NOR = {
							this = scope:host
						}
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
						dynasty ?= { dynasty_prestige_level > scope:host.dynasty.dynasty_prestige_level }
					}
					save_scope_as = first
				}
			}

			add_activity_log_entry = {
				key = dynasty_blessing
				tags = { pulse_action }
				character = scope:first

				scope:host.dynasty = {
					add_dynasty_prestige = medium_dynasty_prestige_gain
				}
				scope:first = {
					add_opinion = {
						modifier = impressed_opinion
						target = scope:host
						opinion = 10
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = dynasty_blessing_title
					left_icon = scope:first
					right_icon = scope:host
					show_as_tooltip = {
						scope:activity = {
							scope:host.dynasty = {
								add_dynasty_prestige = medium_dynasty_prestige_gain
							}
							scope:first = {
								add_opinion = {
									modifier = impressed_opinion
									target = scope:host
									opinion = 10
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_cultural_acceptance = {
	effect = {
		scope:activity = {
			if = {
				limit = {
					scope:activity = { has_activity_type = activity_wedding }
				}
				scope:activity.special_guest:spouse_1 = { save_scope_as = first }
				scope:activity.special_guest:spouse_2 = { save_scope_as = second }
			}

			if = {
				limit = {
					scope:activity = {
						has_activity_type = activity_coronation
						has_variable = officiator
					}
				}
				scope:host = { save_scope_as = first }
				scope:activity.var:officiator = { save_scope_as = second }
			}

			add_activity_log_entry = {
				key = cultural_acceptance
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:first.culture = {
					change_cultural_acceptance = {
						target = scope:second.culture
						value = miniscule_positive_culture_acceptance
						desc = cultural_acceptance_gain_hold_court_event_outcome
					}
				}
				scope:second.culture = {
					change_cultural_acceptance = {
						target = scope:first.culture
						value = miniscule_positive_culture_acceptance
						desc = cultural_acceptance_gain_hold_court_event_outcome
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_stewardship
					title = cultural_acceptance_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:first.culture = {
								change_cultural_acceptance = {
									target = scope:second.culture
									value = miniscule_positive_culture_acceptance
									desc = cultural_acceptance_gain_hold_court_event_outcome

									desc = cultural_acceptance_gain_hold_court_event_outcome
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_royal_gift = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			random_attending_character = {
				limit = {
					NOR = {
						this = scope:host
						scope:activity.special_guest:spouse_1 ?= this
						scope:activity.special_guest:spouse_2 ?= this
					}
					is_physically_able_ai_adult = yes
					location = scope:activity.activity_location
					highest_held_title_tier = tier_empire
					gold > minor_gold_value
				}
				alternative_limit = {
					NOR = {
						this = scope:host
						scope:activity.special_guest:spouse_1 ?= this
						scope:activity.special_guest:spouse_2 ?= this
					}
					is_physically_able_ai_adult = yes
					location = scope:activity.activity_location
					highest_held_title_tier >= tier_kingdom
					gold > minor_gold_value
				}
				save_scope_as = first
			}
			random_list = {
				30 = {
					modifier = {
						scope:host = { is_ai = no }
						add = 70
					}
					scope:host = { save_scope_as = second }
				}
				30 = {
					trigger = {
						scope:activity = { has_activity_type = activity_wedding }
					}
					modifier = {
						scope:activity.special_guest:spouse_1 = { is_ai = no }
						add = 70
					}
					scope:activity.special_guest:spouse_1 = { save_scope_as = second }
				}
				30 = {
					trigger = {
						scope:activity = { has_activity_type = activity_wedding }
					}
					modifier = {
						scope:activity.special_guest:spouse_2 = { is_ai = no }
						add = 70
					}
					scope:activity.special_guest:spouse_2 = { save_scope_as = second }
				}
			}


			add_activity_log_entry = {
				key = royal_gift
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:first = {
					pay_short_term_gold = {
						target = scope:second
						gold = minor_gold_value
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_wedding
					title = royal_gift_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								pay_short_term_gold = {
									target = scope:second
									gold = minor_gold_value
								}
							}
						}
					}
				}

		}
	}

MFA_pulse_poetry_performance = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}
			if = {
				limit = {
					scope:host = {
						has_trait = lifestyle_poet
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
					}
				}
				scope:host = { save_scope_as = first }
			}
			else = {
				random_attending_character = {
					limit = {
						OR = {
							has_trait = lifestyle_poet
							AND = {
								has_court_position = court_poet_court_position
								aptitude = {
									court_position = court_poet_court_position
									value >= 3
								}
								court_owner = scope:host
							}
						}
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
					}
					save_scope_as = first
				}
			}

			add_activity_log_entry = {
				key = poetry_performance
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_prestige = minor_prestige_gain
				}
				if = {
					limit = {
						scope:first != scope:host
					}
					scope:host = {
						add_prestige = minor_prestige_gain
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = poetry_performance_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_prestige = minor_prestige_gain
							}
							if = {
								limit = {
									scope:first != scope:host
								}
								scope:host = {
									add_prestige = minor_prestige_gain
								}
							}
						}
					}
				}
			}
		}
	}
}
