MFA_pulse_apa_cheering_on_competitors = {
	effect = {
		scope:activity = {
			random_guest_subset = {
				name = spectators
				limit = {
					NOR = {
						this = scope:host
						scope:activity.special_guest:chariot_race_honorary_guest ?= this
					}
					is_physically_able_ai_adult = yes
					num_of_relation_friend < 2
					location = scope:activity.activity_location
				}
				save_scope_as = first
			}

			random_guest_subset = {
				name = spectators
				limit = {
					NOR = {
						this = scope:host
						scope:activity.special_guest:chariot_race_honorary_guest ?= this
						this = scope:first
					}
					is_physically_able_ai_adult = yes
					num_of_relation_friend < 2
					location = scope:activity.activity_location
					number_of_traits_in_common = {
						target = scope:first
						value >= 1
					}
					can_set_relation_friend_trigger = { CHARACTER = scope:first }
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = apa_cheering_on_competitors
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:second = {
					progress_towards_friend_effect = {
						REASON = friend_chariot_convincing_display
						CHARACTER = scope:first
						OPINION = default_friend_opinion
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = apa_cheering_on_competitors_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:second = {
								progress_towards_friend_effect = {
									REASON = friend_chariot_convincing_display
									CHARACTER = scope:first
									OPINION = default_friend_opinion
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_apa_thunderous_applause = {
	effect = {
		scope:activity = {
			random_guest_subset = {

				limit = {
					var:current_place_in_race > 1
				}
				save_scope_as = first
			}

			random_guest_subset = {
				name = charioteers
				limit = {
					var:current_place_in_race = scope:first.ahead_charioteer_index

			}

			add_activity_log_entry = {
				key = apa_thunderous_applause

				target = scope:second

MFA_pulse_apa_exchanging_race_commentary = {
	effect = {
		scope:activity = {
			random_guest_subset = {
				name = spectators
				limit = {
					NOR = {
						scope:host = this
						scope:activity.special_guest:chariot_race_honorary_guest ?= this
					}
					is_ai = yes
					is_adult = yes
					might_cheat_on_every_partner_trigger = yes
				}
				save_scope_as = first
			}
			random_guest_subset = {
				name = spectators
				limit = {
					NOR = {
						scope:host = this
						scope:activity.special_guest:chariot_race_honorary_guest ?= this
					}
					is_attracted_to_gender_of = scope:first
					possibly_interested_in_character_trigger = {
						CHARACTER = scope:first
					}
					is_ai = yes
					is_adult = yes
					OR = {
						might_cheat_on_every_partner_trigger = yes
						is_consort_of = scope:first
					}
				}
				save_scope_as = second
			}
			add_activity_log_entry = {
				key = apa_exchanging_race_commentary

					left_icon = scope:first
