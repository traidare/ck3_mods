MFA_pulse_agot_coronation_dynasty_blessing = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			random_attending_character = {
				limit = {
					NOR = {
						this = scope:host
						this = scope:activity.special_guest:coronation_target
					}
					is_physically_able_ai_adult = yes
					location = scope:activity.activity_location
					dynasty ?= { dynasty_prestige_level > scope:host.dynasty.dynasty_prestige_level }
				}
				save_scope_as = first
			}

			add_activity_log_entry = {
				key = dynasty_blessing
				tags = { pulse_action }
				character = scope:first

				scope:host.dynasty = {
					add_dynasty_prestige = medium_dynasty_prestige_gain
				}
				scope:first = {
					add_opinion = {
						modifier = impressed_opinion
						target = scope:host
						opinion = 10
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = agot_coronation_dynasty_blessing_title
					left_icon = scope:first
					right_icon = scope:host
					show_as_tooltip = {
						scope:activity = {
							scope:host.dynasty = {
								add_dynasty_prestige = medium_dynasty_prestige_gain
							}
							scope:first = {
								add_opinion = {
									modifier = impressed_opinion
									target = scope:host
									opinion = 10
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_agot_coronation_seductive_exchange = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			random_attending_character = {
				limit = {
					has_activity_intent = woo_attendee_intent
					is_physically_able_ai_adult = yes
					location = scope:activity.activity_location
					intent_target = {
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
						is_ai = yes
					}
				}
				save_scope_as = first
			}
			scope:first.intent_target = { save_scope_as = second }

			add_activity_log_entry = {
				key = seductive_exchange
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:first = {
					progress_towards_lover_effect = {
						CHARACTER = scope:second
						OPINION = default_lover_opinion

					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no

				}
				send_interface_message = {
					type = MFA_pulse_msg_intrigue

					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {

MFA_pulse_agot_coronation_poetry_performance = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}
			if = {
				limit = {
					scope:host = {
						has_trait = lifestyle_poet
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
					}
				}
				scope:host = { save_scope_as = first }
			}
			else = {
				random_attending_character = {
					limit = {
						OR = {
							has_trait = lifestyle_poet
							AND = {
								has_court_position = court_poet_court_position
								aptitude = {
									court_position = court_poet_court_position
									value >= 3
								}
								court_owner = scope:host
							}
						}
						is_physically_able_ai_adult = yes
						location = scope:activity.activity_location
					}
					save_scope_as = first
				}
			}

			add_activity_log_entry = {
				key = poetry_performance
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_prestige = minor_prestige_gain
				}
				if = {
					limit = {
						NOT = { scope:first = scope:host }
					}
					scope:host = {
						add_prestige = minor_prestige_gain
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = agot_coronation_poetry_performance_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_prestige = minor_prestige_gain
							}
							if = {
								limit = {
									NOT = { scope:first = scope:host }
								}
								scope:host = {
									add_prestige = minor_prestige_gain
								}
							}
						}
					}
				}
			}
		}
	}
}
