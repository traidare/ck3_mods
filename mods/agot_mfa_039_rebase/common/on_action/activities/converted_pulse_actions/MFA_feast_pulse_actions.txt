MFA_pulse_excellent_food = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = root_scope
			}

			random_attending_character = {
				limit = {
					this != scope:host
					is_physically_able_ai_adult = yes
					location = scope:activity.activity_location
					OR = {
						likes_character_trigger = { CHARACTER = scope:host }
						loves_food_trigger = yes
					}
				}
				save_scope_as = first
			}

			add_activity_log_entry = {
				key = excellent_food
				tags = { pulse_action }
				character = scope:first

				scope:host = {
					add_stress = medium_stress_loss
				}
				scope:first = {
					add_opinion = {
						modifier = impressed_opinion
						target = scope:host
						opinion = 10
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = excellent_food_title
					left_icon = scope:first
					right_icon = scope:host
					show_as_tooltip = {
						scope:activity = {
							scope:host = {
								add_stress = medium_stress_loss
							}
							scope:first = {
								add_opinion = {
									modifier = impressed_opinion
									target = scope:host
									opinion = 10
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_hook_trusting_guest = {
	effect = {
		scope:activity = {
			random_attending_character = {
				limit = {
					this != scope:host
					location = scope:activity.activity_location
					has_trait = trusting
					is_ai = yes
					save_temporary_scope_as = first_temp
					scope:activity = {
						any_attending_character = {
							this != scope:first_temp
							location = scope:activity.activity_location
							can_add_hook = {
								target = scope:first_temp
								type = favor_hook
							}
						}
					}
				}
				save_scope_as = first
			}
			random_attending_character = {
				limit = {
					location = scope:activity.activity_location
					this != scope:first
					can_add_hook = {
						target = scope:first
						type = favor_hook
					}
				}
				weight = {
					base = 1
					modifier = {
						add = 10
						is_ai = no
					}
					modifier = {
						add = 10
						this = scope:host
					}
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = hook_trusting_guest
				tags = { pulse_action }
				character = scope:second
				target = scope:first

				scope:second = {
					add_hook = {
						target = scope:first
						type = favor_hook
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_hook
					title = hook_trusting_guest_title
					left_icon = scope:second
					right_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:second = {
								add_hook = {
									target = scope:first
									type = favor_hook
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_antiquarian_admires_artifact = {
	effect = {
		scope:activity = {
			if = {
				limit = {
					any_attending_character = {
						has_court_position = antiquarian_court_position
						location = scope:activity.activity_location
						is_physically_able_ai_adult = yes
					}
				}
				random_attending_character = {
					limit = {
						has_court_position = antiquarian_court_position
						location = scope:activity.activity_location
						is_physically_able_ai_adult = yes
					}
					save_scope_as = first
				}
			}
			else = {
				random_attending_character = {
					limit = {
						location = scope:activity.activity_location
						is_physically_able_ai_adult = yes
						learning >= high_skill_rating
					}
					save_scope_as = first
				}
			}

			scope:host = {
				random_character_artifact = {
					limit = {
						category = inventory
						NOT = { rarity = common }
					}
					save_scope_as = artifact
					save_scope_as = second
				}
			}

			add_activity_log_entry = {
				key = antiquarian_admires_artifact
				tags = { pulse_action }
				character = scope:first
				artifact = scope:artifact

				scope:host = {
					add_prestige = minor_prestige_gain
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = MFA_pulse_msg_public
					title = antiquarian_admires_artifact_title
					left_icon = scope:first
					right_icon = scope:artifact
					show_as_tooltip = {
						scope:activity = {
							scope:host = {
								add_prestige = minor_prestige_gain
							}
						}
					}
				}
			}
		}
	}
}
