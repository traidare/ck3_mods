MFA_pulse_loyalty_pledge = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
			}
			random_guest_subset = {
				name = supporter
				limit = {
					is_playable_character = yes
					is_vassal_of = scope:host
					location = scope:activity.activity_location
					opinion = {
						target = scope:host
						value >= low_positive_opinion
					}
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = loyalty_pledge
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:first = {
					add_prestige = minor_prestige_gain
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = loyalty_pledge_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_prestige = minor_prestige_gain
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_regalia_impresses_attendees = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
				random_character_artifact = {
					limit = {
						artifact_slot_type = regalia
						OR = {
							rarity = illustrious
							rarity = famed
						}
					}
					save_scope_as = regalia
				}
			}

			add_activity_log_entry = {
				key = regalia_impresses_attendees
				tags = { pulse_action }
				character = scope:first

				every_attending_character = {
					limit = { NOT = { this = scope:host } }
					custom = custom.every_attending_character
					add_opinion = {
						target = scope:host
						modifier = impressed_opinion
						opinion = 2
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = regalia_impresses_attendees_title
					left_icon = scope:first
					right_icon = scope:host
					show_as_tooltip = {
						scope:activity = {
							every_attending_character = {
								limit = { NOT = { this = scope:host } }
								custom = custom.every_attending_character
								add_opinion = {
									target = scope:host
									modifier = impressed_opinion

				}
			}
		}
	}
}
