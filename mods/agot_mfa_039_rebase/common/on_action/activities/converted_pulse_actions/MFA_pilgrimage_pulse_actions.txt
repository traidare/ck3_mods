MFA_pulse_apa_overcome_with_emotion = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
			}

			add_activity_log_entry = {
				key = apa_overcome_with_emotion
				tags = { pulse_action }
				character = scope:first

				scope:first = {
					add_piety = 20
					stress_impact = {
						base = minor_stress_impact_gain
						zealous = miniscule_stress_impact_loss
						humble = miniscule_stress_impact_loss
						compassionate = miniscule_stress_impact_loss
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = apa_overcome_with_emotion_title
					left_icon = scope:first
					show_as_tooltip = {
						scope:activity = {
							scope:first = {
								add_piety = 20
								stress_impact = {
									base = minor_stress_impact_gain
									zealous = miniscule_stress_impact_loss
									humble = miniscule_stress_impact_loss
									compassionate = miniscule_stress_impact_loss
								}
							}
						}
					}
				}
			}
		}
	}
}

MFA_pulse_apa_preaching_to_unbelievers = {
	effect = {
		scope:activity = {
			scope:host = {
				save_scope_as = first
			}
			random_pool_character = {
				province = scope:activity.activity_location
				limit = {
					faith != scope:first.faith
					NOT = { has_trait = zealous }
				}
				save_scope_as = second
			}

			add_activity_log_entry = {
				key = apa_preaching_to_unbelievers
				tags = { pulse_action }
				character = scope:first
				target = scope:second

				scope:second = {
					set_character_faith = scope:first.faith
				}
				scope:first = {
					add_piety = 15
				}
			}
			#Now we set a memory so that this conversion does not look to random if one looks at the character
			hidden_effect = { #Memory Management
				scope:first = {
					create_character_memory = {
						type = memory_apa_converted
						participants = {
							converted = scope:second
						}
					}
				}
				if = {
					limit = { exists = scope:new_memory }
					scope:new_memory = {
						set_variable = {
							name = memory_location
							value = scope:activity.activity_location
						}
					}
				}
				scope:second = {
					create_character_memory = {
						type = memory_apa_was_converted
						participants = {
							converter = scope:first
						}
					}
				}
			}

			every_attending_character = {
				limit = {
					is_ai = no
					has_game_rule = MFA_pulse_notifications_enabled
				}
				send_interface_message = {
					type = send_interface_message
					title = apa_preaching_to_unbelievers_title
					left_icon = scope:first
					right_icon = scope:second
					show_as_tooltip = {
						scope:activity = {
							scope:second = {
								set_character_faith = scope:first.faith
							}
							scope:first = {
								add_piety = 15
							}
						}
					}
				}
			}
		}
	}
}
