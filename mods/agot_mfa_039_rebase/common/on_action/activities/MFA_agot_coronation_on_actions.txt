MFA_agot_coronation_1_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_1_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			# Fires for every attendee
			trigger_event = { on_action = coronation_ceremony_ongoing_event_pulse }
		}

		change_variable = {
			name = MFA_phase_1_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
			}
			trigger_event = { on_action = MFA_agot_coronation_1_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_1_remaining
		}
	}
}

MFA_agot_coronation_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_agot_coronation_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_agot_coronation_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
