####################################
# MFA Activity Converter
# Pulse relay for festival activity
# Root scope = the character (same as on_weekly_pulse).
####################################



MFA_festival_0_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_0_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			# Fires for every attendee
			trigger_event = {
				on_action = festival_random_pulse
			}
		}

		change_variable = {
			name = MFA_phase_0_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
			}
			trigger_event = { on_action = MFA_festival_0_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_0_remaining
		}
	}
}

MFA_festival_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_festival_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_festival_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}

							any_courtier_or_guest = {
								is_physically_able_ai = yes

							any_courtier_or_guest = {
								is_physically_able_ai = yes
								is_participant_in_activity = scope:activity

						scope:host.location.county.holder.top_liege = scope:host.top_liege

							scope:host.location.county = {
								has_variable = recent_dev

				}
				trigger_event = { on_action = MFA_pulse_liege_visited_benevolence }
			}
			1 = { # guest_friendship

					scope:activity = {
