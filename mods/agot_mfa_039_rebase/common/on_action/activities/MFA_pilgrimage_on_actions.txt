MFA_pilgrimage_0_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_0_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			if = {
				limit = { this = scope:host }
				#We are in a Pilgrimage event chain or Hajjing
				if = {
					limit = {
						OR = {
							#AGOT Disabled, no hajj references
							#Hajj destination events are railroaded via pilgrimage.3101
							#AGOT Modified, remove hajj references
							# scope:activity = {
							# 	has_activity_option = {
							# 		category = special_type
							# 		option = pilgrimage_type_hajj
							# 	}
							# }
							#No events until you're done with your chain
							has_character_flag = in_pilgrimage_event_chain
						}
					}
					#No events until you're done with your chain
				}
				#We're just doing regular pilgrim destination stuff
				else = {
					trigger_event = { on_action = pilgrimage_destination_events }
				}
			}
		}

		change_variable = {
			name = MFA_phase_0_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
			}
			trigger_event = { on_action = MFA_pilgrimage_0_relay days = MFA_1_month_days }
		}
		else = {
			remove_variable = MFA_phase_0_remaining
		}
	}
}

MFA_pilgrimage_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_pilgrimage_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_pilgrimage_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
