MFA_playdate_0_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_0_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
				exists = scope:activity
				scope:activity = { is_current_phase_active = yes }
			}
			# Fires for every attendee
			trigger_event = {
				on_action = playdate_event_selection
				days = { 1 3 }
			}
			random = {
				chance = {
					value = 10
					if = {
						limit = {
							has_bp2_dlc_trigger = yes
							this = scope:activity.activity_host
						}
						add = 40
					}
				}
				trigger_event = {
					on_action = playdate_event_selection
					days = { 15 20 }
				}
			}
		}

		change_variable = {
			name = MFA_phase_0_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
			}
			trigger_event = { on_action = MFA_playdate_0_relay days = MFA_1_month_days }
		}
		else = {
			remove_variable = MFA_phase_0_remaining
		}
	}
}

MFA_playdate_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_playdate_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_playdate_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
