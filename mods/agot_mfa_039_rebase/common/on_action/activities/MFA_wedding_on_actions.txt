# MFA Activity Converter
# Pulse relay for wedding activity
# Root scope = the character (same as on_weekly_pulse).
####################################



MFA_wedding_1_relay = {
	trigger = {
		exists = scope:activity
		is_ruler = yes
		scope:activity = { is_current_phase_active = yes }
	}

	effect = {
		# Fires for every attendee
		trigger_event = { on_action = wedding_ceremony_ongoing_event_pulse }

		if = {
			limit = {
				scope:activity = { is_current_phase_active = yes }
			}
			trigger_event = { on_action = MFA_wedding_1_relay days = 7 }
		}
	}
}

MFA_wedding_2_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_2_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_2_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			# Fires for every attendee
			trigger_event = { on_action = wedding_banquet_ongoing_event_pulse }
		}

		change_variable = {
			name = MFA_phase_2_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_2_remaining >= 1
			}
			trigger_event = { on_action = MFA_wedding_2_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_2_remaining
		}
	}
}

MFA_wedding_3_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_3_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_3_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			# Bloody Wedding only has a murder event, no regular wnight
			if = {
				limit = {
					is_bloody_wedding = yes
				}
				if = {
					limit = {
						scope:host = scope:activity.special_guest:spouse_1 #if you're the spouse
						scope:host = { var:bloody_wedding_murder ?= flag:spouse } #and you've chosen to kill them during the wedding night
					}
					scope:host = { trigger_event = ep2_wedding.3060 }
				}
			}
			# For a regular wedding
			else = {
				# Fires only for the spouses
				if = {
					limit = {
						OR = {
							root = scope:activity.special_guest:spouse_1
							root = scope:activity.special_guest:spouse_2
						}
					}
					trigger_event = {
						on_action = wedding_wnight_spouse_event_pulse
					}
				}
				# Fires for the guests and host
				else = {
					trigger_event = { on_action = wedding_wnight_ongoing_event_pulse }
				}
			}
		}

		change_variable = {
			name = MFA_phase_3_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_3_remaining >= 1
			}
			trigger_event = { on_action = MFA_wedding_3_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_3_remaining
		}
	}
}

MFA_wedding_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_wedding_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_wedding_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
