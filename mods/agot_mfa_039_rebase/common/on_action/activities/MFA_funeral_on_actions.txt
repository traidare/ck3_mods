# MFA Activity Converter
# Pulse relay for funeral activity
# Root scope = the character (same as on_weekly_pulse).
####################################



MFA_funeral_1_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_1_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			scope:activity.var:body_to_bury = { save_scope_as = deceased }
			root.involved_activity = { save_scope_as = activity }
			trigger_event = { on_action = funeral_wake_pulse }
		}

		change_variable = {
			name = MFA_phase_1_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
			}
			trigger_event = { on_action = MFA_funeral_1_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_1_remaining
		}
	}
}

MFA_funeral_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_funeral_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_funeral_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
