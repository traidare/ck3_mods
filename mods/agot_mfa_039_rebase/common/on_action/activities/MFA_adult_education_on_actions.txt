MFA_adult_education_0_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_0_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			if = {
				limit = { this = scope:host }
				# Let's check that we still have the required teachers and students, and if not let's create them again
				#Set up recurring characters: teachers
				bp2_university_teacher_setup = yes
				#Set up recurring characters: students
				bp2_university_student_setup = yes
				trigger_event = { on_action = adult_education_destination_events }
			}
		}

		change_variable = {
			name = MFA_phase_0_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_0_remaining >= 1
			}
			trigger_event = { on_action = MFA_adult_education_0_relay days = MFA_adult_education_1_month_days }
		}
		else = {
			remove_variable = MFA_phase_0_remaining
		}
	}
}

MFA_adult_education_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_adult_education_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_adult_education_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
