####################################
# MFA Activity Converter
# Pulse relay for chariot_race activity
# Root scope = the character (same as on_weekly_pulse).
####################################



MFA_chariot_race_1_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_1_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			if = {
				limit = {
					this = scope:host
				}
				scope:activity = {
					every_guest_subset = {
						name = spectators
						trigger_event = { on_action = chariot_race_betting_events }
					}
				}
			}
		}

		change_variable = {
			name = MFA_phase_1_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_1_remaining >= 1
			}
			trigger_event = { on_action = MFA_chariot_race_1_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_1_remaining
		}
	}
}

MFA_chariot_race_2_relay = {
	trigger = {
		is_ruler = yes
		exists = var:MFA_phase_2_remaining
	}

	effect = {
		if = {
			limit = {
				var:MFA_phase_2_remaining >= 1
				scope:activity = { is_current_phase_active = yes }
			}
			if = {
				limit = { this = scope:host }
				# This on action handles everything occurs on the race track and then fires informative events to spectators who care about the charioteers affected
				# After all that is sorted, it fires chariot_spectator_on_action for spectators that have not yet received an event
				trigger_event = { on_action = chariot_race_on_action }
			}
		}

		change_variable = {
			name = MFA_phase_2_remaining
			add = -1
		}
		if = {
			limit = {
				var:MFA_phase_2_remaining >= 1
			}
			trigger_event = { on_action = MFA_chariot_race_2_relay days = MFA_1_week_days }
		}
		else = {
			remove_variable = MFA_phase_2_remaining
		}
	}
}

MFA_chariot_race_pulse_relay = {
	effect = {
		if = {
			limit = {
				has_variable = MFA_pulse_relay_tick
			}
			change_variable = { name = MFA_pulse_relay_tick add = 1 }
		}
		else = {
			set_variable = { name = MFA_pulse_relay_tick value = 1 }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_chariot_race_pulse_relay_effects }
		}
		if = {
			limit = {
				exists = scope:activity
			}
			trigger_event = { on_action = MFA_chariot_race_pulse_relay days = 1 }
		}
		else = {
			remove_variable = MFA_pulse_relay_tick
		}
	}
}
