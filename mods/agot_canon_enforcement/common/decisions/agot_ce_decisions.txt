# The interaction cannot target its own actor, so self-marking is a decision.
# Both are gated on the same manual-marking rule.

agot_ce_protect_self_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}

	desc = agot_ce_protect_self_decision_desc
	selection_tooltip = agot_ce_protect_self_decision_tooltip

	is_shown = {
		has_game_rule = agot_ce_manual_marking_enabled
		NOT = { has_character_flag = agot_ce_protected }
	}

	effect = {
		add_character_flag = agot_ce_protected
		agot_ce_apply_protection_effect = yes
	}

	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

agot_ce_unprotect_self_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}

	desc = agot_ce_unprotect_self_decision_desc
	selection_tooltip = agot_ce_unprotect_self_decision_tooltip

	is_shown = {
		has_game_rule = agot_ce_manual_marking_enabled
		has_character_flag = agot_ce_protected
	}

	effect = {
		remove_character_flag = agot_ce_protected
		if = {
			limit = { agot_ce_is_protected_trigger = no }
			agot_ce_clear_protection_effect = yes
		}
	}

	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
