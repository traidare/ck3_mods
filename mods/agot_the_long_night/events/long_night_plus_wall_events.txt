namespace = long_night_plus_wall

# The Long Night+ — #12 the Wall, LAYER B cinematic (Option A: middle break + full animation).
# Fired per human player from agot_long_night_wall_collapse_effect.
#
# Built on AGOT's working widget. Tweaks vs the bare AGOT version:
#   - widget = our copy ln_particle_wall_collapse (AGOT's widget byte-for-byte, delay kept at 0.1 — raising it
#     breaks the widget entirely). Music sync is solved on the AUDIO side instead: we play a trimmed copy of the
#     track (LN_wall_collapse_the_others) whose intro is shaved so the boom lands on the ~instant fall.
#   - window = ln_topleft_wall_event (AGOT's character_event window anchored TOP-LEFT so it no longer masks the
#     animation). The window stays VISIBLE/on-screen — hiding it breaks the widget's clock (whole saga).
# The widget spawns the collapse mesh, breaks the Wall via AGOT's agot_wall_fall (building swap to the broken mesh),
# spawns the dust effect and AUTO-CLOSES the event. We only pan the camera and play the collapse music.
long_night_plus_wall.1 = {
	type = character_event
	window = ln_topleft_wall_event
	title = long_night_plus_wall.1.t
	desc = long_night_plus_wall.1.desc
	theme = default

	immediate = {
		pan_camera_to_province = title:b_castle_black.title_province
		play_music_cue = LN_wall_collapse_the_others # trimmed copy, boom synced on the fall (music/in_game/zz_ln_sounds.txt)
	}

	widgets = {
		widget = {
			gui = "ln_particle_wall_collapse"
			container = "custom_widgets_container"
			controller = default
			setup_scope = { }
		}
	}

	option = {
		name = long_night_plus_wall.1.a
	}
}

# Reliable fallback: with no human player to play the widget (AI-only realm / observer), break the Wall directly if it
# is still standing. Mirrors what the widget's agot_wall_fall does (middle break + building swap to force the refresh).
long_night_plus_wall.3 = {
	hidden = yes
	trigger = {
		title:b_castle_black.title_province = {
			NOT = { has_variable = agot_wall_has_fallen_left }
			NOT = { has_variable = agot_wall_has_fallen_middle }
			NOT = { has_variable = agot_wall_has_fallen_right }
		}
	}
	immediate = {
		title:b_castle_black.title_province = {
			set_variable = agot_wall_has_fallen_middle
			remove_building = special_the_wall
			add_building = special_the_wall_damaged
		}
	}
}
