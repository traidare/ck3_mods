namespace = ln_aftermath

# The Long Night+ — AFTERMATH (resettling) events.
# Design doc: dev/the_long_night_plus_aftermath.md (sibling, not shipped).
#
# ln_aftermath.1  — THE QUESTION OF THE NORTH: the mandatory Crown dilemma (restore the Watch ⇄ welcome the
#                   free folk), scheduled 30-180 days after the Long Night ends. Mutually exclusive forever.
# ln_aftermath.3  — notification to players: the free folk pour south (wildling path taken).
# ln_aftermath.20 — refugee caravan arrival router (hidden): settle here, walk to the next empty county, or
#                   give up when nothing is left.
# ln_aftermath.21 — player refugee arrival choice (the AI settles automatically).
# ln_aftermath.25 — player notification: the road ends, the refugee becomes a plain adventurer.

# ------------------------------------------------------------------------------------------------------------
# THE QUESTION OF THE NORTH
ln_aftermath.1 = {
	type = character_event
	title = ln_aftermath.1.t
	desc = ln_aftermath.1.desc
	theme = realm
	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		NOT = { exists = global_var:ln_nw_dilemma_done }
		ln_aftermath_crown_trigger = yes
		ln_long_night_over_trigger = yes
	}

	# Restore the Night's Watch — the free folk stay outside, the Wall's ruin is garrisoned again.
	option = {
		name = ln_aftermath.1.a
		custom_tooltip = ln_aftermath.1.a.tt
		set_global_variable = { name = ln_nw_dilemma_done value = yes }
		set_global_variable = { name = ln_nights_watch_restored value = yes }
		longnight_reform_nightswatch_effect = yes
		ai_chance = {
			base = 50
			modifier = { add = 25 has_trait = zealous }
			modifier = { add = 15 has_trait = just }
			modifier = { add = 10 has_trait = paranoid }
			modifier = { add = -20 has_trait = compassionate }
			modifier = { add = -10 has_trait = fickle }
		}
	}

	# Welcome the free folk — the Watch never rises again; the clans settle the Wall and the Gifts.
	option = {
		name = ln_aftermath.1.b
		custom_tooltip = ln_aftermath.1.b.tt
		hidden_effect = {
			save_scope_as = ln_unleash_chooser   # the choosing crown skips the ln_aftermath.3 notification
			ln_wildlings_unleashed_effect = yes
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = compassionate }
			modifier = { add = 15 has_trait = brave }
			modifier = { add = 10 has_trait = generous }
			modifier = { add = -30 has_trait = zealous }
			modifier = { add = -15 has_trait = paranoid }
		}
	}
}

# Refugee caravan arrival router (hidden). Fired by the Return North travel plan's on_arrival.
ln_aftermath.20 = {
	hidden = yes

	trigger = {
		has_character_flag = dawn_refugee
		is_landless_adventurer = yes
	}

	immediate = {
		# WAYPOINT stop (Moat Cailin on the causeway, on_arrival_destinations = all fires here too):
		# resupply the caravan and roll on. Final-arrival logic only runs at the stored destination —
		# never settle at a waypoint, even one that happens to lie in scoured wilderness.
		if = {
			limit = {
				is_travelling = yes
				NOT = { location.county ?= var:ln_return_destination }
			}
			domicile ?= { change_provisions = 400 }
		}
		else_if = {
			limit = { location.county ?= { ln_wilderness_county_trigger = yes } }
			# The settler folk gather HERE, at the destination — northern survivors flocking to a returned
			# lord. Raised at departure they would drain the camp's provisions for the whole forced trip.
			# Guard: re-arrivals (after a "not yet") must not stack more regiments.
			if = {
				limit = { agot_current_settler_maa_count < 100 }
				if = { limit = { ln_refugee_settler_size_value >= 4 } create_maa_regiment = { type = laamp_settler_maa size = 4 check_can_recruit = no } }
				else_if = { limit = { ln_refugee_settler_size_value >= 3 } create_maa_regiment = { type = laamp_settler_maa size = 3 check_can_recruit = no } }
				else_if = { limit = { ln_refugee_settler_size_value >= 2 } create_maa_regiment = { type = laamp_settler_maa size = 2 check_can_recruit = no } }
				else = { create_maa_regiment = { type = laamp_settler_maa size = 1 check_can_recruit = no } }
			}
			if = {
				limit = { is_ai = yes }
				ln_refugee_settle_here_effect = yes
			}
			else = {
				trigger_event = ln_aftermath.21
			}
		}
		else = {
			# Claimed while the caravan traveled — walk on to the next empty county.
			location.county ?= { ln_find_wilderness_neighbor_effect = yes }
			if = {
				limit = { NOT = { exists = scope:ln_return_target } }
				random_county_in_region = {
					region = world_westeros_the_north_mainland
					limit = { ln_wilderness_county_trigger = yes }
					save_scope_as = ln_return_target
				}
			}
			if = {
				limit = { exists = scope:ln_return_target }
				# Restock for the extra leg — the re-route is forced too, no starvation on our watch.
				# Already in the North by now, so a direct leg (no waypoint); update the stored destination.
				set_variable = { name = ln_return_destination value = scope:ln_return_target }
				domicile ?= { change_provisions = 500 }
				start_travel_plan = {
					destination = scope:ln_return_target.title_province
					travel_with_domicile = yes
					players_use_planner = no
					on_arrival_event = ln_aftermath.20
					on_arrival_destinations = all
				}
			}
			else = {
				# Nothing left to settle anywhere — the road ends here.
				ln_clear_refugee_state_effect = yes
				if = {
					limit = { is_ai = no }
					trigger_event = ln_aftermath.25
				}
			}
		}
	}
}
