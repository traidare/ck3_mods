others_maintenance.1 = {
	scope = none
	hidden = yes
	immediate = {
		set_global_variable = { name = agot_long_night_enabled value = yes }
		set_global_variable = { name = agot_long_night_eligible_year value = current_year }
		if = {
			limit = { NOT = { has_dlc_feature = roads_to_power } }
			set_global_variable = { name = agot_long_night_dependency_blocked value = yes }
			every_player = { trigger_event = agot_ln.1 }
		}
		else_if = { limit = { has_game_rule = others_invasion_in_50_years } change_global_variable = { name = agot_long_night_eligible_year add = 50 } }
		else_if = { limit = { has_game_rule = others_invasion_in_100_years } change_global_variable = { name = agot_long_night_eligible_year add = 100 } }
		else_if = { limit = { has_game_rule = others_invasion_in_150_years } change_global_variable = { name = agot_long_night_eligible_year add = 150 } }
		else_if = {
			limit = { has_game_rule = others_historical_invasion current_year < 8298 }
			set_global_variable = { name = agot_long_night_eligible_year value = 8298 }
		}
		else_if = {
			limit = { has_game_rule = others_full_random_invasion }
			random_list = {
				20 = { }
				20 = { change_global_variable = { name = agot_long_night_eligible_year add = 25 } }
				20 = { change_global_variable = { name = agot_long_night_eligible_year add = 50 } }
				20 = { change_global_variable = { name = agot_long_night_eligible_year add = 100 } }
				20 = { change_global_variable = { name = agot_long_night_eligible_year add = 150 } }
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = agot_long_night_dependency_blocked }
				NOT = { has_game_rule = others_invasion_wont_happen }
				NOT = { has_game_rule = others_manual_invasion }
			}
			trigger_event = { id = agot_ln.10 days = 1 }
		}
	}
}
