			limit = { has_variable = temp_government_type }

		}
	}

apply_character_title_data = {
	if = {
		limit = {
			exists = $SCOPED_TITLE$
			exists = $SCOPE$
		}
		$SCOPE$ ?= {
			if = {
				limit = { is_titled_character = yes }

				### TEMPLE BARONY HOLDERS ###
				# AM/AGOT+/LoV Compatch delta, carried here too since this replace file wins the
				# effect game-wide: rare AGOT lay-clergy faiths can skip the normal theocracy
				# branch and fall into MISSING_TITLE_LOC.
				if = {
					limit = {
						$SCOPED_TITLE$ ?= {
							tier = tier_barony
							title_province = { has_holding_type = church_holding }
						}
					}
					if = {
						limit = { religion = religion:the_seven_religion }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_male_the_seven_religion" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_female_the_seven_religion" GENDER = female }
					}
					else = {
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_male" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_female" GENDER = female }
					}
				}
				### RULERS & SPOUSES ###
				else_if = {
					limit = { is_titled_ruler_or_spouse = yes }

					### CLAIMANT OVERRIDE ###
					if = {
						limit = {
							OR = {
								has_character_flag = agot_claimant_king
								scope:spouse_scope ?= { has_character_flag = agot_claimant_king }
							}
						}
						if = {
							limit = {
								OR = {
									any_character_war = {
										OR = {
											casus_belli = {
												any_target_title = {
													this = title:e_the_stormlands
												}
											}
											AND = {
												prev = { has_title = title:e_the_stormlands }
												OR = {
													using_cb = independence_war
													using_cb = agot_independence_war
												}
											}
										}
									}
									any_spouse = {
										even_if_dead = yes
										any_character_war = {
											OR = {
												casus_belli = {
													any_target_title = {
														this = title:e_the_stormlands
													}
												}
												AND = {
													prev = { has_title = title:e_the_stormlands }
													OR = {
														using_cb = independence_war
														using_cb = agot_independence_war
													}
												}
											}
										}
									}
								}
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_stormlands" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "queen_stormlands" GENDER = female }
						}
						else_if = {
							limit = {
								OR = {
									AND = {
										culture = {
											OR = {
												has_name_list = name_list_salt_dornish
												has_name_list = name_list_sand_dornish
												has_name_list = name_list_rhoynish
												has_cultural_pillar = heritage_rhoynar
											}
										}
										any_character_war = {
											OR = {
												casus_belli = {
													any_target_title = {
														this = title:e_dorne
													}
												}
												AND = {
													prev = { has_title = title:e_dorne }
													OR = {
														using_cb = independence_war
														using_cb = agot_independence_war
													}
												}
											}
										}
									}
									any_spouse = {
										even_if_dead = yes
										culture = {
											OR = {
												has_name_list = name_list_salt_dornish
												has_name_list = name_list_sand_dornish
												has_name_list = name_list_rhoynish
												has_cultural_pillar = heritage_rhoynar
											}
										}
										any_character_war = {
											OR = {
												casus_belli = {
													any_target_title = {
														this = title:e_dorne
													}
												}
												AND = {
													prev = { has_title = title:e_dorne }
													OR = {
														using_cb = independence_war
														using_cb = agot_independence_war
													}
												}
											}
										}
									}
								}
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_rhoynish" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "princess" GENDER = female }
						}
						else_if = {
							limit = { always = yes }

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "queen" GENDER = female }
						}
					}

					### NEW VALYRIA HOLDER OVERRIDE ###
					else_if = {
						limit = {
							$SCOPED_TITLE$ ?= title:h_new_valyria
							$SCOPED_TITLE$ ?= { holder = $SCOPE$ }
						}

						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "new_valyria_emperor_valyrian_male" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "new_valyria_emperor_valyrian_female" GENDER = female }
					}

					### PAINTED MOUNTAINS KHAL HOLDER OVERRIDE ###
					else_if = {
						limit = {
							$SCOPED_TITLE$ ?= title:k_khalasar_painted_mountains
							$SCOPED_TITLE$ ?= { holder = $SCOPE$ }
						}

						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "khal_king_male" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "khal_king_female" GENDER = female }
					}

					### EE TEMPLOV COMPATCH: FAR EAST LORE TITLES ###
					# Culture-gated overrides for Essos Expanded realms, checked before the government chain.
					# Only fires at realm-ruler tiers; vassals fall through to normal naming.
					else_if = {
						limit = {
							OR = {
								culture = culture:feicui_ren
								culture = culture:lengii
							}
							primary_title ?= { tier >= tier_kingdom }
							$SCOPED_TITLE$.holder ?= $SCOPE$
						}
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ee_god_emperor_male" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ee_god_emperor_female" GENDER = female }
					}
					else_if = {
						limit = {
							OR = {
								culture = culture:feicui_ren
								culture = culture:lengii
							}
							has_government = celestial_government
							$SCOPED_TITLE$.holder ?= $SCOPE$
						}
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ee_yiti_governor_male" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ee_yiti_governor_female" GENDER = female }
					}
					else_if = {
						limit = {
							culture = culture:asshaii
							$SCOPED_TITLE$.holder ?= $SCOPE$
							primary_title ?= { tier >= tier_duchy }
						}
						if = {
							limit = { primary_title ?= { tier >= tier_empire } }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_asshai_male" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_asshai_female" GENDER = female }
						}
						else = {
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_asshai_male" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_asshai_female" GENDER = female }
						}
					}
					else_if = {
						limit = {
							culture = culture:kayakayanayan
							primary_title ?= { tier >= tier_kingdom }

										$SCOPED_TITLE$ ?= title:k_magnar_of_thenn

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_hof" GENDER = male }

							else_if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

						}

						### DUCHY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_duchy } }
							if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_duke" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_duke" GENDER = female }
							}
							else_if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "duke_tribal_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "duke_tribal_female" GENDER = female }
							}
						}

						### COUNTY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_county } }
							if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_count" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_count" GENDER = female }
							}
							else_if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "count_tribal_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "count_tribal_female" GENDER = female }
							}
						}

						### BARONY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_barony } }
							if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_tribal_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_tribal_female" GENDER = female }
							}
						}
					}

					### CLAN ###
					else_if = {
						limit = {
							agot_localization_government_trigger = { GOV = clan SPOUSE_SCOPE = scope:spouse_scope }
						}

						### HEGEMONY ###
						if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_hegemony } }
							if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "hegemon" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "hegemon_female" GENDER = female }
							}
						}

						### EMPIRE ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_empire } }
							if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_emperor" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_emperor" GENDER = female }
							}
							else_if = {
								limit = { $SCOPED_TITLE$ ?= title:e_beyond_the_wall }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_male_beyond_the_wall" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_female_beyond_the_wall" GENDER = female }
							}
							else_if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_clan_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "emperor_clan_female" GENDER = female }
							}
						}

						### KINGDOM ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_kingdom } }
							if = { # Magnar
								limit = {
									OR = {
										$SCOPED_TITLE$ ?= title:k_magnar_of_thenn
										$SCOPED_TITLE$ ?= title:k_thenn
									}
								}

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_hof" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_hof" GENDER = female }
							}
							else_if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_king" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_king" GENDER = female }
							}
							else_if = { # Fallback
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_clan_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "king_clan_female" GENDER = female }
							}
						}

						### DUCHY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_duchy } }
							if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_duke" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_duke" GENDER = female }
							}
							else_if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "duke_clan_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "duke_clan_female" GENDER = female }
							}
						}

						### COUNTY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_county } }
							if = { # Magnar
								limit = { culture ?= { has_name_list = name_list_thenn } }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_count" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "magnar_count" GENDER = female }
							}
							else_if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "count_clan_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "count_clan_female" GENDER = female }
							}
						}

						### BARONY ###
						else_if = {
							limit = { $SCOPED_TITLE$ ?= { tier = tier_barony } }
							if = {
								limit = { always = yes }

								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_clan_male" GENDER = male }
								set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_clan_female" GENDER = female }
							}
						}
					}

					### LANDLESS ADVENTURER ###
					else_if = {
						limit = {
							OR = {
								agot_localization_government_trigger = { GOV = landless_adventurer SPOUSE_SCOPE = scope:spouse_scope }
								$SCOPED_TITLE$ ?= { is_landless_type_title = yes }
							}
						}
						if = { # Captain-General
							limit = {
								$SCOPED_TITLE$ ?= title:d_laamp_merc_golden_company
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "captain_general" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "captain_general" GENDER = female }
						}
						else_if = { # Ser
							limit = {
								is_agot_knight_trigger = yes
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SER" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SER" GENDER = female }
						}
						else_if = { # Captain
							limit = {
								has_realm_law = camp_purpose_mercenaries
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_camp_mercenaries" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duchess_landless_adventurer_camp_mercenaries" GENDER = female }
						}
						else_if = { # Master/Mistress
							limit = {
								has_realm_law = camp_purpose_wanderers
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_male" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_holder_female" GENDER = female }
						}
						else_if = { # Master/Mistress
							limit = {
								has_realm_law = camp_purpose_scholars
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_male" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_holder_female" GENDER = female }
						}
						else_if = { # Leader
							limit = {
								has_realm_law = camp_purpose_explorers
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_camp_explorers" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duchess_landless_adventurer_camp_explorers" GENDER = female }
						}
						else_if = { # Boss
							limit = {
								has_realm_law = camp_purpose_brigands
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_camp_brigands" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duchess_landless_adventurer_camp_brigands" GENDER = female }
						}
						else_if = { # Lord/Lady
							limit = {
								has_realm_law = camp_purpose_legitimists
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lord_non_paramount" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lady_non_paramount" GENDER = female }
						}
						else_if = {
							limit = {
								trigger_if = {
									limit = { exists = primary_title }
									primary_title ?= $SCOPED_TITLE$
								}
								trigger_else = { always = no }
							}

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_male_camp" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_female_camp" GENDER = female }
						}
						else_if = {
							limit = { always = yes }

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_male" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_spouse_landless_adventurer_holder_female" GENDER = female }
						}
						else_if = {
							limit = { always = yes }

							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_male_camp" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "default_duke_landless_adventurer_female_camp" GENDER = female }
						}
					}

					### FALLBACK WARNING ###
					else = {
						$SCOPE$ ?= {
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
							set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
						}
					}
				}

				### SCIONS ###
				else_if = {
					limit = { scope:scion_status ?= flag:royal }

					if = { # Prince/Princess
						limit = {
							trigger_if = {
								limit = { exists = primary_title }
								NOT = { primary_title ?= $SCOPED_TITLE$ }
							}
							OR = {
								trigger_if = {
									limit = { exists = scope:scion_1_title }
									scope:scion_1_title ?= $SCOPED_TITLE$
								}
								trigger_if = {
									limit = { exists = scope:scion_2_title }
									scope:scion_2_title ?= $SCOPED_TITLE$
								}
								trigger_if = {
									limit = { exists = scope:scion_3_title }
									scope:scion_3_title ?= $SCOPED_TITLE$
								}
							}
						}

						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "prince" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "princess" GENDER = female }
					}

					### FALLBACK WARNING ###
					else = {
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
					}
				}

				### FALLBACK WARNING ###
				else = {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
				}
			}

			### HIDDEN TITLED CHARACTERS ###
			else_if = {
				limit = { is_title_hidden_character = yes }

				if = {
					limit = { scope:high_septon ?= yes }
					if = {
						limit = { $SCOPED_TITLE$ ?= title:k_the_most_devout }

						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "HIGH_SEPTON" GENDER = male }
					}

					### FALLBACK WARNING ###
					else = {
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
					}
				}
				else = {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "BLANK_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "BLANK_TITLE_LOC" GENDER = female }
				}
			}

			### FALLBACK WARNING ###
			else = {
				$SCOPE$ ?= {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
				}
			}
		}
	}
	else = {
		if = { # Septon/Septa
			limit = { scope:septon ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SEPTON" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SEPTA" GENDER = female }
		}
		else_if = { # Maester/Archmaester/Grandmaester
			limit = { scope:maester ?= yes }

			if = {
				limit = {
					has_inactive_trait = grandmaester
					NOT = {
						has_trait = archmaester
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "grand_maester" GENDER = male }
			}
			else_if = {
				limit = {
					has_trait = archmaester
					NOT = {
						has_inactive_trait = grandmaester
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ArchaesterTitle" GENDER = male }
			}
			else_if = {
				limit = {
					NOR = {
						has_trait = archmaester
						has_inactive_trait = grandmaester
					}
					OR = {
						AND = {
							has_trait = maester
							has_trait_xp = {
								trait = maester
								value = 100
							}
						}
						has_trait = maester_historical
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MaesterTitle" GENDER = male }
			}
			else_if = {
				limit = {
					NOT = { has_trait = archmaester }
					has_trait = maester
					has_trait_xp = {
						trait = maester
						value = 0
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MaesterTitle" GENDER = male }
			}
			else = {
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "AcolyteTitle" GENDER = male }
			}
		}
		else_if = { # Silent Sister
			limit = { scope:silent_sister ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SILENT_SISTER" GENDER = female }
		}
		else_if = { # Ser
			limit = { scope:knighthood ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SER" GENDER = ALL }
		}
		else_if = { # Councillor
			limit = { scope:councillor ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lord_non_paramount" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lady_non_paramount" GENDER = female }
		}
		else_if = { # Lord/Lady (Scion Honorific)
			limit = { scope:scion_status ?= flag:active }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "LORD_SCION" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "LADY_SCION" GENDER = female }
		}

		else_if = { # Temple barony holders without cached AGOT ruler data (AM/AGOT+/LoV Compatch delta)
			limit = {
				exists = primary_title
				primary_title = {
					tier = tier_barony
					title_province = { has_holding_type = church_holding }
				}
			}
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_male" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "baron_theocracy_female" GENDER = female }
		}

		### FALLBACK WARNING ###
		else = {
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
		}
	}
