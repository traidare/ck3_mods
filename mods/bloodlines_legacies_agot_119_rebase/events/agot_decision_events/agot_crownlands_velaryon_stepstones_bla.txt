agot_crownlands_bla.0198 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		dynasty = dynasty:dynn_Velaryon
		has_title = title:d_driftmark

		OR = {
			############################################
			# Story has never begun
			############################################

			title:d_driftmark = {
				NOT = {
					has_variable = agot_velaryon_stepstones_story_begun_bla
				}
			}

			############################################
			# Campaign active, but ambition needs
			# to be restored to current ruler
			############################################

			AND = {
				title:d_driftmark = {
					has_variable = agot_velaryon_stepstones_campaign_active_bla
				}

				NOT = {
					has_character_flag = agot_velaryon_stepstones_ambition_bla
				}

				NOT = {
					has_title = title:k_the_stepstones
				}
			}

			############################################
			# Bloodstone has been taken
			############################################

			AND = {
				has_title = title:c_bloodstone

				has_character_flag = agot_velaryon_stepstones_ambition_bla

				title:d_driftmark = {
					NOT = {
						has_variable = agot_velaryon_stepstones_bloodstone_taken_bla
					}
				}
			}

			############################################
			# Kingdom has been formed and Bloodstone
			# has already been acknowledged
			############################################

			AND = {
				has_title = title:k_the_stepstones
				has_title = title:c_bloodstone

				has_character_flag = agot_velaryon_stepstones_ambition_bla

				title:d_driftmark = {
					has_variable = agot_velaryon_stepstones_bloodstone_taken_bla
				}

				title:k_the_stepstones = {
					NOT = {
						has_variable = agot_velaryon_stepstones_founded_bla
					}
				}
			}

			############################################
			# First stability check
			############################################

			AND = {
				has_title = title:k_the_stepstones

				title:k_the_stepstones = {
					has_variable = agot_velaryon_stepstones_founded_bla

					NOT = {
						has_variable = agot_velaryon_stepstones_stability_checked_bla
					}
				}
			}

			############################################
			# Long rule underway
			############################################

			AND = {
				has_title = title:k_the_stepstones

				title:k_the_stepstones = {
					has_variable = agot_velaryon_stepstones_long_rule_begun_bla

					NOT = {
						has_variable = agot_velaryon_stepstones_legacy_completed_bla
					}
				}
			}
		}
	}

	immediate = {

		############################################
		# Begin the Stepstones ambition
		############################################

		if = {
			limit = {
				title:d_driftmark = {
					NOT = {
						has_variable = agot_velaryon_stepstones_story_begun_bla
					}
				}
			}

			title:d_driftmark = {
				set_variable = {
					name = agot_velaryon_stepstones_story_begun_bla
					value = 1
				}

				set_variable = {
					name = agot_velaryon_stepstones_campaign_active_bla
					value = 1
				}
			}

			add_character_flag = agot_velaryon_stepstones_ambition_bla

			trigger_event = agot_crownlands_bla.0150
		}

		############################################
		# Bloodstone has been taken
		#
		# IMPORTANT:
		# This must come BEFORE the kingdom check
		# so 0155 cannot be skipped.
		############################################

		else_if = {
			limit = {
				has_title = title:c_bloodstone
				has_character_flag = agot_velaryon_stepstones_ambition_bla

				title:d_driftmark = {
					NOT = {
						has_variable = agot_velaryon_stepstones_bloodstone_taken_bla
					}
				}
			}

			trigger_event = agot_crownlands_bla.0155
		}

		############################################
		# The Stepstones have been united
		############################################

		else_if = {
			limit = {
				has_title = title:k_the_stepstones
				has_title = title:c_bloodstone

				has_character_flag = agot_velaryon_stepstones_ambition_bla

				title:d_driftmark = {
					has_variable = agot_velaryon_stepstones_bloodstone_taken_bla
				}

				title:k_the_stepstones = {
					NOT = {
						has_variable = agot_velaryon_stepstones_founded_bla
					}
				}
			}

			trigger_event = agot_crownlands_bla.0156
		}

		############################################
		# Restore the Velaryon ambition to a
		# successor while the campaign is active
		############################################

		else_if = {
			limit = {
				title:d_driftmark = {
					has_variable = agot_velaryon_stepstones_campaign_active_bla
				}

				NOT = {
					has_character_flag = agot_velaryon_stepstones_ambition_bla
				}

				NOT = {
					has_title = title:k_the_stepstones
				}
			}

			add_character_flag = agot_velaryon_stepstones_ambition_bla
		}

		############################################
		# First year of the new kingdom
		############################################

		else_if = {
			limit = {
				has_title = title:k_the_stepstones

				title:k_the_stepstones = {
					has_variable = agot_velaryon_stepstones_founded_bla

					NOT = {
						has_variable = agot_velaryon_stepstones_stability_checked_bla
					}
				}
			}

			title:k_the_stepstones = {
				set_variable = {
					name = agot_velaryon_stepstones_stability_checked_bla
					value = 1
				}
			}

			trigger_event = agot_crownlands_bla.0160
		}

		############################################
		# Long-rule progression
		############################################

		else_if = {
			limit = {
				has_title = title:k_the_stepstones

				title:k_the_stepstones = {
					has_variable = agot_velaryon_stepstones_long_rule_begun_bla

					NOT = {
						has_variable = agot_velaryon_stepstones_legacy_completed_bla
					}
				}
			}

			title:k_the_stepstones = {
				change_variable = {
					name = agot_velaryon_stepstones_years_since_founding_bla
					add = 1
				}
			}

			if = {
				limit = {
					title:k_the_stepstones = {
						var:agot_velaryon_stepstones_years_since_founding_bla >= 10
					}
				}

				trigger_event = agot_crownlands_bla.0163
			}
		}
	}
}

agot_crownlands_bla.0151 = {
	type = character_event
	title = agot_crownlands_bla.0151.t
	desc = agot_crownlands_bla.0151.desc
	theme = martial

	override_background = {
		reference = docks
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:stepstones_sellsail_captain_bla
		animation = personality_bold
	}

	option = { # Pay them properly.
		name = agot_crownlands_bla.0151.a

		trigger = {
			gold >= 150
		}

		remove_short_term_gold = 150

		scope:stepstones_sellsail_captain_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 35
			}

			add_character_modifier = {
				modifier = agot_stepstones_well_paid_captain_bla
				years = 10
			}
		}

		add_character_flag = agot_velaryon_stepstones_captain_paid_bla

		trigger_event = {
			id = agot_crownlands_bla.0152
			days = { 14 30 }
		}
	}

	option = { # Promise land after victory.
		name = agot_crownlands_bla.0151.b

		scope:stepstones_sellsail_captain_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		add_character_flag = agot_velaryon_stepstones_captain_promised_land_bla

		trigger_event = {
			id = agot_crownlands_bla.0152
			days = { 14 30 }
		}

	}

	option = { # Test their worth first.
		name = agot_crownlands_bla.0151.c

		duel = {
			skill = martial
			value = medium_skill_rating

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0151.c.success

					add_martial_lifestyle_xp = 250

					scope:stepstones_sellsail_captain_bla = {
						add_opinion = {
							target = root
							modifier = impressed_opinion
							opinion = 25
						}

						add_character_modifier = {
							modifier = agot_stepstones_proven_captain_bla
							years = 10
						}
					}

					add_character_flag = agot_velaryon_stepstones_captain_proven_bla
				}
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0151.c.failure

					add_martial_lifestyle_xp = 100

					scope:stepstones_sellsail_captain_bla = {
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -15
						}
					}

					add_character_flag = agot_velaryon_stepstones_captain_unproven_bla
				}
			}
		}

		trigger_event = {
			id = agot_crownlands_bla.0152
				days = { 14 30 }
		}
	}

	option = { # We need no sellsails.
		name = agot_crownlands_bla.0151.d

		add_prestige = 75

		scope:stepstones_sellsail_captain_bla = {
			set_relation_potential_rival = {
				target = root
				reason = rival_disagreement
			}

			move_to_pool = yes
		}

		add_character_flag = agot_velaryon_stepstones_captain_refused_bla

		trigger_event = {
			id = agot_crownlands_bla.0152
			days = { 14 30 }
		}

	}
}

		}

		trigger_event = {
			id = agot_crownlands_bla.0154
			months = { 6 10 }
		}
	}

	############################################
	# Harsh Requisitions
	############################################

	option = {
		name = agot_crownlands_bla.0152.e

		add_dread = 5

		add_character_flag = agot_velaryon_stepstones_requisitioned_supply_bla

		add_character_modifier = {
			modifier = agot_velaryon_stepstones_harsh_requisitions_bla

			id = agot_crownlands_bla.0154
			months = { 6 10 }
		}

				}
			}
		}

		trigger_event = {
			id = agot_crownlands_bla.0154
			months = { 6 10 }
