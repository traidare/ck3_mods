agot_oldstones_bla.0003 = {
	type = character_event

	title = agot_oldstones_bla.0003.t

	desc = {
		desc = agot_oldstones_bla.0003.desc

		triggered_desc = {
			trigger = { has_character_flag = oldstones_collapsed_crypt_bla }
			desc = agot_oldstones_bla.0003.desc_collapsed
		}

		triggered_desc = {
			trigger = { has_character_flag = oldstones_burial_hall_bla }
			desc = agot_oldstones_bla.0003.desc_burial
		}

			trigger = { has_character_flag = oldstones_flooded_vault_bla }

			trigger = { has_character_flag = oldstones_royal_chamber_bla }

		}
	}

	theme = dynasty

	override_background = {
		reference = ce1_catacombs
	}

	left_portrait = {
		character = root
		animation = fear
	}

	immediate = {
		if = {
			limit = {
				NOT = { exists = global_var:oldstones_excavation_stage }
			}
			set_global_variable = {
				name = oldstones_excavation_stage
				value = flag:first
			}
		}
		else_if = {
			limit = {
				global_var:oldstones_excavation_stage = flag:first
			}
			set_global_variable = {
				name = oldstones_excavation_stage
				value = flag:second
			}
		}
		else = {
			set_global_variable = {
				name = oldstones_excavation_stage
				value = flag:third
			}
		}

		if = {
			limit = { has_character_flag = oldstones_collapsed_crypt_bla }

			random_list = {
				65 = { add_character_flag = oldstones_ruined_find_bla }
				25 = { add_character_flag = oldstones_historical_fragment_bla }
				8 = { add_character_flag = oldstones_identifiable_relic_bla }
				2 = { add_character_flag = oldstones_royal_relic_bla }

				1 = {
					trigger = { NOT = { exists = global_var:oldstones_hammer_recovered } }
					add_character_flag = oldstones_monumental_discovery_bla
				}

				1 = {
					trigger = {
						NOT = { exists = global_var:oldstones_royal_crypt_discovered }
						global_var:oldstones_excavation_stage = flag:third
					}
					add_character_flag = oldstones_king_beneath_bla
				}
			}
		}

		else_if = {
			limit = { has_character_flag = oldstones_burial_hall_bla }

			random_list = {
				45 = { add_character_flag = oldstones_ruined_find_bla }
				30 = { add_character_flag = oldstones_historical_fragment_bla }
				17 = { add_character_flag = oldstones_identifiable_relic_bla }
				6 = { add_character_flag = oldstones_royal_relic_bla }

				2 = {
					trigger = { NOT = { exists = global_var:oldstones_hammer_recovered } }
					add_character_flag = oldstones_monumental_discovery_bla
				}

				1 = {
					trigger = {
						NOT = { exists = global_var:oldstones_royal_crypt_discovered }
						global_var:oldstones_excavation_stage = flag:third
					}
					add_character_flag = oldstones_king_beneath_bla
				}
			}
		}

		else_if = {
			limit = { has_character_flag = oldstones_flooded_vault_bla }

			random_list = {
				50 = { add_character_flag = oldstones_ruined_find_bla }
				30 = { add_character_flag = oldstones_historical_fragment_bla }
				14 = { add_character_flag = oldstones_identifiable_relic_bla }
				4 = { add_character_flag = oldstones_royal_relic_bla }

				2 = {
					trigger = { NOT = { exists = global_var:oldstones_hammer_recovered } }
					add_character_flag = oldstones_monumental_discovery_bla
				}

				1 = {
					trigger = {
						NOT = { exists = global_var:oldstones_royal_crypt_discovered }
						global_var:oldstones_excavation_stage = flag:third
					}
					add_character_flag = oldstones_king_beneath_bla
				}
			}
		}

		else_if = {
			limit = { has_character_flag = oldstones_royal_chamber_bla }

			random_list = {
				30 = { add_character_flag = oldstones_ruined_find_bla }
				25 = { add_character_flag = oldstones_historical_fragment_bla }
				25 = { add_character_flag = oldstones_identifiable_relic_bla }
				14 = { add_character_flag = oldstones_royal_relic_bla }

				4 = {
					trigger = { NOT = { exists = global_var:oldstones_hammer_recovered } }
					add_character_flag = oldstones_monumental_discovery_bla
				}

				2 = {
					trigger = {
						NOT = { exists = global_var:oldstones_royal_crypt_discovered }
						global_var:oldstones_excavation_stage = flag:third
					}
					add_character_flag = oldstones_king_beneath_bla
				}
			}
		}
	}

	option = {
		name = agot_oldstones_bla.0003.a

		trigger_event = {
			id = agot_oldstones_bla.0004
			days = { 10 10 }
		}
	}
}
