			flag = agot_crabb_blood_clarence_cooldown_bla

			location = root.capital_province
			age = { 22 32 }

			martial = { 10 14 }
			prowess = { 16 22 }

			trait = strong
			trait = brave

			save_scope_as = crabb_clarence_descendant_bla
		}
	}

	option = { # Bring him into the household

		hidden_effect = {
			add_courtier = scope:crabb_clarence_descendant_bla
		}

		add_prestige = 100
		add_martial_lifestyle_xp = 125

		custom_tooltip = agot_crownlands_bla.0504.a.tt
	}

	option = { # Publicly encourage the story
		name = agot_crownlands_bla.0504.b

		add_prestige = 175
		dynasty ?= { add_dynasty_prestige = 75 }

		scope:crabb_clarence_descendant_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 40
			}
		}

		hidden_effect = {
			add_courtier = scope:crabb_clarence_descendant_bla
		}

		custom_tooltip = agot_crownlands_bla.0504.b.tt
	}

	option = { # Test him
		name = agot_crownlands_bla.0504.c

		trigger = {
			prowess >= 8
		}

		skill = prowess

		duel = {
			skill = prowess
			value = scope:crabb_clarence_descendant_bla.prowess

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1
				}

				add_prestige = 250
				add_martial_lifestyle_xp = 175

				hidden_effect = {
					add_courtier = scope:crabb_clarence_descendant_bla
				}

				custom_tooltip = agot_crownlands_bla.0504.c.success
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -1
				}

				add_martial_lifestyle_xp = 125

				hidden_effect = {
					add_courtier = scope:crabb_clarence_descendant_bla
				}

				custom_tooltip = agot_crownlands_bla.0504.c.failure
			}
		}
	}

	option = { # Reject the genealogy
		name = agot_crownlands_bla.0504.d

		add_prestige = 75
