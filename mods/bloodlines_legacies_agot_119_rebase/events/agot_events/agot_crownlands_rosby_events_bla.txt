agot_crownlands_bla.0999 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Rosby

		NOT = {
			has_character_flag = rosby_any_event_recently_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = rosby_any_event_recently_bla
			years = 3
		}

		random_list = {
			10 = { trigger_event = agot_crownlands_bla.0900 }
			10 = { trigger_event = agot_crownlands_bla.0901 }
			10 = { trigger_event = agot_crownlands_bla.0902 }
			8 = { trigger_event = agot_crownlands_bla.0903 }

			90 = { }
		}
	}
}

agot_crownlands_bla.0901 = {
	type = character_event
	title = agot_crownlands_bla.0901.t
	desc = agot_crownlands_bla.0901.desc
	theme = intrigue

	override_background = {
		reference = throne_room
	}

	left_portrait = {
		character = root
		animation = personality_suspicious
	}

	right_portrait = {
		character = scope:rosby_fugitive_courtier_bla
		animation = personality_fear
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Rosby

		NOT = {
			has_character_flag = agot_rosby_guest_capital_cooldown_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = agot_rosby_guest_capital_cooldown_bla
			years = 10
		}

		create_character = {
			location = root.capital_province
			age = { 25 45 }
			dynasty = none
			gender_female_chance = 50

			culture = root.culture
			faith = root.faith

			diplomacy = { 9 14 }
			intrigue = { 11 16 }
			stewardship = { 7 12 }

			trait = shrewd

			save_scope_as = rosby_fugitive_courtier_bla
		}
	}

	option = { # Shelter them
		name = agot_crownlands_bla.0901.a

		hidden_effect = {
			add_courtier = scope:rosby_fugitive_courtier_bla
		}

		add_intrigue_lifestyle_xp = 150

		scope:rosby_fugitive_courtier_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 40
			}
		}

		custom_tooltip = agot_crownlands_bla.0901.a.tt
	}

	option = { # Hear their secrets
		name = agot_crownlands_bla.0901.b

		add_intrigue_lifestyle_xp = 200
		add_prestige = 75

		random_list = {
			65 = {
				add_gold = 75
				custom_tooltip = agot_crownlands_bla.0901.b.success
			}

			35 = {
				custom_tooltip = agot_crownlands_bla.0901.b.failure
			}
		}
	}

	option = { # Return them
		name = agot_crownlands_bla.0901.c

		add_diplomacy_lifestyle_xp = 125
		add_prestige = 100

		if = {
			limit = {
				exists = liege
			}

			liege = {
				add_opinion = {
					target = root
					modifier = grateful_opinion
					opinion = 20
				}
			}
		}
	}

	option = { # Send them onward
		name = agot_crownlands_bla.0901.d

		add_diplomacy_lifestyle_xp = 100
		add_piety = 75

		custom_tooltip = agot_crownlands_bla.0901.d.tt
	}
}
