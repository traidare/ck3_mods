agot_crownlands_bla.0301 = {
	type = character_event

	theme = stewardship

		character = root

		is_available_adult = yes

		NOT = {
			has_character_flag = agot_darklyn_dun_fort_cooldown_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = agot_darklyn_dun_fort_cooldown_bla
			years = 9
		}
	}

	############################################
	# The Dun Fort comes first.
	############################################

	option = {
		name = agot_crownlands_bla.0301.a

		trigger = {
			gold >= 125
		}

		remove_short_term_gold = 125
		add_martial_lifestyle_xp = 200
		add_prestige = 75

		add_character_modifier = {
			modifier = agot_darklyn_dun_fort_restored_bla
			years = 8
		}

		add_character_flag = agot_darklyn_dun_fort_strengthened_bla
	}

	############################################
	# Duskendale is our true strength.
	############################################

	option = {
		name = agot_crownlands_bla.0301.b

		trigger = {
			gold >= 125
		}

		remove_short_term_gold = 125
		add_stewardship_lifestyle_xp = 250

		add_character_modifier = {
			modifier = agot_darklyn_duskendale_investment_bla
			years = 8
		}

		add_character_flag = agot_darklyn_duskendale_before_dun_fort_bla
	}

	############################################
	# Fund both.
	############################################

	option = {
		name = agot_crownlands_bla.0301.c

		trigger = {
			gold >= 200
		}

		remove_short_term_gold = 200

		duel = {
			skill = stewardship
			value = medium_skill_rating

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0301.c.success

					add_stewardship_lifestyle_xp = 350
					add_prestige = 125

					add_character_modifier = {
						modifier = agot_darklyn_dun_fort_and_town_bla
						years = 10
					}

					add_character_flag = agot_darklyn_dun_fort_balanced_bla
				}
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0301.c.failure

					add_stewardship_lifestyle_xp = 100
					add_stress = minor_stress_gain

					add_character_modifier = {
						modifier = agot_darklyn_overextended_works_bla
						years = 4
					}

					add_character_flag = agot_darklyn_dun_fort_overextended_bla
				}
			}
		}
	}

	############################################
	# Neither. Preserve what already stands.
	############################################

	option = {
		name = agot_crownlands_bla.0301.d

		add_gold = 50
		add_learning_lifestyle_xp = 150

		add_character_flag = agot_darklyn_dun_fort_preserved_bla
	}
}

agot_crownlands_bla.0303 = {
	type = character_event
	title = agot_crownlands_bla.0303.t

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_character_flag = agot_darklyn_guilds_result_prosperity_bla
				}
				desc = agot_crownlands_bla.0303.desc_prosperity
			}

			triggered_desc = {
				trigger = {
					has_character_flag = agot_darklyn_guilds_result_stability_bla
				}
				desc = agot_crownlands_bla.0303.desc_stability
			}

			triggered_desc = {
				trigger = {
					has_character_flag = agot_darklyn_guilds_result_compromise_bla
				}
				desc = agot_crownlands_bla.0303.desc_compromise
			}

			desc = agot_crownlands_bla.0303.desc_dispute
		}
	}

	theme = stewardship

	override_background = {
		reference = market
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {

		############################################
		# Merchant policy
		############################################

		if = {
			limit = {
				has_character_flag = agot_darklyn_guilds_merchants_bla
			}

			random_list = {
				65 = {
					modifier = {
						add = 20
						has_character_flag = agot_darklyn_duskendale_before_dun_fort_bla
					}

					modifier = {
						add = 10
						has_character_flag = agot_darklyn_dun_fort_balanced_bla
					}

					add_character_flag = agot_darklyn_guilds_result_prosperity_bla
				}

				35 = {
					add_character_flag = agot_darklyn_guilds_result_dispute_bla
				}
			}
		}

		############################################
		# Craftsmen policy
		############################################

		else_if = {
			limit = {
				has_character_flag = agot_darklyn_guilds_craftsmen_bla
			}

			random_list = {
				60 = {
					add_character_flag = agot_darklyn_guilds_result_stability_bla
				}

				40 = {
					add_character_flag = agot_darklyn_guilds_result_dispute_bla
				}
			}
		}

		############################################
		# Old privileges
		############################################

		else_if = {
			limit = {
				has_character_flag = agot_darklyn_guilds_old_privileges_bla
			}

			random_list = {
				65 = {
					modifier = {
						add = 15
						has_character_flag = agot_darklyn_dun_fort_preserved_bla
					}

					add_character_flag = agot_darklyn_guilds_result_stability_bla
				}

				35 = {
					add_character_flag = agot_darklyn_guilds_result_dispute_bla
				}
			}
		}

		############################################
		# Compromise
		############################################

		else_if = {
			limit = {
				has_character_flag = agot_darklyn_guilds_compromise_bla
			}

			random_list = {
				55 = {
					modifier = {
						add = 25
						has_character_flag = agot_darklyn_dun_fort_balanced_bla
					}

					add_character_flag = agot_darklyn_guilds_result_compromise_bla
				}

				45 = {
					add_character_flag = agot_darklyn_guilds_result_dispute_bla
				}
			}
		}
	}

	############################################
	# Prosperity
	############################################

	option = {
		name = agot_crownlands_bla.0303.a

		trigger = {
			has_character_flag = agot_darklyn_guilds_result_prosperity_bla
		}

		add_gold = 125
		add_stewardship_lifestyle_xp = 250

		add_character_modifier = {
			modifier = agot_darklyn_open_duskendale_trade_bla
			years = 7
		}

		add_character_flag = agot_darklyn_merchants_grateful_bla
	}

	############################################
	# Stability
	############################################

	option = {
		name = agot_crownlands_bla.0303.b

		trigger = {
			has_character_flag = agot_darklyn_guilds_result_stability_bla
		}

		add_diplomacy_lifestyle_xp = 250
		add_prestige = 100

		add_character_modifier = {
			modifier = agot_darklyn_guild_compact_bla
			years = 7
		}

		add_character_flag = agot_darklyn_guilds_loyal_bla
	}

	############################################
	# Successful compromise
	############################################

	option = {
		name = agot_crownlands_bla.0303.c

		trigger = {
			has_character_flag = agot_darklyn_guilds_result_compromise_bla
		}

		add_diplomacy_lifestyle_xp = 250
		add_stewardship_lifestyle_xp = 250

		add_character_modifier = {
			modifier = agot_darklyn_balanced_guild_charter_bla
			years = 8
		}

		add_character_flag = agot_darklyn_guild_compromise_success_bla
	}

	############################################
	# Dispute
	############################################

	option = {
		name = agot_crownlands_bla.0303.d

		trigger = {
			has_character_flag = agot_darklyn_guilds_result_dispute_bla
		}

		add_stress = minor_stress_gain

		add_character_modifier = {
			modifier = agot_darklyn_guild_resentment_bla
			years = 4
		}

		add_character_flag = agot_darklyn_guild_dispute_bla
	}

	after = {

		remove_character_flag = agot_darklyn_guilds_merchants_bla
		remove_character_flag = agot_darklyn_guilds_craftsmen_bla
		remove_character_flag = agot_darklyn_guilds_old_privileges_bla
		remove_character_flag = agot_darklyn_guilds_compromise_bla
	}
}
