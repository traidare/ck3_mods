agot_riverlands_bla.0199 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		OR = {
			dynasty = dynasty:dynn_Blackwood
			dynasty = dynasty:dynn_Bracken
		}

		blackwood_bracken_rival_exists_trigger = yes

		NOT = { has_character_flag = blackwood_bracken_any_feud_event_recently_bla }
	}

	immediate = {

		add_character_flag = {
			flag = blackwood_bracken_any_feud_event_recently_bla
			years = 3
		}

		random_list = {
			15 = { trigger_event = agot_riverlands_bla.0200 }
			15 = { trigger_event = agot_riverlands_bla.0201 }
			15 = { trigger_event = agot_riverlands_bla.0202 }
			15 = { trigger_event = agot_riverlands_bla.0203 }
			15 = { trigger_event = agot_riverlands_bla.0204 }

			10 = {
				trigger = { var:blackwood_bracken_feud_intensity_bla >= 3 }

			}

				trigger = { var:blackwood_bracken_feud_intensity_bla >= 4 }
				trigger_event = agot_riverlands_bla.0206
			}

			5 = {

			}

agot_riverlands_bla.0200 = {
	type = character_event
	title = agot_riverlands_bla.0200.t
	desc = agot_riverlands_bla.0200.desc
	theme = rival_relation

	override_background = {
		reference = agot_river
	}

	left_portrait = {
		character = root
		animation = anger
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		OR = {
			dynasty = dynasty:dynn_Blackwood
			dynasty = dynasty:dynn_Bracken
		}

		blackwood_bracken_rival_exists_trigger = yes

		NOT = { has_character_flag = blackwood_bracken_any_feud_event_recently_bla }
		NOT = { has_character_flag = blackwood_bracken_border_dead_recently_bla }
	}

	immediate = {
		add_character_flag = { flag = blackwood_bracken_any_feud_event_recently_bla years = 2 }
		add_character_flag = { flag = blackwood_bracken_border_dead_recently_bla years = 5 }

		random_living_character = {
			limit = {
				is_house_head = yes
				is_landed = yes
				NOT = { this = root }

				OR = {
					AND = {
						root = { dynasty = dynasty:dynn_Blackwood }
						dynasty = dynasty:dynn_Bracken
					}
					AND = {
						root = { dynasty = dynasty:dynn_Bracken }
						dynasty = dynasty:dynn_Blackwood
					}
				}
			}

			save_scope_as = feud_rival_bla
		}
	}
	option = {
		name = agot_riverlands_bla.0200.a

		add_prestige = 100
		add_dread = 10

		change_variable = {
			name = blackwood_bracken_feud_intensity_bla
			add = 1
		}

		scope:feud_rival_bla = {
			add_opinion = {
				target = root
				modifier = insulted_opinion
				opinion = -30
			}
		}

		add_character_modifier = {
			modifier = blackwood_bracken_rekindled_hatred_bla
			years = 5
		}
	}

	option = {
		name = agot_riverlands_bla.0200.b

		add_piety = 75
		add_prestige = 25

		change_variable = {
			name = blackwood_bracken_feud_intensity_bla
			add = -1
		}

		scope:feud_rival_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}

		add_character_modifier = {
			modifier = blackwood_bracken_fragile_truce_bla
			years = 3
		}
	}

	option = {
		name = agot_riverlands_bla.0200.c

		trigger = {
			OR = {
				intrigue >= 10
				has_trait = deceitful
				has_trait = schemer
			}
		}

		random_list = {
			70 = {
				change_variable = {
					name = blackwood_bracken_feud_intensity_bla
					add = 2
				}

				add_hook = {
					target = scope:feud_rival_bla
					type = fabrication_hook
				}

				scope:feud_rival_bla = {
					add_opinion = {
						target = root
						modifier = suspicious_opinion
						opinion = -40
					}
				}

				custom_tooltip = agot_riverlands_bla.0200.c.success_tt
			}

			30 = {
				add_prestige = -75

				scope:feud_rival_bla = {
					add_opinion = {
						target = root
						modifier = exposed_criminal_opinion
						opinion = -50
					}
				}

				custom_tooltip = agot_riverlands_bla.0200.c.failure_tt
			}
		}
	}

	option = {
		name = agot_riverlands_bla.0200.d

		trigger = {
			OR = {
				martial >= 10
				prowess >= 8
				has_trait = wrathful
				has_trait = brave
			}
		}

		add_prestige = 150
		add_dread = 5

		change_variable = {
			name = blackwood_bracken_feud_intensity_bla
			add = 2
		}

		add_character_flag = {
			flag = blackwood_bracken_duel_demands_bla
			years = 2
		}

		scope:feud_rival_bla = {
			add_opinion = {
				target = root
				modifier = rival_opinion
				opinion = -40
			}
		}
	}
}
