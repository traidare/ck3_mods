agot_crownlands_bla.1099 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		dynasty = dynasty:dynn_Bar_Emmon

		NOT = {
			has_character_flag = bar_emmon_any_event_recently_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = bar_emmon_any_event_recently_bla
			years = 3
		}

		random_list = {
			10 = { trigger_event = agot_crownlands_bla.1000 }
			10 = { trigger_event = agot_crownlands_bla.1001 }
			10 = { trigger_event = agot_crownlands_bla.1002 }

			80 = { }
		}
	}
}

agot_crownlands_bla.1002 = {

	theme = dynasty

	override_background = {
		reference = bp3_coast
	}

	left_portrait = {
		character = root
		animation = personality_honorable
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		dynasty = dynasty:dynn_Bar_Emmon

		NOT = {
			has_character_flag = agot_bar_emmon_point_and_bay_cooldown_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = agot_bar_emmon_point_and_bay_cooldown_bla
			years = 15
		}
	}

	option = {
		name = agot_crownlands_bla.1002.a

		add_martial_lifestyle_xp = 175
		add_prestige = 150
		dynasty ?= { add_dynasty_prestige = 75 }

		add_character_modifier = {
			modifier = agot_bar_emmon_watchers_of_blackwater_bay_bla
			years = 10
		}
	}

	option = {
		name = agot_crownlands_bla.1002.b

		add_stewardship_lifestyle_xp = 175
		dynasty ?= { add_dynasty_prestige = 75 }

		add_character_modifier = {
			modifier = agot_bar_emmon_harbor_of_sharp_point_bla
			years = 10
		}
	}

	option = {
		name = agot_crownlands_bla.1002.c

		trigger = {
			exists = liege
		}

		add_diplomacy_lifestyle_xp = 175
		add_prestige = 125
		dynasty ?= { add_dynasty_prestige = 75 }

		liege = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}

		add_character_modifier = {
			modifier = agot_bar_emmon_first_light_of_warning_bla
			years = 10
		}
	}

	option = {
		name = agot_crownlands_bla.1002.d

		add_diplomacy_lifestyle_xp = 125
		add_stewardship_lifestyle_xp = 100
		dynasty ?= { add_dynasty_prestige = 100 }

		add_character_modifier = {
			modifier = agot_bar_emmon_lords_of_sharp_point_bla
			years = 10
		}
	}
}
