agot_crownlands_bla.0200 = {
	type = character_event
	title = agot_crownlands_bla.0200.t
	desc = agot_crownlands_bla.0200.desc
	theme = stewardship

	override_background = {
		reference = study
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:celtigar_tax_collector_bla
		animation = personality_rational
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Celtigar

		NOT = {
			has_character_flag = agot_celtigar_crabs_due_cooldown_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = agot_celtigar_crabs_due_cooldown_bla
			years = 7
		}

		create_character = {
			gender_female_chance = 0
			employer = root
			dynasty = none
			culture = root.culture
			faith = root.faith
			age = { 28 52 }

			stewardship = { 11 16 }
			intrigue = { 7 12 }

			trait = diligent

			save_scope_as = celtigar_tax_collector_bla
		}
		set_variable = {
			name = celtigar_tax_collector_bla
			value = scope:celtigar_tax_collector_bla
		}
	}

	############################################
	# Audit the books personally
	############################################

	option = {
		name = agot_crownlands_bla.0200.a

		duel = {
			skill = stewardship
			value = medium_skill_rating

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0200.a.success

					add_gold = 125
					add_stewardship_lifestyle_xp = 300
				}
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0200.a.failure

					add_stewardship_lifestyle_xp = 100

					scope:celtigar_tax_collector_bla = {
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -10
						}
					}
				}
			}
		}
	}

	############################################
	# Give the collector authority
	############################################

	option = {
		name = agot_crownlands_bla.0200.b

		scope:celtigar_tax_collector_bla = {
			add_character_modifier = {
				modifier = agot_celtigar_empowered_tax_collector_bla
				years = 8
			}

			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		add_character_flag = agot_celtigar_empowered_collector_bla
	}

	############################################
	# Take an immediate settlement
	############################################

	option = {
		name = agot_crownlands_bla.0200.c

		add_gold = 100

		scope:celtigar_tax_collector_bla = {
			move_to_pool = yes
		}

		add_character_flag = agot_celtigar_tax_compromise_bla
	}

	############################################
	# Find leverage in the accounts
	############################################

	option = {
		name = agot_crownlands_bla.0200.d

		trigger = {
			intrigue >= 11

			can_add_hook = {
				target = scope:celtigar_tax_collector_bla
				type = loyalty_hook
			}
		}

		add_intrigue_lifestyle_xp = 250

		add_hook = {
			target = scope:celtigar_tax_collector_bla
			type = loyalty_hook
		}

		scope:celtigar_tax_collector_bla = {
			add_opinion = {
				target = root
				modifier = suspicious_opinion
				opinion = -20
			}
		}

		add_character_flag = agot_celtigar_collector_bound_bla
	}
}

agot_crownlands_bla.0201 = {
	type = character_event

		animation = personality_rational
	}

	right_portrait = {
		character = scope:celtigar_tax_collector_bla
		animation = personality_rational
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Celtigar

		OR = {
			has_character_flag = agot_celtigar_empowered_collector_bla
			has_character_flag = agot_celtigar_collector_bound_bla
		}

		exists = var:celtigar_tax_collector_bla

		var:celtigar_tax_collector_bla = {
			is_alive = yes
		}

		NOT = {
			has_character_flag = agot_celtigar_taxman_followup_bla
		}
	}

	immediate = {
		add_character_flag = agot_celtigar_taxman_followup_bla

		var:celtigar_tax_collector_bla = {
			save_scope_as = celtigar_tax_collector_bla
		}
	}

	############################################
	# Let the collector continue
	############################################

	option = {
		name = agot_crownlands_bla.0201.a

		add_gold = 125

		scope:celtigar_tax_collector_bla = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}

			add_character_modifier = {
				modifier = agot_celtigar_relentless_collector_bla
				years = 8
			}
		}

		add_character_flag = agot_celtigar_taxman_unrestrained_bla
	}

	############################################
	# Rein the collector in
	############################################

	option = {
		name = agot_crownlands_bla.0201.b

		add_diplomacy_lifestyle_xp = 200

		scope:celtigar_tax_collector_bla = {
			remove_character_modifier = agot_celtigar_empowered_tax_collector_bla

			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -15
			}
		}

		add_character_flag = agot_celtigar_taxman_restrained_bla
	}

	############################################
	# Audit the collector
	############################################

	option = {
		name = agot_crownlands_bla.0201.c

		duel = {
			skill = stewardship
			value = medium_skill_rating

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0201.c.success

					add_gold = 100
					add_stewardship_lifestyle_xp = 300

					add_character_flag = agot_celtigar_taxman_corruption_found_bla
				}

				scope:celtigar_tax_collector_bla = {
					add_opinion = {
						target = root
						modifier = suspicious_opinion
						opinion = -25
					}
				}
			}

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}

				send_interface_toast = {
					title = agot_crownlands_bla.0201.c.failure

					add_stewardship_lifestyle_xp = 100

					scope:celtigar_tax_collector_bla = {
						add_opinion = {
							target = root
							modifier = offended_opinion
							opinion = -15
						}
					}
				}
			}
		}
	}

	############################################
	# The collector knows too much
	############################################

	option = {
		name = agot_crownlands_bla.0201.d

		trigger = {
			has_character_flag = agot_celtigar_collector_bound_bla
		}

		add_intrigue_lifestyle_xp = 250

		scope:celtigar_tax_collector_bla = {
			add_character_modifier = {
				modifier = agot_celtigar_keeper_of_the_ledgers_bla
				years = 10
			}
		}

		add_character_flag = agot_celtigar_taxman_kept_close_bla
	}

	############################################
	# Dismiss the collector
	############################################

	option = {
		name = agot_crownlands_bla.0201.e

		scope:celtigar_tax_collector_bla = {
			remove_character_modifier = agot_celtigar_empowered_tax_collector_bla

			set_relation_potential_rival = {
				target = root
				reason = rival_disagreement
			}

			move_to_pool = yes
		}

		add_prestige = 50

		add_character_flag = agot_celtigar_taxman_dismissed_bla
	}
}
