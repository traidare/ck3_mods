		name = agot_riverlands_bla.0650.b

		add_prestige = 50

		add_character_modifier = {
			modifier = piper_measured_defense_bla
			years = 3
		}
	}

	option = {
		name = agot_riverlands_bla.0650.c

		trigger = {
			OR = {
				martial >= 10
				has_trait = brave
				has_trait = gallant
				has_trait = strategist
			}
		}

		add_prestige = 200
		add_martial_lifestyle_xp = 500

		add_character_modifier = {
			modifier = piper_reckless_glory_bla
			years = 5
		}
	}

	option = {
		name = agot_riverlands_bla.0650.d

		trigger = {
			OR = {
				has_trait = craven
				has_trait = content
				has_trait = patient
			}
		}

		add_prestige = 25
		add_stress = -25

		add_character_modifier = {
			modifier = piper_measured_defense_bla
			years = 5
		}
	}

# =========================================================
# 0653 - A Knight of Summer
# =========================================================

agot_riverlands_bla.0653 = {
	type = character_event
	title = agot_riverlands_bla.0653.t
	desc = agot_riverlands_bla.0653.desc
	theme = martial

	override_background = {
		reference = army_camp
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Piper

		NOT = { has_character_flag = piper_any_event_recently_bla }
		NOT = { has_character_flag = piper_recent_knight_of_summer_bla }

		any_courtier = {
			is_adult = yes
			is_knight = yes
		}
	}

	immediate = {
		add_character_flag = {
			flag = piper_any_event_recently_bla
			years = 3
		}

		add_character_flag = {
			flag = piper_recent_knight_of_summer_bla
			years = 6
		}

		random_courtier = {
			limit = {
				is_adult = yes
				is_knight = yes
			}
			save_scope_as = piper_summer_knight
		}
	}

	right_portrait = {
		character = scope:piper_summer_knight
		animation = personality_bold
	}

	option = {
		name = agot_riverlands_bla.0653.a

		scope:piper_summer_knight = {

			add_trait = lifestyle_blademaster
	
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}

		add_prestige = 100
	}

	option = {
		name = agot_riverlands_bla.0653.b

		trigger = {
			OR = {
				martial >= 10
				has_trait = brave
				has_trait = ambitious
				has_trait = gallant
			}
		}

		scope:piper_summer_knight = {
			if = {
				limit = {
					NOT = { has_trait = brave }
				}
				add_trait = brave
			}

			add_trait = lifestyle_blademaster

			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 25
			}
		}

		add_prestige = 150
		add_martial_lifestyle_xp = 300
	}

	option = {
		name = agot_riverlands_bla.0653.c

		trigger = {
			OR = {
				has_trait = patient
				has_trait = diligent
				learning >= 10
			}
		}

		scope:piper_summer_knight = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 15
			}
		}

		add_learning_lifestyle_xp = 300
		add_prestige = 75
	}

	option = {
		name = agot_riverlands_bla.0653.d

		trigger = {
			OR = {
				has_trait = craven
				has_trait = content
				has_trait = compassionate
			}
		}

		scope:piper_summer_knight = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -15
			}
		}

		add_prestige = -50
		add_stress = -30
	}
}
