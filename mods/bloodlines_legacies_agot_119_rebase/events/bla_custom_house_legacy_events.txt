# ------------------------------------------------------------
# 0001 — House Age (framing + renown + splendor + unlock count)
# ------------------------------------------------------------
agot_bla_custom_house_legacy.0001 = {
	type = character_event
	title = agot_bla_custom_house_legacy.0001.t
	desc  = agot_bla_custom_house_legacy.0001.desc
	theme = family
	override_background = study

	trigger = {
		is_ai = no
		has_trait = bla_legacy_customizer
		exists = dynasty
		NOT = { dynasty = { has_variable = bla_custom_house_legacy_done } }
	}

	immediate = {

		 create_character = {
        		template = maester_character
        		location = root.capital_province
        		dynasty = none
        		save_scope_as = new_maester
    		}
		scope:new_maester = {
				add_character_flag = traveling_maester
		}

		dynasty = {
			# reset scores
			set_variable = { name = bla_score_coin value = 0 }
			set_variable = { name = bla_score_banner value = 0 }
			set_variable = { name = bla_score_steel value = 0 }
			set_variable = { name = bla_score_veil value = 0 }
			set_variable = { name = bla_score_candle value = 0 }

			# clear track flags
			remove_variable = bla_custom_track_coin
			remove_variable = bla_custom_track_banner
			remove_variable = bla_custom_track_steel
			remove_variable = bla_custom_track_veil
			remove_variable = bla_custom_track_candle
			remove_variable = bla_custom_track_mosaic

			# age + framing
			set_variable = { name = bla_custom_age_unlocks value = 0 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
		}
	}

	right_portrait = {
		character = scope:new_maester
		animation = throne_room_writer
	}

	# NEW: 0 unlocked, +0 renown
	option = {
		name = agot_bla_custom_house_legacy.0001.a1
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 0 }
			set_variable = { name = bla_custom_house_is_new value = 1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~100: 1 unlocked, +250 renown
	option = {
		name = agot_bla_custom_house_legacy.0001.a2
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 1 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 250
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~250: 2 unlocked, +1000 renown, -2 splendor level
	option = {
		name = agot_bla_custom_house_legacy.0001.a3
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 2 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 1000
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~500: 3 unlocked, +2250 renown, -2 splendor level
	option = {
		name = agot_bla_custom_house_legacy.0001.a4
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 3 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 2250
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~1000: 4 unlocked, +4000 renown, -3 splendor levels
	option = {
		name = agot_bla_custom_house_legacy.0001.a5
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 4 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 4000
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}
}

# ------------------------------------------------------------
# 0007 — Resolve winner + unlock N perks + summary tooltips
# ------------------------------------------------------------
agot_bla_custom_house_legacy.0007 = {
	type = character_event
	title = agot_bla_custom_house_legacy.0007.t
	desc  = agot_bla_custom_house_legacy.0007.desc
	theme = family
	override_background = {
		reference = study
	}
	right_portrait = {
		character = scope:new_maester
		animation = throne_room_writer
	}
	option = {
		name = agot_bla_custom_house_legacy.0007.a

		hidden_effect = {

			# Clear any prior build flags on the character
			remove_character_flag = bla_can_build_coin_dynasty_seat
			remove_character_flag = bla_can_build_banner_dynasty_seat
			remove_character_flag = bla_can_build_steel_dynasty_seat
			remove_character_flag = bla_can_build_veil_dynasty_seat
			remove_character_flag = bla_can_build_candle_dynasty_seat
			remove_character_flag = bla_can_build_mosaic_dynasty_seat

			dynasty = {

				remove_variable = bla_custom_track_coin
				remove_variable = bla_custom_track_banner
				remove_variable = bla_custom_track_steel
				remove_variable = bla_custom_track_veil
				remove_variable = bla_custom_track_candle
				remove_variable = bla_custom_track_mosaic

				set_variable = { name = bla_best_track value = 3 } # 1 coin, 2 banner, 3 steel, 4 veil, 5 candle, 6 mosaic
				set_variable = { name = bla_best_score value = var:bla_score_steel }

				if = {
					limit = { var:bla_score_coin >= var:bla_best_score }
					set_variable = { name = bla_best_score value = var:bla_score_coin }
					set_variable = { name = bla_best_track value = 1 }
				}
				if = {
					limit = { var:bla_score_banner >= var:bla_best_score }
					set_variable = { name = bla_best_score value = var:bla_score_banner }
					set_variable = { name = bla_best_track value = 2 }
				}
				if = {
					limit = { var:bla_score_veil >= var:bla_best_score }
					set_variable = { name = bla_best_score value = var:bla_score_veil }
					set_variable = { name = bla_best_track value = 4 }
				}
				if = {
					limit = { var:bla_score_candle >= var:bla_best_score }
					set_variable = { name = bla_best_score value = var:bla_score_candle }
					set_variable = { name = bla_best_track value = 5 }
				}
				if = {
					limit = { var:bla_best_score < 3 }
					set_variable = { name = bla_best_track value = 6 }
				}

				if = { limit = { var:bla_best_track = 6 } set_variable = { name = bla_custom_track_mosaic value = 1 } }
				else_if = { limit = { var:bla_best_track = 1 } set_variable = { name = bla_custom_track_coin value = 1 } }
				else_if = { limit = { var:bla_best_track = 2 } set_variable = { name = bla_custom_track_banner value = 1 } }
				else_if = { limit = { var:bla_best_track = 4 } set_variable = { name = bla_custom_track_veil value = 1 } }
				else_if = { limit = { var:bla_best_track = 5 } set_variable = { name = bla_custom_track_candle value = 1 } }
				else = { set_variable = { name = bla_custom_track_steel value = 1 } }

				if = {
					limit = { has_variable = bla_custom_track_mosaic }
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_mosaic_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_mosaic_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_mosaic_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_mosaic_legacy_4 }
				}
				else_if = {
					limit = { has_variable = bla_custom_track_steel }
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_steel_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_steel_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_steel_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_steel_legacy_4 }
				}
				else_if = {
					limit = { has_variable = bla_custom_track_coin }
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_coin_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_coin_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_coin_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_coin_legacy_4 }
				}
				else_if = {
					limit = { has_variable = bla_custom_track_banner }
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_banner_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_banner_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_banner_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_banner_legacy_4 }
				}
				else_if = {
					limit = { has_variable = bla_custom_track_veil }
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_veil_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_veil_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_veil_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_veil_legacy_4 }
				}
				else = {
					if = { limit = { var:bla_custom_age_unlocks >= 1 } add_dynasty_perk = bla_custom_candle_legacy_1 }
					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_candle_legacy_2 }
					if = { limit = { var:bla_custom_age_unlocks >= 3 } add_dynasty_perk = bla_custom_candle_legacy_3 }
					if = { limit = { var:bla_custom_age_unlocks >= 4 } add_dynasty_perk = bla_custom_candle_legacy_4 }
				}

				set_variable = { name = bla_custom_house_legacy_done value = 1 }
			}

			if = {
				limit = { dynasty = { has_variable = bla_custom_track_coin } }
				add_character_flag = bla_can_build_coin_dynasty_seat
			}
			else_if = {
				limit = { dynasty = { has_variable = bla_custom_track_banner } }
				add_character_flag = bla_can_build_banner_dynasty_seat
			}
			else_if = {
				limit = { dynasty = { has_variable = bla_custom_track_steel } }
				add_character_flag = bla_can_build_steel_dynasty_seat
			}
			else_if = {
				limit = { dynasty = { has_variable = bla_custom_track_veil } }
				add_character_flag = bla_can_build_veil_dynasty_seat
			}
			else_if = {
				limit = { dynasty = { has_variable = bla_custom_track_candle } }
				add_character_flag = bla_can_build_candle_dynasty_seat
			}
			else_if = {
				limit = { dynasty = { has_variable = bla_custom_track_mosaic } }
				add_character_flag = bla_can_build_mosaic_dynasty_seat
			}

			remove_trait = bla_legacy_customizer
			bla_apply_custom_dynasty_seat_effect = yes
		}

		if = { limit = { dynasty = { var:bla_custom_age_unlocks = 0 } } custom_tooltip = bla_custom_summary_new_intro }

		if = { limit = { dynasty = { has_variable = bla_custom_track_coin } } custom_tooltip = bla_custom_summary_coin }
		if = { limit = { dynasty = { has_variable = bla_custom_track_banner } } custom_tooltip = bla_custom_summary_banner }
		if = { limit = { dynasty = { has_variable = bla_custom_track_steel } } custom_tooltip = bla_custom_summary_steel }
		if = { limit = { dynasty = { has_variable = bla_custom_track_veil } } custom_tooltip = bla_custom_summary_veil }
		if = { limit = { dynasty = { has_variable = bla_custom_track_candle } } custom_tooltip = bla_custom_summary_candle }
