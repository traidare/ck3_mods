agot_bla_house_mudd = {
	effect = {
		if = {
			limit = {
				has_game_rule = agot_dynasty_reclaimed_bla_enabled
			}

				every_player = {
					limit = { game_start_date > 8258.1.1 }
					trigger_event = {
						id = agot_john_mudd.0001
						days = 20
					}
				}
		}
	}
}

agot_bla_house_strong = {
	effect = {
		if = {
			limit = {

			}
				every_player = {
					limit = { game_start_date > 8170.1.1 }

						days = 40
					}
				}
		}
	}
}

agot_bla_house_lothston = {
	effect = {
		if = {
			limit = {

			}
				every_player = {
					limit = { game_start_date > 8239.1.1 }

						days = 60

		}
	}
}
agot_bla_custom_house_legacy_on_start = {
	effect = {
		every_player = {
			if = {
				limit = {
					is_ai = no
					has_trait = bla_legacy_customizer
					exists = dynasty
					NOT = { dynasty = { has_variable = bla_custom_house_legacy_done } }
				}
				trigger_event = { id = agot_bla_custom_house_legacy.0001 days = 2 }
			}
		}
	}
}
