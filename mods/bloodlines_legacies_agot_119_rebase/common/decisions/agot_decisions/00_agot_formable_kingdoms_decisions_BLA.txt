## Crownlands
# Dusklands
kingdom_of_dusklands_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}
	decision_group_type = major
	ai_check_interval = 120
	desc = kingdom_of_dusklands_decision_desc
	selection_tooltip = kingdom_of_dusklands_tooltip

	is_shown = {
		dynasty = dynasty:dynn_Darklyn
		primary_title.tier = tier_duchy
		has_title = title:d_duskendale
		NOT = {
			has_variable = pre_war_liege 
			has_global_variable = pk_dusklands_decision
			}
		
	}
	is_valid = {
		OR = {
			is_independent_ruler = yes
			AND = {
				is_vassal_of = title:h_the_iron_throne.holder
				OR = {
					has_strong_hook = title:h_the_iron_throne.holder
					has_relation_friend	= title:h_the_iron_throne.holder
				}
			}
		}
		dynasty = {
			has_dynasty_perk = darklyn_legacy_5
		}
		completely_controls = title:d_duskendale
		completely_controls = title:d_stokeworth
		completely_controls = title:d_deerfield
		completely_controls = title:d_rooks_rest
		prestige_level >= 3
		gold >= 2000
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 1000
	}

	effect = {
		add_prestige = 1000
		add_legitimacy = 250
		give_nickname = nick_dusk_king
		gain_heroic_legend_seed_tooltip_effect = yes
		get_title = title:k_dusklands_bla
		title:h_the_iron_throne = { save_temporary_scope_as = empire }
		title:k_dusklands_bla = { 
			set_de_jure_liege_title = scope:empire
			save_temporary_scope_as = k_dusklands_bla
		}
		title:k_dusklands_bla = { save_temporary_scope_as = kingdom }
		title:d_duskendale = { 
			set_de_jure_liege_title = scope:kingdom	
		}
		title:d_stokeworth = { 
			set_de_jure_liege_title = scope:kingdom	
		}
		title:d_deerfield = { 
			set_de_jure_liege_title = scope:kingdom	
		}
		title:d_rooks_rest = { 
			set_de_jure_liege_title = scope:kingdom	
		}
		scope:kingdom = {
				set_capital_county = title:c_duskendale
			}
		add_pressed_claim = title:d_kings_landing
		add_pressed_claim = title:d_rosby
		add_pressed_claim = title:d_crackclaw_point
		set_global_variable = { name = pk_dusklands_decision value = 1 }
	
		add_character_modifier = {
            		modifier = restored_dusk_dynasty
           		years = 20
        	}
	}

	cost = { gold = 2000 }
}
