# Grand Remembrance 1.8.1 / CK3 1.19 runtime repair.
# Keep the chronicle visibility query safe while no GUI character is present.

gr_chronicle_window = {
	scope = character

	is_shown = {
		save_temporary_scope_as = character
		trigger_if = {
			limit = { exists = scope:character }
			scope:character ?= {
				is_alive = yes
				has_variable = gr_chronicle_open
				trigger_if = {
					limit = { has_variable = gr_chronicle_open }
					var:gr_chronicle_open > 0
				}
			}
		}
		trigger_else = { always = no }
	}
}
