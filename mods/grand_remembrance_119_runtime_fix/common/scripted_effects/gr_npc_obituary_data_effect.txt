# Grand Remembrance — NPC Obituary Data Gathering Effect
# Called ONCE per death during on_death (from the reference-player scope
# chosen by gr_internal_npc_obituary_processing) while scope:dead_npc
# is still a valid (alive) scope. Sets all gr_* variables on scope:dead_npc
# for chronicle page artifact creation.
#
# Scope context:
#   ROOT = the dying NPC (on_death's original ROOT — iterators/scope blocks
#          never rebind ROOT, only THIS)
#   scope:dead_npc = the dying NPC (still valid during on_death)
#   scope:gr_iter_player = the reference player (chronicler/page owner),
#          chosen once per death in gr_on_actions.txt; ROOT cannot be used
#          to reach them from here
#
# This effect was extracted from gr_npc_obituary.1 to fix the dead-scope
# bug where variables couldn't be set/read on a character 2 days after death.

gr_npc_obituary_data_effect = {
		# Generate random seed for text variant selection
		scope:dead_npc = {
			# Random seed for text-variant selection -- one uniform roll (audit M12).
			# Was stacked percent coin-flips: binomial, so the v5/v6 variant pools
			# almost never displayed (P(v6) ~ 0.16% on the player path).
			random_list = {
				1 = { set_variable = { name = gr_random_seed value = 0 } }
				1 = { set_variable = { name = gr_random_seed value = 20 } }
				1 = { set_variable = { name = gr_random_seed value = 40 } }
				1 = { set_variable = { name = gr_random_seed value = 60 } }
				1 = { set_variable = { name = gr_random_seed value = 80 } }
				1 = { set_variable = { name = gr_random_seed value = 100 } }
			}
		}

		# =============================================
		# SECTION A: BASIC INFO
		# =============================================
		scope:dead_npc = {
			set_variable = { name = gr_predecessor value = this }
			set_variable = { name = gr_death_year value = current_year }
			set_variable = { name = gr_death_age value = age }
			set_variable = { name = gr_gold value = gold }

			# Reign years (if they were a ruler)
			if = {
				limit = { is_ruler = yes }
				set_variable = { name = gr_reign_years value = years_as_ruler }
			}
			else = {
				set_variable = { name = gr_reign_years value = 0 }
			}

			# Title tier (guard for unlanded chars)
			if = {
				limit = { exists = primary_title }
				set_variable = { name = gr_tier value = primary_title.tier }
				set_variable = { name = gr_realm_name value = primary_title }
			}
			else = {
				set_variable = { name = gr_tier value = 0 }
			}

			if = {
				limit = { exists = house }
				set_variable = { name = gr_house_name value = house }
			}
			set_variable = { name = gr_culture value = culture }
			set_variable = { name = gr_predecessor_faith value = faith }

			if = {
				limit = { has_variable = gr_culture_origin }
				# Already set from tracking
			}

			# Admin government detection
			if = {
				limit = { government_has_flag = government_is_administrative }
				set_variable = { name = gr_is_admin_govt value = yes }
			}
		}

		# =============================================
		# SECTION B: TITLE LIST (up to 3)
		# Guard for unlanded NPCs
		# =============================================
		scope:dead_npc = {
			set_variable = { name = gr_title_count value = 0 }
			if = {
				limit = { exists = primary_title }
				set_variable = { name = gr_title_1 value = primary_title }
				change_variable = { name = gr_title_count add = 1 }

				every_held_title = {
					limit = {
						tier = scope:dead_npc.primary_title.tier
						NOT = { this = scope:dead_npc.primary_title }
						scope:dead_npc.var:gr_title_count < 3
					}
					scope:dead_npc = {
						if = {
							limit = { NOT = { has_variable = gr_title_2 } }
							set_variable = { name = gr_title_2 value = prev }
							change_variable = { name = gr_title_count add = 1 }
						}
						else_if = {
							limit = { NOT = { has_variable = gr_title_3 } }
							set_variable = { name = gr_title_3 value = prev }
							change_variable = { name = gr_title_count add = 1 }
						}
					}
				}
			}
		}

		# =============================================
		# SECTION C: DEATH REASON
		# =============================================
		scope:dead_npc = {
			if = {
				limit = { OR = { death_reason = death_natural_causes death_reason = death_old_age } }
				set_variable = { name = gr_death_reason value = flag:old_age }
			}
			else_if = {
				limit = { death_reason = death_battle }
				set_variable = { name = gr_death_reason value = flag:battle }
			}
			else_if = {
				limit = { OR = { death_reason = death_murder_known death_reason = death_murder death_reason = death_assassination death_reason = death_mysterious death_reason = death_plotting death_reason = death_poison } }
				set_variable = { name = gr_death_reason value = flag:murder }
			}
			else_if = {
				limit = { OR = { death_reason = death_execution death_reason = death_execution_blood_eagle death_reason = death_execution_blot } }
				set_variable = { name = gr_death_reason value = flag:execution }
			}
			else_if = {
				limit = { OR = { death_reason = death_childbirth death_reason = death_delivery } }
				set_variable = { name = gr_death_reason value = flag:childbirth }
			}
			else_if = {
				limit = { OR = {
					death_reason = death_bubonic_plague
					death_reason = death_smallpox
					death_reason = death_consumption
					death_reason = death_cancer
					death_reason = death_leper
					death_reason = death_great_pox
					death_reason = death_typhus
					death_reason = death_pneumonic
					death_reason = death_ill
					death_reason = death_measles
					death_reason = death_dysentery
					death_reason = death_ergotism
					death_reason = death_camp_fever
					death_reason = death_horrible_disease
				} }
				set_variable = { name = gr_death_reason value = flag:disease }
			}
			else_if = {
				limit = { OR = { death_reason = death_wounds death_reason = death_duel } }
				set_variable = { name = gr_death_reason value = flag:wounds }
			}
			else = {
				set_variable = { name = gr_death_reason value = flag:unknown }
			}
		}

		# =============================================
		# SECTION C2: PARENTS
		# =============================================
		scope:dead_npc = {
			if = {
				limit = { exists = father }
				set_variable = { name = gr_father value = father }
			}
			if = {
				limit = { exists = mother }
				set_variable = { name = gr_mother value = mother }
			}
		}

		# =============================================
		# SECTION C2b: HEIR DETECTION (best effort)
		# CK3 doesn't expose AI heir as a scope during on_death,
		# so we approximate: prefer eldest adult male child.
		# Won't be 100% accurate for elective/ultimogeniture
		# but handles primogeniture/partition (the common case).
		# =============================================
		scope:dead_npc = {
			if = {
				limit = { any_child = { is_alive = yes } }
				random_child = {
					limit = { is_alive = yes }
					weight = {
						base = 1
						modifier = { add = 100 is_male = yes }
						modifier = { add = 50 age >= 16 }
						modifier = { add = 25 age >= 30 }
					}
					scope:dead_npc = {
						set_variable = { name = gr_heir_char value = prev }
					}
				}
			}
		}

		# =============================================
		# SECTION C3: CHRONICLER & OPINION
		# For NPCs, use court_chronicler type so the obituary
		# has a named author (the viewing player). Opinion is
		# from that player's perspective of the dead NPC.
		# =============================================
		scope:dead_npc = {
			set_variable = { name = gr_chronicler_type value = flag:court_chronicler }
			set_variable = { name = gr_chronicler value = scope:gr_iter_player }
		}
		# Capture the player's opinion while both character scopes are valid.
		# save_temporary_opinion_value_as is the CK3 1.19 interface for
		# materializing an opinion as a numeric temporary scope.
		scope:gr_iter_player = {
			save_temporary_opinion_value_as = {
				name = gr_opinion_of_dead_npc
				target = scope:dead_npc
			}
			scope:dead_npc = {
				set_variable = {
					name = gr_opinion
					value = scope:gr_opinion_of_dead_npc
				}
			}
		}

		# =============================================
		# SECTION D: COPY TRACKED MEMORIES
		# Most memory vars are already on scope:dead_npc from
		# the gr_is_tracked_character tracking system.
		# Only renames needed here.
		# =============================================
		scope:dead_npc = {
			# Battle casualties rename
			if = {
				limit = { has_variable = gr_total_casualties }
				set_variable = { name = gr_casualties value = var:gr_total_casualties }
			}
			else = {
				set_variable = { name = gr_casualties value = 0 }
			}

			# Battle tier (already named correctly)
			# gr_memory_great_battle, gr_battle_tier, gr_best_battle_strength -- already on dead_npc
			# gr_memory_led_troops, gr_memory_won_war -- already on dead_npc
			# gr_memory_varangian + sub-vars -- already on dead_npc
			# gr_memory_adventure + sub-vars -- already on dead_npc
			# gr_memory_created_kingdom + sub-vars -- already on dead_npc
			# gr_memory_created_empire, gr_memory_claimed_hegemony -- already on dead_npc
			# Activity counts (feast, hunt, tournament, etc.) -- already on dead_npc
			# Intrigue counts (murder, kidnap, etc.) -- already on dead_npc

			# Capital development at death
			if = {
				limit = {
					exists = capital_county
					has_variable = gr_capital_dev_start
				}
				set_variable = {
					name = gr_capital_dev_growth
					value = {
						value = capital_county.development_level
						subtract = var:gr_capital_dev_start
					}
				}
			}

			# Became king young
			if = {
				limit = { exists = primary_title primary_title.tier >= 3 }
				set_variable = {
					name = gr_became_king_age
					value = {
						value = age
						subtract = years_as_ruler
					}
				}
			}
		}

		# =============================================
		# SECTION E: STATS (snapshot at death)
		# =============================================
		scope:dead_npc = {
			set_variable = { name = gr_stat_diplomacy value = diplomacy }
			set_variable = { name = gr_stat_martial value = martial }
			set_variable = { name = gr_stat_stewardship value = stewardship }
			set_variable = { name = gr_stat_intrigue value = intrigue }
			set_variable = { name = gr_stat_learning value = learning }
			set_variable = { name = gr_stat_prowess value = prowess }

			# Determine top stat
			set_variable = { name = gr_top_stat value = flag:diplomacy }
			set_variable = { name = gr_top_stat_val value = diplomacy }

			if = {
				limit = { martial > var:gr_top_stat_val }
				set_variable = { name = gr_top_stat value = flag:martial }
				set_variable = { name = gr_top_stat_val value = martial }
			}
			if = {
				limit = { stewardship > var:gr_top_stat_val }
				set_variable = { name = gr_top_stat value = flag:stewardship }
				set_variable = { name = gr_top_stat_val value = stewardship }
			}
			if = {
				limit = { intrigue > var:gr_top_stat_val }
				set_variable = { name = gr_top_stat value = flag:intrigue }
				set_variable = { name = gr_top_stat_val value = intrigue }
			}
			if = {
				limit = { learning > var:gr_top_stat_val }
				set_variable = { name = gr_top_stat value = flag:learning }
				set_variable = { name = gr_top_stat_val value = learning }
			}
		}

		# =============================================
		# SECTION F: PRESTIGE & PIETY LEVELS
		# =============================================
		scope:dead_npc = {
			set_variable = { name = gr_prestige_level value = prestige_level }
			set_variable = { name = gr_prestige value = prestige }
			set_variable = { name = gr_piety_level value = piety_level }
			set_variable = { name = gr_piety value = piety }
			set_variable = { name = gr_dread value = dread }
			set_variable = { name = gr_monthly_income value = monthly_character_income }
		}

		# =============================================
		# SECTION G: PERSONALITY TRAITS (35 total)
		# =============================================
		scope:dead_npc = {
			if = { limit = { has_trait = lustful } set_variable = { name = gr_trait_lustful value = yes } }
			if = { limit = { has_trait = chaste } set_variable = { name = gr_trait_chaste value = yes } }
			if = { limit = { has_trait = gluttonous } set_variable = { name = gr_trait_gluttonous value = yes } }
			if = { limit = { has_trait = temperate } set_variable = { name = gr_trait_temperate value = yes } }
			if = { limit = { has_trait = greedy } set_variable = { name = gr_trait_greedy value = yes } }
			if = { limit = { has_trait = generous } set_variable = { name = gr_trait_generous value = yes } }
			if = { limit = { has_trait = lazy } set_variable = { name = gr_trait_lazy value = yes } }
			if = { limit = { has_trait = diligent } set_variable = { name = gr_trait_diligent value = yes } }
			if = { limit = { has_trait = wrathful } set_variable = { name = gr_trait_wrathful value = yes } }
			if = { limit = { has_trait = calm } set_variable = { name = gr_trait_calm value = yes } }
			if = { limit = { has_trait = patient } set_variable = { name = gr_trait_patient value = yes } }
			if = { limit = { has_trait = impatient } set_variable = { name = gr_trait_impatient value = yes } }
			if = { limit = { has_trait = arrogant } set_variable = { name = gr_trait_arrogant value = yes } }
			if = { limit = { has_trait = humble } set_variable = { name = gr_trait_humble value = yes } }
			if = { limit = { has_trait = deceitful } set_variable = { name = gr_trait_deceitful value = yes } }
			if = { limit = { has_trait = honest } set_variable = { name = gr_trait_honest value = yes } }
			if = { limit = { has_trait = craven } set_variable = { name = gr_trait_craven value = yes } }
			if = { limit = { has_trait = brave } set_variable = { name = gr_trait_brave value = yes } }
			if = { limit = { has_trait = shy } set_variable = { name = gr_trait_shy value = yes } }
			if = { limit = { has_trait = gregarious } set_variable = { name = gr_trait_gregarious value = yes } }
			if = { limit = { has_trait = ambitious } set_variable = { name = gr_trait_ambitious value = yes } }
			if = { limit = { has_trait = content } set_variable = { name = gr_trait_content value = yes } }
			if = { limit = { has_trait = arbitrary } set_variable = { name = gr_trait_arbitrary value = yes } }
			if = { limit = { has_trait = just } set_variable = { name = gr_trait_just value = yes } }
			if = { limit = { has_trait = cynical } set_variable = { name = gr_trait_cynical value = yes } }
			if = { limit = { has_trait = zealous } set_variable = { name = gr_trait_zealous value = yes } }
			if = { limit = { has_trait = paranoid } set_variable = { name = gr_trait_paranoid value = yes } }
			if = { limit = { has_trait = trusting } set_variable = { name = gr_trait_trusting value = yes } }
			if = { limit = { has_trait = compassionate } set_variable = { name = gr_trait_compassionate value = yes } }
			if = { limit = { has_trait = callous } set_variable = { name = gr_trait_callous value = yes } }
			if = { limit = { has_trait = sadistic } set_variable = { name = gr_trait_sadistic value = yes } }
			if = { limit = { has_trait = stubborn } set_variable = { name = gr_trait_stubborn value = yes } }
			if = { limit = { has_trait = fickle } set_variable = { name = gr_trait_fickle value = yes } }
			if = { limit = { has_trait = vengeful } set_variable = { name = gr_trait_vengeful value = yes } }
			if = { limit = { has_trait = forgiving } set_variable = { name = gr_trait_forgiving value = yes } }
		}

		# =============================================
		# SECTION H: EDUCATION TRAITS
		# =============================================
		scope:dead_npc = {
			# Diplomacy education
			if = { limit = { has_trait = education_diplomacy_1 } set_variable = { name = gr_edu value = flag:diplomacy } set_variable = { name = gr_edu_tier value = 1 } }
			if = { limit = { has_trait = education_diplomacy_2 } set_variable = { name = gr_edu value = flag:diplomacy } set_variable = { name = gr_edu_tier value = 2 } }
			if = { limit = { has_trait = education_diplomacy_3 } set_variable = { name = gr_edu value = flag:diplomacy } set_variable = { name = gr_edu_tier value = 3 } }
			if = { limit = { has_trait = education_diplomacy_4 } set_variable = { name = gr_edu value = flag:diplomacy } set_variable = { name = gr_edu_tier value = 4 } }
			if = { limit = { has_trait = education_diplomacy_5 } set_variable = { name = gr_edu value = flag:diplomacy } set_variable = { name = gr_edu_tier value = 5 } }
			# Martial education
			if = { limit = { has_trait = education_martial_1 } set_variable = { name = gr_edu value = flag:martial } set_variable = { name = gr_edu_tier value = 1 } }
			if = { limit = { has_trait = education_martial_2 } set_variable = { name = gr_edu value = flag:martial } set_variable = { name = gr_edu_tier value = 2 } }
			if = { limit = { has_trait = education_martial_3 } set_variable = { name = gr_edu value = flag:martial } set_variable = { name = gr_edu_tier value = 3 } }
			if = { limit = { has_trait = education_martial_4 } set_variable = { name = gr_edu value = flag:martial } set_variable = { name = gr_edu_tier value = 4 } }
			if = { limit = { has_trait = education_martial_5 } set_variable = { name = gr_edu value = flag:martial } set_variable = { name = gr_edu_tier value = 5 } }
			# Stewardship education
			if = { limit = { has_trait = education_stewardship_1 } set_variable = { name = gr_edu value = flag:stewardship } set_variable = { name = gr_edu_tier value = 1 } }
			if = { limit = { has_trait = education_stewardship_2 } set_variable = { name = gr_edu value = flag:stewardship } set_variable = { name = gr_edu_tier value = 2 } }
			if = { limit = { has_trait = education_stewardship_3 } set_variable = { name = gr_edu value = flag:stewardship } set_variable = { name = gr_edu_tier value = 3 } }
			if = { limit = { has_trait = education_stewardship_4 } set_variable = { name = gr_edu value = flag:stewardship } set_variable = { name = gr_edu_tier value = 4 } }
			if = { limit = { has_trait = education_stewardship_5 } set_variable = { name = gr_edu value = flag:stewardship } set_variable = { name = gr_edu_tier value = 5 } }
			# Intrigue education
			if = { limit = { has_trait = education_intrigue_1 } set_variable = { name = gr_edu value = flag:intrigue } set_variable = { name = gr_edu_tier value = 1 } }
			if = { limit = { has_trait = education_intrigue_2 } set_variable = { name = gr_edu value = flag:intrigue } set_variable = { name = gr_edu_tier value = 2 } }
			if = { limit = { has_trait = education_intrigue_3 } set_variable = { name = gr_edu value = flag:intrigue } set_variable = { name = gr_edu_tier value = 3 } }
			if = { limit = { has_trait = education_intrigue_4 } set_variable = { name = gr_edu value = flag:intrigue } set_variable = { name = gr_edu_tier value = 4 } }
			if = { limit = { has_trait = education_intrigue_5 } set_variable = { name = gr_edu value = flag:intrigue } set_variable = { name = gr_edu_tier value = 5 } }
			# Learning education
			if = { limit = { has_trait = education_learning_1 } set_variable = { name = gr_edu value = flag:learning } set_variable = { name = gr_edu_tier value = 1 } }
			if = { limit = { has_trait = education_learning_2 } set_variable = { name = gr_edu value = flag:learning } set_variable = { name = gr_edu_tier value = 2 } }
			if = { limit = { has_trait = education_learning_3 } set_variable = { name = gr_edu value = flag:learning } set_variable = { name = gr_edu_tier value = 3 } }
			if = { limit = { has_trait = education_learning_4 } set_variable = { name = gr_edu value = flag:learning } set_variable = { name = gr_edu_tier value = 4 } }
			if = { limit = { has_trait = education_learning_5 } set_variable = { name = gr_edu value = flag:learning } set_variable = { name = gr_edu_tier value = 5 } }
		}

		# =============================================
		# SECTION I: CONGENITAL TRAITS
		# =============================================
		scope:dead_npc = {
			# Intelligence
			if = { limit = { has_trait = intellect_good_3 } set_variable = { name = gr_intel value = flag:intellect_good_3 } }
			if = { limit = { has_trait = intellect_good_2 } set_variable = { name = gr_intel value = flag:intellect_good_2 } }
			if = { limit = { has_trait = intellect_good_1 } set_variable = { name = gr_intel value = flag:intellect_good_1 } }
			if = { limit = { has_trait = intellect_bad_1 } set_variable = { name = gr_intel value = flag:intellect_bad_1 } }
			if = { limit = { has_trait = intellect_bad_2 } set_variable = { name = gr_intel value = flag:intellect_bad_2 } }
			if = { limit = { has_trait = intellect_bad_3 } set_variable = { name = gr_intel value = flag:intellect_bad_3 } }
			# Physique
			if = { limit = { has_trait = physique_good_3 } set_variable = { name = gr_physique value = flag:physique_good_3 } }
			if = { limit = { has_trait = physique_good_2 } set_variable = { name = gr_physique value = flag:physique_good_2 } }
			if = { limit = { has_trait = physique_good_1 } set_variable = { name = gr_physique value = flag:physique_good_1 } }
			if = { limit = { has_trait = physique_bad_1 } set_variable = { name = gr_physique value = flag:physique_bad_1 } }
			if = { limit = { has_trait = physique_bad_2 } set_variable = { name = gr_physique value = flag:physique_bad_2 } }
			if = { limit = { has_trait = physique_bad_3 } set_variable = { name = gr_physique value = flag:physique_bad_3 } }
			# Appearance
			if = { limit = { has_trait = beauty_good_3 } set_variable = { name = gr_beauty value = flag:beauty_good_3 } }
			if = { limit = { has_trait = beauty_good_2 } set_variable = { name = gr_beauty value = flag:beauty_good_2 } }
			if = { limit = { has_trait = beauty_good_1 } set_variable = { name = gr_beauty value = flag:beauty_good_1 } }
			if = { limit = { has_trait = beauty_bad_1 } set_variable = { name = gr_beauty value = flag:beauty_bad_1 } }
			if = { limit = { has_trait = beauty_bad_2 } set_variable = { name = gr_beauty value = flag:beauty_bad_2 } }
			if = { limit = { has_trait = beauty_bad_3 } set_variable = { name = gr_beauty value = flag:beauty_bad_3 } }
			# Special congenital
			if = { limit = { has_trait = giant } set_variable = { name = gr_con_giant value = yes } }
