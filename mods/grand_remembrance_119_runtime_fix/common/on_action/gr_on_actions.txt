# Use on_actions nesting (not direct effect block) to avoid
# overriding vanilla's on_death effect. The on_actions list
# is additive across files, so this safely extends vanilla behavior.
on_death = {
	on_actions = {
		gr_internal_obituary_processing
		gr_internal_npc_obituary_processing
	}
}

# ================================================
# RICE COMPATIBILITY: disabled for this AGOT playset
# The RICE-only placeholder faith cannot be queried safely when RICE is absent.

# NPC Obituary Processing
# Fires on every death; guards itself to only process
# qualifying NPC dynasty members or close family.
#
# IMPORTANT: All data gathering, page creation, and dynasty list
# management happens HERE during on_death while scope:dead_npc is
# still a valid (alive) scope. The delayed event (gr_npc_obituary.1)
# only sends the toast notification.
#
# Previous bug: data gathering was deferred to a 2-day delayed event.
# By then the NPC was a dead character (weak scope), so has_variable
# and var:X failed — producing empty chronicle pages.
# ================================================
gr_internal_npc_obituary_processing = {
	effect = {
		if = {
			limit = {
				gr_is_qualifying_npc = yes
			}

			# Save dead NPC while still a valid scope
			save_scope_as = dead_npc

			# ---- Reference player selection ----
			# One player is chosen as chronicler/owner for this death: the
			# obituary data, the chronicle page artifact, its dynasty-list
			# registration, and the pedestal below all happen exactly ONCE,
			# on this player. (Running them inside every_player duplicated
			# dynasty-list entries in multiplayer and desynced the
			# rulers/pages lockstep lists that gr_cw_page_setup_effect pages
			# through.) Toasts remain per-player at the end.
			# Preference: a player of the deceased's own dynasty (their
			# chronicle window is where this page will be browsed), then any
			# close-family player, then any player at all.
			random_player = {
				limit = {
					exists = dynasty
					exists = scope:dead_npc.dynasty
					dynasty = scope:dead_npc.dynasty
				}
				save_scope_as = gr_iter_player
			}
			if = {
				limit = { NOT = { exists = scope:gr_iter_player } }
				random_player = {
					limit = { is_close_family_of = scope:dead_npc }
					save_scope_as = gr_iter_player
				}
			}
			if = {
				limit = { NOT = { exists = scope:gr_iter_player } }
				random_player = {
					save_scope_as = gr_iter_player
				}
			}

			scope:gr_iter_player = {
				# ---- Step 1: Gather all obituary data (once) ----
				# Sets gr_* variables on scope:dead_npc (traits, stats,
				# titles, scores, gr_score / gr_score_tier, etc.) while the
				# character is still alive. MUST run before Steps 3/3b below,
				# which read gr_score and gr_score_tier off scope:dead_npc.
				# Chronicler + opinion inside read scope:gr_iter_player.
				gr_npc_obituary_data_effect = yes
			}

				# ---- Step 1b: Secret deeds analysis (secrets wave 2026-07-03) ----
				# Same resolution as the player path's Section C4, computed on
				# scope:dead_npc. The chronicler here is the viewing player, so
				# "knows" means that player is a knower of the secret. Counters
				# (gr_seduce_count etc.) are already on scope:dead_npc via tracking.
				scope:dead_npc = {
					set_variable = { name = gr_secret_murders value = 0 }
					set_variable = { name = gr_secret_affairs value = 0 }
					every_secret = {
						limit = {
							OR = {
								secret_type = secret_murder
								secret_type = secret_murder_attempt
								secret_type = secret_coup_plotter
								secret_type = secret_raid_estate
							}
						}
						scope:dead_npc = { change_variable = { name = gr_secret_murders add = 1 } }
					}
					every_secret = {
						limit = { secret_type = secret_lover }
						scope:dead_npc = { change_variable = { name = gr_secret_affairs add = 1 } }
					}

					if = {
						limit = { has_variable = gr_murder_count }
						set_variable = { name = gr_public_murders value = var:gr_murder_count }
						change_variable = { name = gr_public_murders subtract = var:gr_secret_murders }
						if = {
							limit = { var:gr_public_murders < 0 }
							set_variable = { name = gr_public_murders value = 0 }
						}
					}
					else = {
						set_variable = { name = gr_public_murders value = 0 }
					}

					set_variable = { name = gr_secret_deeds value = var:gr_secret_murders }
					change_variable = { name = gr_secret_deeds add = var:gr_secret_affairs }

					set_variable = { name = gr_chronicler_knows value = no }
					if = {
						limit = { has_variable = gr_chronicler }
						if = {
							limit = {
								any_secret = {
									OR = {
										secret_type = secret_murder
										secret_type = secret_murder_attempt
										secret_type = secret_coup_plotter
										secret_type = secret_raid_estate
										secret_type = secret_lover
									}
									is_known_by = scope:dead_npc.var:gr_chronicler
								}
							}
							set_variable = { name = gr_chronicler_knows value = yes }
						}
					}

					set_variable = { name = gr_chronicler_liked value = no }
					if = {
						limit = { has_variable = gr_opinion var:gr_opinion >= 20 }
						set_variable = { name = gr_chronicler_liked value = yes }
					}

					if = {
						limit = { has_game_rule = gr_secret_never }
						if = {
							limit = { var:gr_secret_deeds >= 1 }
							set_variable = { name = gr_secrets_buried value = yes }
						}
					}
					else_if = {
						limit = { has_game_rule = gr_secret_always }
						if = {
							limit = { var:gr_secret_deeds >= 1 }
							set_variable = { name = gr_secrets_revealed value = yes }
						}
					}
					else = {
						if = {
							limit = { var:gr_secret_deeds >= 1 }
							if = {
								limit = { var:gr_chronicler_knows = yes }
								if = {
									limit = { var:gr_chronicler_liked = yes }
									set_variable = { name = gr_secrets_buried value = yes }
								}
								else = {
									set_variable = { name = gr_secrets_revealed value = yes }
								}
							}
							else_if = {
								limit = { var:gr_secret_deeds >= 3 }
								set_variable = { name = gr_secrets_hinted value = yes }
							}
						}
					}
				}

			# ---- Step 3: Add to dynasty remembered rulers list (once) ----
			# Chronicle filter game rule (audit M4): under "Qualifying Only",
			# Forgettable-tier rulers get no chronicle entry at all -- the
			# ruler append here and the page append in Step 2b are gated by
			# the SAME condition so the two lists stay in lockstep.
			if = {
				limit = {
					scope:dead_npc = { exists = dynasty }
					OR = {
						NOT = { has_game_rule = gr_chronicle_qualifying }
						scope:dead_npc = { NOT = { has_variable = gr_score_tier } }
						scope:dead_npc = { NOT = { var:gr_score_tier = flag:forgettable } }
					}
				}
				scope:dead_npc.dynasty = {
					add_to_variable_list = {
						name = gr_remembered_rulers
						target = scope:dead_npc
					}
					set_variable = { name = gr_has_remembered_rulers value = yes }
				}
			}

			# ---- Step 3b: Increment dynasty total score (once) ----
			# Runs AFTER Step 1 so scope:dead_npc.var:gr_score exists.
			# Not gated by the chronicle filter: a hidden ruler still
			# contributed their legacy to the dynasty's total.
			if = {
				limit = {
					scope:dead_npc = { exists = dynasty }
					scope:dead_npc = { has_variable = gr_score }
				}
				scope:dead_npc.dynasty = {
					if = {
						limit = { NOT = { has_variable = gr_dynasty_total } }
						set_variable = { name = gr_dynasty_total value = 0 }
					}
					change_variable = {
						name = gr_dynasty_total
						add = scope:dead_npc.var:gr_score
					}
				}
			}

			scope:gr_iter_player = {
				# ---- Step 2: Create chronicle page artifact (once) ----
				# Copies all gr_* vars from scope:dead_npc to the page.
				# Owned by the reference player (THIS) so it persists.
				# Gated identically to Step 3 (audit M4) so no orphan page
				# is created for a filtered-out ruler.
				if = {
					limit = {
						OR = {
							NOT = { has_game_rule = gr_chronicle_qualifying }
							scope:dead_npc = { NOT = { has_variable = gr_score_tier } }
							scope:dead_npc = { NOT = { var:gr_score_tier = flag:forgettable } }
						}
					}
					gr_create_npc_chronicle_page_effect = yes
				}
			}


			# ---- Step 2b: Register the page on the dynasty (once) ----
			# Exactly one page entry per death so the ruler entry appended in
			# Step 3 above and this page entry stay lockstep-paired at the
			# same list index (gr_cw_page_setup_effect walks both lists by
			# index).
			if = {
				limit = {
					scope:dead_npc = { exists = dynasty }
					exists = scope:chronicle_page
				}
				scope:dead_npc.dynasty = {
					add_to_variable_list = {
						name = gr_remembered_chronicle_pages
						target = scope:chronicle_page
					}
				}
				scope:chronicle_page = {
					set_variable = {
						name = gr_dynasty_total
						value = scope:dead_npc.dynasty.var:gr_dynasty_total
					}
				}
			}

			# ---- Step 5: Fire toast notification (2-day delay) ----
			# Genuinely per-player: every player gets their own toast.
			# Gated on the page existing: a ruler hidden by the chronicle
			# filter (audit M4) got no page, so announcing one would lie.
			if = {
				limit = { exists = scope:chronicle_page }
				every_player = {
					trigger_event = {
						id = gr_npc_obituary.1
						days = 2
					}
				}
			}

			# ---- Step 4: Chronicle pedestal create/upgrade ----
			# Dynasty COUNTERS run exactly once per death (previously they
			# incremented once per player, inflating in multiplayer). The
			# pedestal ARTIFACT is a player-held keepsake: it is destroyed
			# and recreated on the reference player (scope:gr_iter_player)
			# chosen above, never on the dead NPC.
			if = {
				limit = {
					OR = {
						scope:dead_npc = { var:gr_score_tier = flag:legendary }
						has_game_rule = gr_pedestal_all
					}
					scope:dead_npc = { exists = dynasty }
					exists = scope:gr_iter_player
				}

				# UPGRADE existing chronicle
				if = {
					limit = { scope:dead_npc.dynasty = { has_variable = gr_has_chronicle } }

					scope:dead_npc.dynasty = {
						change_variable = { name = gr_chronicles_recorded add = 1 }
					}

					scope:gr_iter_player = {
						# Find and destroy the player's old chronicle artifact
						random_character_artifact = {
							limit = { has_variable = gr_is_chronicle }
							save_scope_as = old_chronicle
						}
						if = {
							limit = { exists = scope:old_chronicle }
							destroy_artifact = scope:old_chronicle
						}

						# Create upgraded artifact based on chronicle count
						# ILLUSTRIOUS (10+ chronicles)
						if = {
							limit = { scope:dead_npc.dynasty = { var:gr_chronicles_recorded >= 10 } }
							create_artifact = {
								name = gr_chronicle_artifact_name
								description = gr_chronicle_artifact_desc_illustrious
								type = book
								visuals = book
								quality = 100
								wealth = 80
								max_durability = 100
								modifier = gr_chronicle_illustrious
								save_scope_as = new_chronicle
							}
						}
						# FAMED (6+ chronicles)
						else_if = {
							limit = { scope:dead_npc.dynasty = { var:gr_chronicles_recorded >= 6 } }
							create_artifact = {
								name = gr_chronicle_artifact_name
								description = gr_chronicle_artifact_desc_famed
								type = book
								visuals = book
								quality = 80
								wealth = 50
								max_durability = 100
								modifier = gr_chronicle_famed
								save_scope_as = new_chronicle
							}
						}
						# MASTERWORK (3+ chronicles)
						else_if = {
							limit = { scope:dead_npc.dynasty = { var:gr_chronicles_recorded >= 3 } }
							create_artifact = {
								name = gr_chronicle_artifact_name
								description = gr_chronicle_artifact_desc_masterwork
								type = book
								visuals = book
								quality = 40
								wealth = 20
								max_durability = 100
								modifier = gr_chronicle_masterwork
								save_scope_as = new_chronicle
							}
						}
						# STILL COMMON (under 3 chronicles)
						else = {
							create_artifact = {
								name = gr_chronicle_artifact_name
								description = gr_chronicle_artifact_desc_common
								type = book
								visuals = book
								quality = 10
								max_durability = 100
								modifier = gr_chronicle_common
								save_scope_as = new_chronicle
							}
						}

						# Tag new artifact
						if = {
							limit = { exists = scope:new_chronicle }
							scope:new_chronicle = {
								set_variable = { name = gr_is_chronicle value = yes }
							}
						}
					}
				}
				# CREATE new chronicle
				else = {
					scope:gr_iter_player = {
						create_artifact = {
							name = gr_chronicle_artifact_name
							description = gr_chronicle_artifact_desc_common
							type = book
							visuals = book
							quality = 10
							max_durability = 100
							modifier = gr_chronicle_common
							save_scope_as = new_chronicle
						}

						scope:new_chronicle = {
							set_variable = { name = gr_is_chronicle value = yes }
						}
					}

					scope:dead_npc.dynasty = {
						set_variable = { name = gr_has_chronicle value = yes }
						set_variable = { name = gr_chronicles_recorded value = 1 }
					}
				}

			# ---- Step 6: Prune raw tracking vars off the dead NPC (audit M3) ----
			# The chronicle page artifact (Step 2) now holds a full snapshot of
			# every gr_* var, and the pedestal (Step 4) has already read
			# gr_score_tier. Nothing downstream reads the dead NPC's raw vars, so
			# strip the ~350-var lifetime snapshot off the corpse to cut save
			# bloat. gr_obit_* compact vars are NOT in the pruned list, so the
			# chronicle reader summary is untouched. The heir is never in scope
			# here, so it is never pruned.
			scope:dead_npc = {
				gr_dead_ruler_prune_effect = yes
			}
			}
		}
	}
}

gr_internal_obituary_processing = {
	effect = {
		# Only track landed player character deaths
		# is_landed/is_ruler covers all government types
		# player_heir fallback covers administrative government
		if = {
			limit = {
				is_ai = no
				OR = {
					is_landed = yes
					is_ruler = yes
				}
				OR = {
					exists = primary_heir
					exists = player_heir
				}
			}

			# Save to global so event can access it. Single global variable
			# holding only the most-recently-dead ruler (was previously also
			# appended to a global list, gr_has_last_dead, that was never
			# cleared and only ever read by debug tooling — replaced by
			# this single var; see gr_debug.99 in gr_debug_events.txt).
			save_scope_as = dead_character
			set_global_variable = {
				name = gr_last_dead_var
				value = root
			}

			# Resolve heir - prefer player_heir (who the player continues as),
			# fall back to primary_heir for elective/admin governments.
			if = {
				limit = { exists = player_heir }
				player_heir = { save_scope_as = heir }
			}
			else_if = {
				limit = { exists = primary_heir }
				primary_heir = { save_scope_as = heir }
			}
			scope:heir = {

				# =============================================
				# CLEANUP: Remove any inherited memory variables
				# from previous generation's obituary processing.
				# Without this, achievements "leak" across rulers.
				# Primary cleanup happens in gr_obituary.10 after block.
				# This is a secondary defense for edge cases.
				# =============================================
				# Memory flags
				if = { limit = { has_variable = gr_memory_great_battle } remove_variable = gr_memory_great_battle }
				if = { limit = { has_variable = gr_succession_lock } remove_variable = gr_succession_lock }
				if = { limit = { has_variable = gr_battle_tier } remove_variable = gr_battle_tier }
				if = { limit = { has_variable = gr_best_battle_strength } remove_variable = gr_best_battle_strength }
				if = { limit = { has_variable = gr_memory_led_troops } remove_variable = gr_memory_led_troops }
				if = { limit = { has_variable = gr_total_casualties } remove_variable = gr_total_casualties }
				if = { limit = { has_variable = gr_holding_count } remove_variable = gr_holding_count }
				if = { limit = { has_variable = gr_holding_1 } remove_variable = gr_holding_1 }
				if = { limit = { has_variable = gr_holding_2 } remove_variable = gr_holding_2 }
				if = { limit = { has_variable = gr_holding_3 } remove_variable = gr_holding_3 }
				if = { limit = { has_variable = gr_holding_4 } remove_variable = gr_holding_4 }
				if = { limit = { has_variable = gr_holding_5 } remove_variable = gr_holding_5 }
				if = { limit = { has_variable = gr_building_count } remove_variable = gr_building_count }
				if = { limit = { has_variable = gr_temple_count } remove_variable = gr_temple_count }
				if = { limit = { has_variable = gr_university_built } remove_variable = gr_university_built }
				if = { limit = { has_variable = gr_alliance_count } remove_variable = gr_alliance_count }
				if = { limit = { has_variable = gr_friend_count } remove_variable = gr_friend_count }
				if = { limit = { has_variable = gr_feast_count } remove_variable = gr_feast_count }
				if = { limit = { has_variable = gr_tournament_count } remove_variable = gr_tournament_count }
				if = { limit = { has_variable = gr_hunt_count } remove_variable = gr_hunt_count }
				if = { limit = { has_variable = gr_pilgrimage_count } remove_variable = gr_pilgrimage_count }
				if = { limit = { has_variable = gr_university_attended } remove_variable = gr_university_attended }
				if = { limit = { has_variable = gr_murder_count } remove_variable = gr_murder_count }
				if = { limit = { has_variable = gr_kidnap_count } remove_variable = gr_kidnap_count }
				if = { limit = { has_variable = gr_execution_count } remove_variable = gr_execution_count }
				if = { limit = { has_variable = gr_torture_count } remove_variable = gr_torture_count }
				if = { limit = { has_variable = gr_hook_count } remove_variable = gr_hook_count }
				if = { limit = { has_variable = gr_language_count } remove_variable = gr_language_count }
				if = { limit = { has_variable = gr_monthly_income } remove_variable = gr_monthly_income }
				if = { limit = { has_variable = gr_capital_dev_growth } remove_variable = gr_capital_dev_growth }
				if = { limit = { has_variable = gr_capital_dev_start } remove_variable = gr_capital_dev_start }
				if = { limit = { has_variable = gr_initial_capital } remove_variable = gr_initial_capital }
				if = { limit = { has_variable = gr_tour_count } remove_variable = gr_tour_count }
				if = { limit = { has_variable = gr_wedding_count } remove_variable = gr_wedding_count }
				if = { limit = { has_variable = gr_debate_count } remove_variable = gr_debate_count }
				if = { limit = { has_variable = gr_coronation_count } remove_variable = gr_coronation_count }
				if = { limit = { has_variable = gr_inspection_count } remove_variable = gr_inspection_count }
				if = { limit = { has_variable = gr_expedition_count } remove_variable = gr_expedition_count }
				if = { limit = { has_variable = gr_ritual_count } remove_variable = gr_ritual_count }
				if = { limit = { has_variable = gr_blot_count } remove_variable = gr_blot_count }
				if = { limit = { has_variable = gr_festival_count } remove_variable = gr_festival_count }
				if = { limit = { has_variable = gr_hike_count } remove_variable = gr_hike_count }
				if = { limit = { has_variable = gr_chariot_count } remove_variable = gr_chariot_count }
				if = { limit = { has_variable = gr_memory_won_war } remove_variable = gr_memory_won_war }
				if = { limit = { has_variable = gr_memory_varangian } remove_variable = gr_memory_varangian }
				if = { limit = { has_variable = gr_varangian_target } remove_variable = gr_varangian_target }
				if = { limit = { has_variable = gr_varangian_first_target } remove_variable = gr_varangian_first_target }
				if = { limit = { has_variable = gr_varangian_realm } remove_variable = gr_varangian_realm }
				if = { limit = { has_variable = gr_varangian_first_realm } remove_variable = gr_varangian_first_realm }
				if = { limit = { has_variable = gr_varangian_count } remove_variable = gr_varangian_count }
				if = { limit = { has_variable = gr_memory_adventure } remove_variable = gr_memory_adventure }
				if = { limit = { has_variable = gr_adventure_target } remove_variable = gr_adventure_target }
				if = { limit = { has_variable = gr_adventure_first_target } remove_variable = gr_adventure_first_target }
				if = { limit = { has_variable = gr_adventure_realm } remove_variable = gr_adventure_realm }
				if = { limit = { has_variable = gr_adventure_first_realm } remove_variable = gr_adventure_first_realm }
				if = { limit = { has_variable = gr_adventure_count } remove_variable = gr_adventure_count }
				if = { limit = { has_variable = gr_adventure_tier } remove_variable = gr_adventure_tier }
				if = { limit = { has_variable = gr_memory_created_kingdom } remove_variable = gr_memory_created_kingdom }
				if = { limit = { has_variable = gr_created_kingdom } remove_variable = gr_created_kingdom }
				if = { limit = { has_variable = gr_memory_created_duchy } remove_variable = gr_memory_created_duchy }
				if = { limit = { has_variable = gr_created_duchy } remove_variable = gr_created_duchy }
				if = { limit = { has_variable = gr_memory_created_empire } remove_variable = gr_memory_created_empire }
				if = { limit = { has_variable = gr_empire_title } remove_variable = gr_empire_title }
				if = { limit = { has_variable = gr_empire_type } remove_variable = gr_empire_type }
				if = { limit = { has_variable = gr_memory_claimed_hegemony } remove_variable = gr_memory_claimed_hegemony }

				if = { limit = { has_variable = gr_festival_count }

				# --- LEARNING (knowledge & faith) ---
				set_variable = { name = gr_score_learning value = 0 }
				# Pilgrimages (40 each)
				if = { limit = { has_variable = gr_pilgrimage_count }

				# Expeditions (25 each, cap 75)
				if = { limit = { has_variable = gr_expedition_count }

				# Torture (15 each, cap 75)

				}
				# Executions (10 each, cap 75)
				if = { limit = { has_variable = gr_execution_count }
					change_variable = { name = gr_score_intrigue add = { value = var:gr_execution_count multiply = 10 max = 75 } }

				# Hooks (10 each, cap 75)
				if = { limit = { has_variable = gr_hook_count }

				if = { limit = { has_variable = gr_dread var:gr_dread >= 90 } change_variable = { name = gr_score_intrigue add = 100 } }
				else_if = { limit = { has_variable = gr_dread var:gr_dread >= 75 } change_variable = { name = gr_score_intrigue add = 60 } }
				else_if = { limit = { has_variable = gr_dread var:gr_dread >= 50 } change_variable = { name = gr_score_intrigue add = 30 } }

				# --- DETERMINE TOP & SECOND CATEGORY ---
				set_variable = { name = gr_score_top_val value = var:gr_score_foundation }
				set_variable = { name = gr_score_top_cat value = flag:foundation }
				set_variable = { name = gr_score_second_val value = 0 }
				set_variable = { name = gr_score_second_cat value = flag:none }

				if = { limit = { var:gr_score_warfare > var:gr_score_top_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_top_val }
					set_variable = { name = gr_score_second_cat value = var:gr_score_top_cat }
					set_variable = { name = gr_score_top_val value = var:gr_score_warfare }
					set_variable = { name = gr_score_top_cat value = flag:warfare }
				}
				else_if = { limit = { var:gr_score_warfare > var:gr_score_second_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_warfare }
					set_variable = { name = gr_score_second_cat value = flag:warfare }
				}
				if = { limit = { var:gr_score_stewardship > var:gr_score_top_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_top_val }
					set_variable = { name = gr_score_second_cat value = var:gr_score_top_cat }
					set_variable = { name = gr_score_top_val value = var:gr_score_stewardship }
					set_variable = { name = gr_score_top_cat value = flag:stewardship }
				}
				else_if = { limit = { var:gr_score_stewardship > var:gr_score_second_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_stewardship }
					set_variable = { name = gr_score_second_cat value = flag:stewardship }
				}
				if = { limit = { var:gr_score_diplomacy > var:gr_score_top_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_top_val }
					set_variable = { name = gr_score_second_cat value = var:gr_score_top_cat }
					set_variable = { name = gr_score_top_val value = var:gr_score_diplomacy }
					set_variable = { name = gr_score_top_cat value = flag:diplomacy }
				}
				else_if = { limit = { var:gr_score_diplomacy > var:gr_score_second_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_diplomacy }
					set_variable = { name = gr_score_second_cat value = flag:diplomacy }

					set_variable = { name = gr_score_second_val value = var:gr_score_top_val }
					set_variable = { name = gr_score_second_cat value = var:gr_score_top_cat }
					set_variable = { name = gr_score_top_val value = var:gr_score_learning }
					set_variable = { name = gr_score_top_cat value = flag:learning }

					set_variable = { name = gr_score_second_val value = var:gr_score_learning }

					set_variable = { name = gr_score_second_val value = var:gr_score_top_val }
					set_variable = { name = gr_score_second_cat value = var:gr_score_top_cat }
					set_variable = { name = gr_score_top_val value = var:gr_score_intrigue }
					set_variable = { name = gr_score_top_cat value = flag:intrigue }

				else_if = { limit = { var:gr_score_intrigue > var:gr_score_second_val }
					set_variable = { name = gr_score_second_val value = var:gr_score_intrigue }
					set_variable = { name = gr_score_second_cat value = flag:intrigue }

					set_variable = { name = gr_score_tier value = flag:legendary }

				else_if = {
					limit = { var:gr_score >= 800 }
					set_variable = { name = gr_score_tier value = flag:magnificent }
				}
				else_if = {
					limit = { var:gr_score >= 500 }
					set_variable = { name = gr_score_tier value = flag:distinguished }
				}
				else_if = {
					limit = { var:gr_score >= 200 }
					set_variable = { name = gr_score_tier value = flag:notable }

						value = root.dynasty.var:gr_dynasty_total

					set_variable = { name = gr_dynasty_total value = var:gr_score }

							change_variable = { name = gr_renown_edu add = 3 }
						}
					}

					if = { limit = { has_variable = gr_memory_great_battle }
						if = { limit = { has_variable = gr_battle_tier var:gr_battle_tier = flag:legendary }
							change_variable = { name = gr_renown_edu add = 10 }
						}
						else_if = { limit = { has_variable = gr_battle_tier var:gr_battle_tier = flag:massive }
							change_variable = { name = gr_renown_edu add = 7 }
						}
						else = {
							change_variable = { name = gr_renown_edu add = 4 }
						}
					}
					# Martial lifestyle trait
					if = { limit = { OR = { has_variable = gr_life_strategist has_variable = gr_life_overseer has_variable = gr_life_gallant } }
						change_variable = { name = gr_renown_edu add = 8 }
					}
					# Martial is top stat
					if = { limit = { has_variable = gr_top_stat var:gr_top_stat = flag:martial }
						change_variable = { name = gr_renown_edu add = 6 }
					}
				}

				# DIPLOMACY education
				if = {
					limit = { has_variable = gr_edu var:gr_edu = flag:diplomacy }

						change_variable = { name = gr_renown_edu add = 10 }
					}
					else_if = { limit = { has_variable = gr_prestige_level var:gr_prestige_level >= 4 }
						change_variable = { name = gr_renown_edu add = 7 }
					}
					else_if = { limit = { has_variable = gr_prestige_level var:gr_prestige_level >= 2 }

						change_variable = { name = gr_renown_edu add = 10 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 20 }
						change_variable = { name = gr_renown_edu add = 7 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 10 }
						change_variable = { name = gr_renown_edu add = 4 }

						change_variable = { name = gr_renown_edu add = 8 }
					}

					if = { limit = { has_variable = gr_top_stat var:gr_top_stat = flag:diplomacy }
						change_variable = { name = gr_renown_edu add = 6 }

						change_variable = { name = gr_renown_edu add = 10 }

					}
					else_if = { limit = { has_variable = gr_gold var:gr_gold >= 200 }
						change_variable = { name = gr_renown_edu add = 4 }
					}

					if = { limit = { has_variable = gr_memory_created_kingdom var:gr_tier >= 5 }
						change_variable = { name = gr_renown_edu add = 10 }
					}
					else_if = { limit = { has_variable = gr_memory_created_kingdom }
						change_variable = { name = gr_renown_edu add = 7 }
					}
					else_if = { limit = { has_variable = gr_memory_conquered_realm }
						change_variable = { name = gr_renown_edu add = 4 }
					}
					# Stewardship lifestyle trait
					if = { limit = { OR = { has_variable = gr_life_architect has_variable = gr_life_administrator has_variable = gr_life_avaricious } }
						change_variable = { name = gr_renown_edu add = 8 }
					}
					# Stewardship is top stat
					if = { limit = { has_variable = gr_top_stat var:gr_top_stat = flag:stewardship }

				if = {
					limit = { has_variable = gr_edu var:gr_edu = flag:intrigue }
					# Fear
					if = { limit = { has_variable = gr_dread var:gr_dread >= 80 }
						change_variable = { name = gr_renown_edu add = 10 }
					}
					else_if = { limit = { has_variable = gr_dread var:gr_dread >= 50 }
						change_variable = { name = gr_renown_edu add = 7 }
					}
					else_if = { limit = { has_variable = gr_dread var:gr_dread >= 20 }
						change_variable = { name = gr_renown_edu add = 4 }
					}
					# Survival
					if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 25 }
						change_variable = { name = gr_renown_edu add = 10 }
					}

						change_variable = { name = gr_renown_edu add = 7 }
					}

					if = { limit = { OR = { has_variable = gr_life_schemer has_variable = gr_life_seducer has_variable = gr_life_torturer } }

					}

					if = { limit = { OR = { has_variable = gr_life_scholar has_variable = gr_life_theologian has_variable = gr_life_wholebody } }
						change_variable = { name = gr_renown_edu add = 8 }
					}

					if = { limit = { has_variable = gr_top_stat var:gr_top_stat = flag:learning }

						change_variable = { name = gr_renown_edu multiply = { value = 0.8 } }
					}
					else_if = { limit = { var:gr_edu_tier = 1 }
						change_variable = { name = gr_renown_edu multiply = { value = 0.6 } }
					}

				if = { limit = { var:gr_renown_edu > 40 }

				set_variable = { name = gr_renown_personality value = 0 }

				# Each aligned trait: Bronze +4, Silver +8, Gold +20
				# Cap at 60 (3 traits at Gold)

				# Ambitious
				if = { limit = { has_variable = gr_trait_ambitious }
					if = { limit = { has_variable = gr_memory_created_kingdom var:gr_tier >= 5 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_memory_created_kingdom }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_memory_conquered_realm }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Content
				if = { limit = { has_variable = gr_trait_content }
					if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 35 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 25 }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 15 }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Brave
				if = { limit = { has_variable = gr_trait_brave }
					if = { limit = { has_variable = gr_memory_led_troops has_variable = gr_memory_great_battle has_variable = gr_battle_tier var:gr_battle_tier = flag:legendary }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_memory_led_troops has_variable = gr_memory_great_battle }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_memory_led_troops }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Craven
				if = { limit = { has_variable = gr_trait_craven }
					if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 35 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 25 }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 15 }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Greedy
				if = { limit = { has_variable = gr_trait_greedy }
					if = { limit = { has_variable = gr_gold var:gr_gold >= 800 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_gold var:gr_gold >= 400 }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_gold var:gr_gold >= 150 }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Generous
				if = { limit = { has_variable = gr_trait_generous }
					if = { limit = { has_variable = gr_prestige_level var:gr_prestige_level >= 6 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_prestige_level var:gr_prestige_level >= 4 }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_prestige_level var:gr_prestige_level >= 2 }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Zealous
				if = { limit = { has_variable = gr_trait_zealous }
					if = { limit = { has_variable = gr_piety_level var:gr_piety_level >= 6 }
						change_variable = { name = gr_renown_personality add = 20 }
					}
					else_if = { limit = { has_variable = gr_piety_level var:gr_piety_level >= 4 }
						change_variable = { name = gr_renown_personality add = 8 }
					}
					else_if = { limit = { has_variable = gr_piety_level var:gr_piety_level >= 2 }
						change_variable = { name = gr_renown_personality add = 4 }
					}
				}
				# Cynical

					change_variable = { name = gr_renown_achieve add = 3 }
				}
				# Reign length
				if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 40 }
					change_variable = { name = gr_renown_achieve add = 10 }
				}
				else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 30 }
					change_variable = { name = gr_renown_achieve add = 6 }
				}
				else_if = { limit = { has_variable = gr_reign_years var:gr_reign_years >= 20 }
					change_variable = { name = gr_renown_achieve add = 3 }
				}
				# Treasury
				if = { limit = { has_variable = gr_gold var:gr_gold >= 1000 }
					change_variable = { name = gr_renown_achieve add = 8 }
				}
				else_if = { limit = { has_variable = gr_gold var:gr_gold >= 500 }
					change_variable = { name = gr_renown_achieve add = 5 }
				}
				else_if = { limit = { has_variable = gr_gold var:gr_gold >= 200 }
					change_variable = { name = gr_renown_achieve add = 2 }
				}

				# Cap achievement at 150
				if = { limit = { var:gr_renown_achieve > 150 }
					set_variable = { name = gr_renown_achieve value = 150 }
				}

				# --- TOTAL RENOWN REWARD ---
				set_variable = {
					name = gr_renown_reward
					value = {
						value = var:gr_renown_edu
						add = var:gr_renown_personality
						add = var:gr_renown_achieve
					}
				}
				# Hard cap at 250
				if = { limit = { var:gr_renown_reward > 250 }
					set_variable = { name = gr_renown_reward value = 250 }
				}

				# =============================================
				# SECTION Q: CHRONICLE STORAGE
				# Save compact summary on the dead character so
				# the dynasty chronicle page reader can display
				# each past ruler's page without regenerating the
				# full obituary text. Also add dead char to dynasty
				# variable list for iteration.
				# =============================================
				root = {
					# Score and tier (computed on heir, copy back to dead char)
					set_variable = { name = gr_obit_score value = scope:heir.var:gr_score }
					set_variable = { name = gr_obit_score_tier value = scope:heir.var:gr_score_tier }
					set_variable = { name = gr_obit_top_cat value = scope:heir.var:gr_score_top_cat }
					set_variable = { name = gr_obit_reign_years value = scope:heir.var:gr_reign_years }
					set_variable = { name = gr_obit_death_reason value = scope:heir.var:gr_death_reason }
					set_variable = { name = gr_obit_death_year value = scope:heir.var:gr_death_year }
					set_variable = { name = gr_obit_death_age value = scope:heir.var:gr_death_age }
					set_variable = { name = gr_obit_tier value = scope:heir.var:gr_tier }
					set_variable = { name = gr_obit_children value = scope:heir.var:gr_children_count }
					# Whether a chronicler wrote the obituary
					if = {
						limit = { scope:heir = { has_variable = gr_chronicler } }
						set_variable = { name = gr_obit_had_chronicler value = yes }
					}
					# Key achievements as flags
					if = {
						limit = { scope:heir = { has_variable = gr_memory_created_empire } }
						set_variable = { name = gr_obit_created_empire value = yes }
					}
					if = {
						limit = { scope:heir = { has_variable = gr_memory_created_kingdom } }
						set_variable = { name = gr_obit_created_kingdom value = yes }
					}
					if = {
						limit = { scope:heir = { has_variable = gr_memory_great_battle } }
						set_variable = { name = gr_obit_great_battle value = yes }
					}
					if = {
						limit = { scope:heir = { has_variable = gr_memory_faith_created } }
						set_variable = { name = gr_obit_created_faith value = yes }
					}
					if = {
						limit = { scope:heir = { has_variable = gr_memory_culture_created } }
						set_variable = { name = gr_obit_created_culture value = yes }
					}
					# Mark as having been processed by Grand Remembrance
					set_variable = { name = gr_obit_processed value = yes }
				}

				# =============================================
				# FIRE THE OBITUARY EVENT
				# Placed before chronicle artifact creation so that
				# any errors in the artifact/dynasty section cannot
				# prevent the popup from appearing.
				# =============================================
				trigger_event = {
					id = gr_obituary.10
					days = 1
				}

				# Create new story for the heir (this code runs ON the heir).
				# By the time on_actions finish, player control has switched.
				# Uses same timing window as the obituary popup.
				if = {
					limit = {
						is_ai = no
						gr_story_supported = yes
						NOT = {
							any_owned_story = {
								story_type = story_cycle_grand_remembrance
							}
						}
					}
					create_story = {
						type = story_cycle_grand_remembrance
					}
				}

				# Add to dynasty remembered rulers list AND create chronicle page artifact.
				# The page artifact carries a full snapshot of the heir's gr_* obit
				# vars, enabling the Dynasty Chronicle window to replay the complete
				# obituary narrative via gr_obituary.11 (scope = artifact).
				#
				# gr_remembered_rulers (char list) and gr_remembered_chronicle_pages
				# (artifact list) MUST stay in lockstep — every ruler added gets a page.
				# See CHRONICLE_ARTIFACT_ARCHITECTURE.md for the full design rationale.
				if = {
					limit = { root = { exists = dynasty } }
					# Chronicle filter game rule (audit M4): under "Qualifying Only",
					# Forgettable-tier rulers are hidden from the Dynasty Chronicle --
					# no list entry, no page artifact. Applied at registration time so
					# the rulers/pages lists stay lockstep and existing saves keep what
					# they already recorded. gr_score_tier lives on THIS (the heir) here.
					if = {
						limit = {
							OR = {
								NOT = { has_game_rule = gr_chronicle_qualifying }
								NOT = { has_variable = gr_score_tier }
								NOT = { var:gr_score_tier = flag:forgettable }
							}
						}

						root.dynasty = {
							add_to_variable_list = {
								name = gr_remembered_rulers
								target = root
							}
							set_variable = { name = gr_has_remembered_rulers value = yes }
						}

						# Create chronicle page artifact, copying all gr_* vars from scope:heir.
						# See common/scripted_effects/gr_chronicle_page_effect.txt
						gr_create_chronicle_page_effect = yes

						# Push the page artifact into the dynasty list in lockstep with
						# gr_remembered_rulers (same index = same ruler on both lists).
						root.dynasty = {
							add_to_variable_list = {
								name = gr_remembered_chronicle_pages
								target = scope:chronicle_page
							}
						}
					}

						# ---- Prune raw tracking vars off the dead ruler (audit M3) ----
						# We are inside the `root has dynasty` branch, so the chronicle
						# page artifact was just created above
						# (gr_create_chronicle_page_effect) and holds a full snapshot of
						# every gr_* var (copied from scope:heir). Nothing post-death reads
						# the dead ruler's OWN raw vars: gr_obituary.10 (deferred 1 day)
						# reads scope:heir's copies, and no script chain reads
						# gr_predecessor.var:* off the corpse. So strip the ~350-var
						# lifetime snapshot off root to cut save bloat. gr_obit_* compact
						# vars (set on root above) are NOT in the pruned list, so the
						# chronicle reader summary survives. Runs on root (the dead ruler),
						# never on scope:heir.
						root = {
							gr_dead_ruler_prune_effect = yes
						}
				}
			}
		}
	}
}
