# Grand Remembrance - Decisions
# Runtime rebase: mirror the character open flag to a global GUI-safe flag.

gr_open_dynasty_chronicle = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}

	desc = gr_open_dynasty_chronicle_desc
	selection_tooltip = gr_open_dynasty_chronicle_tooltip

	is_shown = {
		exists = dynasty
		is_ai = no
	}

	is_valid = {
		custom_description = {
			text = gr_chronicle_need_remembered_rulers
			dynasty = { has_variable = gr_has_remembered_rulers }
		}
	}

	effect = {
		custom_tooltip = gr_open_dynasty_chronicle_tooltip
		hidden_effect = {
			set_variable = { name = gr_cw_count value = 0 }
			dynasty = {
				every_in_list = {
					variable = gr_remembered_rulers
					limit = {
						exists = this
					}
					root = { change_variable = { name = gr_cw_count add = 1 } }
				}
			}
			if = {
				limit = { var:gr_cw_count > 0 }
				set_variable = {
					name = gr_chronicle_page
					value = {
						value = var:gr_cw_count
						subtract = 1
					}
				}
			}
			else = {
				set_variable = { name = gr_chronicle_page value = 0 }
			}
			remove_variable = gr_cw_count

			gr_cw_page_setup_effect = yes

			set_variable = { name = gr_chronicle_open value = 1 }
			set_global_variable = {
				name = gr_chronicle_window_open
				value = yes
			}
		}
	}

	ai_check_interval = 0
}
