# ============================================================
# .0001 — Eligible candidates comes of age
# ============================================================
agot_summerhall.0001 = {
	hidden = yes

	trigger = {
		title:h_the_iron_throne.holder ?= { save_temporary_scope_as = ruler }
		scope:ruler = {
			dynasty = dynasty:dynn_Targaryen
			OR = {
				has_title = title:k_summerhall
				NOT = { exists = title:k_summerhall.holder }
			}
			has_title = title:d_summerhall
			has_title = title:c_summerfield
			OR = {
				# Every male child (excluding primary heir) of the ruler must be 16+
				AND = {
					NOT = {
						any_child = {
							is_male = yes
							age < 16
							NOT = { is_primary_heir_of = scope:ruler }
						}
					}
					any_child = {
						is_male = yes
						NOT = { is_primary_heir_of = scope:ruler }
					}
				}
				# At least 3 male children (excluding primary heir) of the ruler must be 16+
				any_child = {
					count >= 3
					is_male = yes
					age >= 16
					NOT = { is_primary_heir_of = scope:ruler }
				}
			}
		}
	}

	immediate = {
		title:k_dummy_summerhall = {
            copy_title_history = title:k_summerhall
        }
        title:d_dummy_summerhall = {
            copy_title_history = title:d_summerhall
        }
        title:c_dummy_summerfield = {
            copy_title_history = title:c_summerfield
        }
		title:h_the_iron_throne.holder = {
			trigger_event = agot_summerhall.0002
		}
	}
}

		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change10

		}

		#Avoid repeats
		hidden_effect = {
			title:c_summerfield = {
				change_title_holder_include_vassals = {
					holder = scope:candidate_4
					change = scope:change10
				}
				copy_title_history = title:c_dummy_summerfield
			}

				change_title_holder_include_vassals = {

			}

					type = created

			type = granted

			add_claim_on_loss = no
		}

		title:k_summerhall = {
			every_in_de_jure_hierarchy = {
				limit = {
					holder = root
				}

				change_title_holder_include_vassals = {
					holder = scope:candidate_4
					change = scope:change12
				}
			}
		}
		resolve_title_and_vassal_change = scope:change12

		title:k_summerhall.holder = { save_scope_as = new_prince_of_summerhall }

		if = {

	}

	}
