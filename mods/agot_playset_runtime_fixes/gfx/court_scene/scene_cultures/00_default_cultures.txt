steppe = {
	trigger = {
		# Scene selection is briefly queried without a valid court owner during cleanup.
		trigger_if = {
			limit = { exists = root }
			has_mpo_dlc_trigger = yes
			OR = {
				government_has_flag = government_is_nomadic
				# COMPATCH: tribal Dothraki khalasars (Essos Expanded's Sarne khalasars etc.)
				# were falling into the generic western hall because their capitals fail the
				# nomad/tribal holding-type check below. Route them to the steppe court directly.
				AND = {
					has_government = tribal_government
					culture = culture:dothraki
				}
				capital_county ?= {
					culture = {
						has_graphical_steppe_culture_group_trigger = yes
					}
					title_province ?= {
						has_holding = yes
						OR = {
							has_holding_type = nomad_holding
							has_holding_type = tribal_holding
						}
					}
				}
				#For tribal northerns... this is better than a big feudal western castle, right?
				capital_county ?= {
					culture = {
						OR = {
							has_graphical_steppe_culture_group_trigger = yes
							has_graphical_western_culture_group_trigger = yes
						}
						#AGOT Modified
						# OR = {

						# 	has_cultural_pillar = heritage_balto_finnic
						# }

					}
					title_province ?= {
						has_holding = yes
						OR = {
							has_holding_type = nomad_holding
							has_holding_type = tribal_holding
						}
					}
				}
			}
			#AGOT Disabled
			# trigger_if = {
			# 	limit = {
