steppe = {
	trigger = {
		# Scene selection is briefly queried without a valid court owner during cleanup.
		trigger_if = {
			limit = { exists = root }
			has_mpo_dlc_trigger = yes
			OR = {
				government_has_flag = government_is_nomadic
				# COMPATCH: tribal Dothraki khalasars (Essos Expanded's Sarne khalasars etc.)
				# were falling into the generic western hall because their capitals fail the
				# nomad/tribal holding-type check below. Route them to the steppe court directly.
				AND = {
					has_government = tribal_government
					culture = culture:dothraki
				}
				capital_county ?= {
					culture = {
						has_graphical_steppe_culture_group_trigger = yes
					}
					title_province ?= {
						has_holding = yes
						OR = {
							has_holding_type = nomad_holding
							has_holding_type = tribal_holding
						}
					}
				}
				#For tribal northerns... this is better than a big feudal western castle, right?
				capital_county ?= {
					culture = {
						OR = {
							has_graphical_steppe_culture_group_trigger = yes
							has_graphical_western_culture_group_trigger = yes
						}
						#AGOT Modified
						# OR = {
						# 	has_cultural_pillar = heritage_ugro_permian
						# 	has_cultural_pillar = heritage_volga_finnic
						# 	has_cultural_pillar = heritage_balto_finnic
						# }
						agot_is_wildling_culture = yes
					}
					title_province ?= {
						has_holding = yes
						OR = {
							has_holding_type = nomad_holding
							has_holding_type = tribal_holding
						}
					}
				}
			}
			#AGOT Disabled
			# trigger_if = {
			# 	limit = {
			# 		has_tgp_dlc_trigger = yes
			# 		NOT = {
			# 			government_has_flag = government_is_nomadic
			# 		}
			# 		tgp_is_ceremonial_regent_trigger = no
			# 	}
			# 	NOT = { primary_title ?= title:h_china }
			# }
			#AGOT Added
			agot_has_throne_room = no
			amsb_has_throne_room = no
			# Legacy of Valyria: reserve dedicated LoV court scenes
			lv_has_throne_room = no
		}
		trigger_else = { always = no }
	}
}

mena = {
	trigger = {
		# Scene selection is briefly queried without a valid court owner during cleanup.
		trigger_if = {
			limit = { exists = root }
			# COMPATCH: re-enabled for DORNE ONLY (AGOT uses this room as the Dornish court).
			# Still disabled everywhere else (Essos, Asshai...).
			capital_province ?= { geographical_region = world_westeros_dorne }
			trigger_if = {
				limit = {
					has_mpo_dlc_trigger = no
				}
				exists = capital_county
				capital_county = {
					OR = {
						culture = {
							OR = {
								has_graphical_mena_culture_group_trigger = yes
								has_graphical_iranian_culture_group_trigger = yes
								has_graphical_african_culture_group_trigger = yes
								AND = {
									has_graphical_iberian_culture_group_trigger  = yes
									#AGOT Modified
									# prev.religion = religion:islam_religion
									any_parent_culture_or_above = {
										this = culture:rhoynar_original
									}
								}
							}
						}
						AND = {
							culture = {
								has_graphical_steppe_culture_group_trigger = yes
							}
							title_province = {
								steppe_building_gfx_use_mediterranean_trigger = no
								steppe_building_gfx_use_indian_trigger = no
								steppe_building_gfx_use_western_trigger = no
								steppe_building_gfx_use_east_asian_trigger = no
								steppe_building_gfx_use_japanese_trigger = no
								steppe_building_gfx_use_southeast_asian_trigger = no
							}
						}
					}
				}
			}
			trigger_else = {
				NOT = {
					government_has_flag = government_is_nomadic
				}
				exists = capital_county
				OR = {
					capital_county = {
						culture = { has_graphical_steppe_culture_group_trigger = yes }
						title_province = {
							steppe_building_gfx_use_mediterranean_trigger = no
							steppe_building_gfx_use_indian_trigger = no
							steppe_building_gfx_use_western_trigger = no
							steppe_building_gfx_use_east_asian_trigger = no
							steppe_building_gfx_use_japanese_trigger = no
							steppe_building_gfx_use_southeast_asian_trigger = no
						}
						title_province ?= {
							has_holding = yes
							NOT = {
								has_holding_type = tribal_holding
							}
						}
					}
					capital_county.culture = {
						OR = {
							has_graphical_mena_culture_group_trigger = yes
							has_graphical_iranian_culture_group_trigger = yes
							has_graphical_african_culture_group_trigger = yes
							AND = {
								has_graphical_iberian_culture_group_trigger  = yes
								#AGOT Modified
								# prev.religion = religion:islam_religion
								any_parent_culture_or_above = {
									this = culture:rhoynar_original
								}
							}
						}
					}
				}
			}
			#AGOT Disabled
			# trigger_if = {
			# 	limit = {
			# 		has_tgp_dlc_trigger = yes
			# 		NOT = {
			# 			government_has_flag = government_is_nomadic
			# 		}
			# 		tgp_is_ceremonial_regent_trigger = no
			# 	}
			# 	NOT = { primary_title ?= title:h_china }
			# }
			#AGOT Added
			agot_has_throne_room = no
			amsb_has_throne_room = no
			# Legacy of Valyria: reserve dedicated LoV court scenes
			lv_has_throne_room = no
		}
		trigger_else = { always = no }
	}
}

western = {
	trigger = {
		# Scene selection is briefly queried without a valid court owner during cleanup.
		trigger_if = {
			limit = { exists = root }
			# COMPATCH v1.1.8: broadened into the universal fallback court so no ruler
			# ever falls into the vanilla chinese/japanese/indian/SEA/mena/mediterranean rooms.
			NOT = {
				government_has_flag = government_is_nomadic
			}
			# Compatch: steppe court claims tribal Dothraki khalasars - keep the universal
			# fallback from tying with (and beating) the steppe scene for them.
			NOT = {
				AND = {
					has_government = tribal_government
					culture = culture:dothraki
				}
			}
			#AGOT Added
			agot_has_throne_room = no
			amsb_has_throne_room = no
			#LEGACY OF VALYRIA ADDED
			lv_has_throne_room = no
		}
		trigger_else = { always = no }
	}
}
