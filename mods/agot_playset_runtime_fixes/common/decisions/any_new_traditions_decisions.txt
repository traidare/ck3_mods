	selection_tooltip = any_new_traditions_1_tooltip

		OR = {

				name = unavailable_unique_decisions

		prestige_level >= 5
		OR = {
			custom_tooltip = {
				text = 1000_of_riders_plains
				number_maa_soldiers_of_type = {
					target = riders_plains
					value >= 1000
				}
			}
			custom_tooltip = {
				text = 500_of_riders_plains
				primary_title = {
					any_owned_title_maa_regiment = {
						is_maa_type = riders_plains

						maa_size >= 5
					}
				}
			}
			custom_tooltip = {
				text = 1000_of_horse_archers_plains
				number_maa_soldiers_of_type = {
					target = h_archers_p
					value >= 1000
				}
			}
			custom_tooltip = {
				text = 500_of_horse_archers_plains
				primary_title = {
					any_owned_title_maa_regiment = {
						is_maa_type = h_archers_p
						count >= 2
						maa_size >= 5
					}
				}
			}
		}
		OR = {
			custom_tooltip = {
				text = 500_of_knights_plains
				number_maa_soldiers_of_type = {
					target = knights_plains
					value >= 500
				}
			}
			custom_tooltip = {
				text = 250_of_knights_plains
				primary_title = {
					any_owned_title_maa_regiment = {
						is_maa_type = knights_plains
						count >= 2
						maa_size >= 5
					}
				}
			}
		}
	}

	effect = {
        root.culture = {
            custom_tooltip = {
                text = remove_tradition_by_lance
                remove_culture_tradition = tradition_by_lance
            }
        }
        root.culture = {
            custom_tooltip = {
                text = add_tradition_by_lance2
                add_culture_tradition = tradition_by_lance2
            }
			custom_tooltip = { text = change_h_archers_p }
			custom_tooltip = { text = change_riders_plains }
			custom_tooltip = { text = change_knights_plains }
        }

			add_dynasty_modifier = ary_traditions_1_modifier
