# A travel party crosses the same sea province as the kraken.
kraken.0100 = {
	type = character_event
	title = kraken.0100.t
	desc = kraken.0100.desc
	theme = travel_danger

	immediate = {
		if = {
			limit = {
				exists = current_travel_plan
				current_travel_plan = {
					any_entourage_character = {
						is_alive = yes
						NOT = { this = root }
					}
				}
			}
			current_travel_plan = {
				random_entourage_character = {
					limit = {
						is_alive = yes
						NOT = { this = root }
					}
					save_scope_as = kraken_endangered_traveler
				}
			}
		}
	}

	option = {
		name = kraken.0100.a
		custom_tooltip = kraken.0100.a.tt
		hidden_effect = {
			if = {
				limit = { exists = scope:kraken_endangered_traveler }
				random = {
					chance = 35
					scope:kraken_endangered_traveler = {
						if = {
							limit = { agot_ce_event_death_protected_trigger = no }
							death = { death_reason = death_kraken_attack }
						}
					}
				}
			}
			else = {
				random = {
					chance = 15
					if = {
						limit = { agot_ce_event_death_protected_trigger = no }
						death = { death_reason = death_kraken_attack }
					}
				}
			}
		}
	}
}

# The Kraken's terror is checked separately from combat skill.
kraken.1001 = {
	type = character_event
	title = kraken.1001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = kraken_challenge_fear_failed }
				desc = kraken.1001.desc.failed
			}
			desc = kraken.1001.desc.held
		}
	}
	theme = dread
	override_background = { reference = kraken_sea_surface }
	right_portrait = {
		character = scope:kraken
		camera = camera_kraken_event_right
	}

	immediate = {
		random_list = {
			60 = {
				modifier = { factor = 2 has_trait = brave }
				modifier = { factor = 0.5 has_trait = craven }
				modifier = { factor = 1.5 has_character_flag = kraken_challenge_has_horn }
				modifier = { factor = 1.5 var:kraken_challenge_balance >= 2 }
				modifier = { factor = 0.65 var:kraken_challenge_balance <= -2 }
				modifier = { factor = 0.7 scope:kraken = { var:kraken_dread >= 200 } }
				remove_character_flag = kraken_challenge_fear_failed
			}
			40 = {
				add_character_flag = kraken_challenge_fear_failed
			}
		}
	}

	option = {
		name = kraken.1001.a
		trigger = { NOT = { has_character_flag = kraken_challenge_fear_failed } }
		hidden_effect = { trigger_event = kraken.1010 }
	}
	option = {
		name = kraken.1001.b
		trigger = { has_character_flag = kraken_challenge_fear_failed }
		add_stress = medium_stress_gain
		hidden_effect = {
			change_variable = { name = kraken_challenge_score subtract = 10 }
			kraken_calculate_challenge_balance_effect = yes
			trigger_event = kraken.1010
		}
	}
	option = {
		name = kraken.1001.c
		trigger = { has_character_flag = kraken_challenge_fear_failed }
		hidden_effect = { kraken_cleanup_challenge_effect = yes }
	}
}

	type = character_event

	desc = {
		first_valid = {
			triggered_desc = { trigger = { has_character_flag = kraken_challenge_target_eye } desc = kraken.1101.desc.eye }

			trigger_event = kraken.1010
		}
	}

		name = kraken.challenge.withdraw

	}
