zzz_great_councils_event.0001 = { # A flavor event of a courtier telling you the emperor has died or been deposed and the heir is controversial (or that you called the council)
	type = character_event
	title = zzz_great_councils_event.0001.t
	desc = {
		first_valid = { # Display the localization of the first valid desc block which returns true
			triggered_desc = {
				trigger = {
					scope:zzz_great_councils_temp_emp = { has_character_flag = zzz_great_councils_inherit }
				}
				desc = zzz_great_councils_event.0001.a.desc
			}
			triggered_desc = {
				trigger = {
					scope:zzz_great_councils_temp_emp = { has_character_flag = zzz_great_councils_depose }
				}
				desc = zzz_great_councils_event.0001.b.desc
			}
			triggered_desc = {
				trigger = {
					scope:zzz_great_councils_emperor = { has_character_flag = zzz_great_councils_called_council }
					root = scope:zzz_great_councils_emperor
				}
				desc = zzz_great_councils_event.0001.c.desc
			}
			triggered_desc = {
				trigger = {
					scope:zzz_great_councils_emperor = { has_character_flag = zzz_great_councils_called_council }
					NOT = { root = scope:zzz_great_councils_emperor }
				}
				desc = zzz_great_councils_event.0001.d.desc
			}
		}
	}
	theme = crown
	override_background = garden

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = {
							OR = {
								scope:zzz_great_councils_temp_emp = { has_character_flag = zzz_great_councils_inherit }
								scope:zzz_great_councils_temp_emp = { has_character_flag = zzz_great_councils_depose }
							}
						}
						desc = zzz_great_councils_event.0001.a
					}
					triggered_desc = {
						trigger = {
							scope:zzz_great_councils_emperor = { has_character_flag = zzz_great_councils_called_council }
						}
						desc = zzz_great_councils_event.0001.c
					}
				}
			}
		}

		if = {
			limit = {
				scope:zzz_great_councils_pregnant ?= { is_alive = yes }
			}
			scope:zzz_great_councils_pregnant ?= {
				set_local_variable = {
					name = zzz_great_councils_preg_var
					value = {
						add = 281
						subtract = pregnancy_days
					}
				}
			}
		}
		if = {
			limit = {
				scope:zzz_great_councils_pregnant ?= { has_local_variable = zzz_great_councils_preg_var }
			}
			trigger_event = {
				id = zzz_great_councils_event.0002
				days = scope:zzz_great_councils_pregnant.var:zzz_great_councils_preg_var
			}
		}
		else_if = {
			limit = { NOT = { any_player = { is_in_list = zzz_great_councils_electors } } }
			trigger_event = zzz_great_councils_event.0002
		}
		else = {
			trigger_event = {
				id = zzz_great_councils_event.0002
				days = 15
			}
		}
	}
}
