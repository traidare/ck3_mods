# AGOT 0.4.39 / CK3 1.19 runtime guard.

# CK3 can query this rule without a potential-raider character, for example
# while character-dependent UI is being rebuilt. Preserve AGOT's rule for every
# valid character and return false for the missing-character case.
can_raid = {
	save_temporary_scope_as = potential_raider
	trigger_if = {
		limit = { exists = scope:potential_raider }
		scope:potential_raider ?= {
			can_raid_trigger = yes
		}
	}
	trigger_else = { always = no }
}
