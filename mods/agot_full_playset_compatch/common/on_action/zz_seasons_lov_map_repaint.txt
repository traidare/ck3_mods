# LoV & Seasons compatch: keep the seasonal map mode painted.
# Repaints yearly once the player has opened the seasons map mode at least once.
yearly_global_pulse = {
	on_actions = {
		zz_seasons_lov_yearly_map_repaint
	}
}

zz_seasons_lov_yearly_map_repaint = {
	trigger = {
		exists = global_var:seasons_map_mode
	}
	effect = {
		seasons_map_startup = yes
	}
}
