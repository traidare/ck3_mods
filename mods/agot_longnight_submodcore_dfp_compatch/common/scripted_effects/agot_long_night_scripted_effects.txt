

# The Long Night+: when the Others fall, the lands the Night King held are left scoured. Ordinary holdings
# collapse into wilderness (secondary holdings razed, development wiped); the great de jure seats are instead
# left as ruins, scaled to their standing (kingdom seat -> a renowned large ruin, duchy seat -> a medium ruin).
# The living who return south inherit a ravaged, depopulated waste to reclaim through AGOT's colonization and
# ruin systems. Must run in the_long_night on_end BEFORE the wights die, so the dead are still scopeable holders.
agot_long_night_scour_others_lands_effect = {
	every_county = {
		limit = {
			exists = holder
			holder = {
				OR = {
					has_trait = wight_trait
					has_trait = other_trait
				}
			}
		}
		save_scope_as = scour_county

		if = {
			# De jure seat of a kingdom -> a renowned great ruin (e.g. Winterfell, Storm's End).
			limit = {
				de_jure_liege ?= {
					de_jure_liege ?= {
						tier = tier_kingdom
						title_capital_county = scope:scour_county
					}
				}
			}
			agot_ruin_county_effect = yes
			title_province = {
				remove_building = medium_ruin_01
				add_building = large_ruin_01
				add_building = ruin_renowned_01
			}
		}
		else_if = {
			# De jure seat of a duchy -> a sizeable ruin.
			limit = {
				de_jure_liege ?= {
					tier = tier_duchy
					title_capital_county = scope:scour_county
				}
			}
			agot_ruin_county_effect = yes
		}
		else = {
			# Everything else simply returns to the wild.
			make_settlement_county_wilderness = { COUNTY = this }
		}
	}
}

escape_noble_effect = {
	# A landless courtier flees south to SAFETY rather than being raised — they join a safe southern court instead
	# of founding an adventurer camp. This keeps the laamp count (and the player's performance) in check, so no cap
	# gate is needed on this path. Priority: a relative, then a ruler of their own culture, then any safe ruler. If
	# no safe host exists at all they simply do not flee (and are raised as a wight) — the flag is set only on success.
	save_scope_as = fleeing_courtier
	if = {
		limit = { any_close_or_extended_family_member = { ln_safe_flee_host_trigger = yes } }
		random_close_or_extended_family_member = {
			limit = { ln_safe_flee_host_trigger = yes }
			save_scope_as = flee_host
		}
	}
	else_if = {
		limit = { any_ruler = { ln_safe_flee_host_trigger = yes  culture = scope:fleeing_courtier.culture } }
		random_ruler = {
			limit = { ln_safe_flee_host_trigger = yes  culture = scope:fleeing_courtier.culture }
			save_scope_as = flee_host
		}
	}
	else_if = {
		limit = { any_ruler = { ln_safe_flee_host_trigger = yes } }
		random_ruler = {
			limit = { ln_safe_flee_host_trigger = yes }
			save_scope_as = flee_host
		}
	}
	if = {
		limit = { exists = scope:flee_host }
		add_stress = medium_stress_impact_gain   # the shock of fleeing (these courtiers are never brave — gated in mutate_into_other_effect)
		add_character_flag = dawn_refugee
		add_character_flag = dawn_refugee_courtier       # path marker: fled as a landless courtier (vs dawn_refugee_lord) — for saves debugging + aftermath flavor
		add_character_modifier = dawn_refugee_modifier   # visible "Fled South" marker
		set_employer = scope:flee_host                   # take refuge in a safe southern court — no adventurer camp
	}
}

# Landed variant (used by the Flee South decision): a lord abandons their seat ahead of the dead.
flee_noble_south_effect = {
	# Remember the lord's former standing BEFORE the titles are lost below — it scales both the starting camp and
	# (later) the retinue they bring south. Camp level: count → 1 (default), duke → 2, king → 3, emperor/hegemon → 4.
	if = { limit = { highest_held_title_tier >= tier_empire } save_scope_value_as = { name = ln_camp_level value = 4 } }
	else_if = { limit = { highest_held_title_tier >= tier_kingdom } save_scope_value_as = { name = ln_camp_level value = 3 } }
	else_if = { limit = { highest_held_title_tier >= tier_duchy } save_scope_value_as = { name = ln_camp_level value = 2 } }
	else = { save_scope_value_as = { name = ln_camp_level value = 1 } }
	# PERSIST the standing and the lost seat as real variables (the scope value above dies with this effect):
	# the aftermath "Return North" pull reads them YEARS later to scale the settler retinue and aim the lord back
	# at his own ancestral county (the seat COUNTY, not the primary title — a fled king targets his capital county).
	set_variable = { name = ln_camp_level value = scope:ln_camp_level }
	if = {
		limit = { is_landed = yes  exists = capital_county }
		set_variable = { name = ln_lost_seat value = capital_county }
	}
	# Note the lord's cultural men-at-arms BEFORE he loses his realm, so the reduced retinue he brings south is of his
	# own people. We can't clone regiments across the loss of the realm, so we record the regional type here and
	# respawn a scaled echo once he is landless. No match (a minor or exotic lord) → no retinue, just gold to recruit.
	if = { limit = { "number_maa_regiments_of_type(frog_spears)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:north } }
	else_if = { limit = { "number_maa_regiments_of_type(river_bows)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:riverlands } }
	else_if = { limit = { "number_maa_regiments_of_type(finger_scouts)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:vale } }
	else_if = { limit = { "number_maa_regiments_of_type(royal_crossbowmen)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:crownlands } }
	else_if = { limit = { "number_maa_regiments_of_type(rain_bringers)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:westerlands } }
	else_if = { limit = { "number_maa_regiments_of_type(honeywine_quirrels)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:reach } }
	else_if = { limit = { "number_maa_regiments_of_type(marcher_longbowmen)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:stormlands } }
	else_if = { limit = { "number_maa_regiments_of_type(sun_spears)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:dorne } }
	else_if = { limit = { "number_maa_regiments_of_type(ironborn_reavers)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:ironman } }
	else_if = { limit = { "number_maa_regiments_of_type(clan_raiders)" >= 1 } save_scope_value_as = { name = ln_maa_region value = flag:wildling } }
	# The shock of fleeing — the brave are shamed to the core, the rest still shaken.
	if = { limit = { has_trait = brave } add_stress = massive_stress_impact_gain }
	else = { add_stress = medium_stress_impact_gain }
	# Their lands do NOT fall to the Night King — a kinsman or retainer left behind takes them up, so the dead must
	# still conquer them (the Night King keeps grinding). Priority: a capable adult vassal, else an adult courtier,
	# else a generated heir (never leave a title without a holder).
	if = {
		limit = { is_landed = yes }
		if = {
			limit = { any_vassal = { ln_flee_successor_trigger = yes } }
			ordered_vassal = {
				limit = { ln_flee_successor_trigger = yes }
				order_by = highest_held_title_tier
				max = 1
				check_range_bounds = no
				save_scope_as = flee_successor
			}
		}
		else_if = {
			limit = { any_courtier = { ln_flee_successor_trigger = yes } }
			random_courtier = {
				limit = { ln_flee_successor_trigger = yes }
				save_scope_as = flee_successor
			}
		}
		else = {
			create_character = {
				age = { 25 45 }
				culture = root.culture
				faith = root.faith
				location = capital_province
				save_scope_as = flee_successor
			}
		}
		create_title_and_vassal_change = { type = granted save_scope_as = flee_south_change }
		every_held_title = {
			change_title_holder = { holder = scope:flee_successor  change = scope:flee_south_change }
		}
		resolve_title_and_vassal_change = scope:flee_south_change
	}

# The Long Night+ — #12 the Wall. Fired from long_night_story when the dead take Castle Black: the middle of the Wall
# comes crashing down, a permanent scar (never restored). AGOT already builds the whole collapse system (meshes,
# animation, particles, sound, conditional rendering) — we only trigger it.
#   LAYER A (reliable, here): the exact building swap AGOT's own agot_wall_fall GUI performs — flag the fallen
#   variable + remove special_the_wall + add special_the_wall_damaged. The swap is MANDATORY, not just the variable:
#   graphical backgrounds only refresh when a building CHANGES in the holding (per 98_special_buildings.txt header).
#   The broken-middle graphical is already active in AGOT, so this renders the shattered Wall.
#   LAYER B (cinematic anim + sound CK3AGOT_wall_collapse_the_others + camera) is layered on once the agot_debug.1000
#   animation test confirms the falling animation actually plays.
agot_long_night_wall_collapse_effect = {
	# One-shot: only if the Wall still stands.
	if = {
		limit = {
			title:b_castle_black.title_province = {
				NOT = { has_variable = agot_wall_has_fallen_left }
				NOT = { has_variable = agot_wall_has_fallen_middle }
				NOT = { has_variable = agot_wall_has_fallen_right }
			}
		}
		# Cinematic for every human player. The widget event (long_night_plus_wall.1) plays the middle collapse
		# animation, pans the camera, plays the music, AND breaks the Wall ITSELF on finish (its GUI calls AGOT's
		# agot_wall_fall scripted GUI = sets agot_wall_has_fallen_middle + swaps the building). So we must NOT swap or
		# set any variable here — a second fallen variable would conflict (two set => no broken graphical renders, the
		# bug we hit). The breach is always the middle (the only animated collapse entity AGOT ships) — i.e. at Castle
		# Black itself.
		every_player = {
			trigger_event = { id = long_night_plus_wall.1 }
		}
		# Reliable fallback: with NO human player to play the widget (AI-only realm, or observer), the Wall would never
		# fall. Two days on, break it directly if it is still standing (mirrors what the widget's agot_wall_fall does).
		trigger_event = { id = long_night_plus_wall.3 days = 2 }
	}
}

mutate_into_other_effect = {
	# Nobles have a small chance to flee south ahead of the dead instead of being raised as
	# wights — they become flagged refugee adventurers (fodder for the War for the Dawn).
	if = {
		limit = {
			is_lowborn = no
			is_landed = no                       # landed rulers become wights keeping their title; only their landless kin/courtiers may flee
			is_adult = yes                       # no fleeing children
			dynasty ?= { dynasty_prestige_level >= 3 }   # only nobles of real standing flee — refugees worth rallying to the Dawn
			NOT = { has_trait = brave }                  # the brave stand and fall — they do not flee
			NOT = { has_trait = wight_trait }
			NOT = { has_trait = other_trait }
			NOT = { has_character_flag = dawn_refugee }
		}
		random = {
			chance = 3    # tunable: noble escape chance (rare — only notable houses flee). They join a safe southern
			              # court (escape_noble_effect), so this path creates NO adventurer camp and needs no cap gate.
			escape_noble_effect = yes
		}
	}
	# Everyone who did not flee is raised as a wight (undead thrall; keeps their title, becomes
	# the Night King's vassal). Identical to the old Other conversion but with the temporary
	# `wight` trait instead of `other_trait` (wights die when the Long Night ends).
	if = {
		limit = {
			NOT = { has_character_flag = dawn_refugee }   # did not flee
			NOT = { has_trait = other_trait }             # spare the true Others (Night King, spawned Others)
			NOT = { house = house:house_theothers }  # never raise actual House of the Others members as wights
		}
	if = {
		limit = { exists = capital_province }
		capital_province = {
			if = {
				limit = { OR = { has_holding_type = wilderness_holding has_holding_type = ruin_holding } }
				set_holding_type = tribal_holding
			}
		}
	}
	
	# All wights use wight_government — a NON-DYNASTIC clone of tribal_government. It keeps full realm mechanics
	# for the Night King's AI host, but non-dynastic = UNPLAYABLE (like theocracy/republic), so a conquered
	# PLAYER hands off to their heir/dynasty, or hits game over if none remain (heirs get conquered too → cascade).
	change_government = wight_government

	# NOTE: the wight is raised (add_trait = wight_trait + one-shot setup) AFTER the disease/age/wound purge
	# below — trait `opposites` BLOCK add_trait, so a diseased/old conquered noble could never become a wight
	# while those traits were still on them. See the one-shot block just before set_faith/set_culture.
	if = { limit = { has_trait = ruin } remove_trait = ruin }
	if = { limit = { has_trait = wilderness } remove_trait = wilderness }
	if = { limit = { has_trait = unknown } remove_trait = unknown }
	if = { limit = { has_trait = agot_dummy_trait } remove_trait = agot_dummy_trait }
	
	# Purge old age/disease traits
	if = { limit = { has_trait = withering_mind } remove_trait = withering_mind }
	if = { limit = { has_trait = fragile_bones } remove_trait = fragile_bones }
	if = { limit = { has_trait = faltering_heart } remove_trait = faltering_heart }
	if = { limit = { has_trait = clouded_eyes } remove_trait = clouded_eyes }
	if = { limit = { has_trait = infirm } remove_trait = infirm }
	if = { limit = { has_trait = consumption } remove_trait = consumption }
	if = { limit = { has_trait = leper } remove_trait = leper }
	if = { limit = { has_trait = bubonic_plague } remove_trait = bubonic_plague }
	if = { limit = { has_trait = ill } remove_trait = ill }
	if = { limit = { has_trait = pneumonic } remove_trait = pneumonic }
	if = { limit = { has_trait = typhus } remove_trait = typhus }
	if = { limit = { has_trait = smallpox } remove_trait = smallpox }
	if = { limit = { has_trait = measles } remove_trait = measles }
	if = { limit = { has_trait = greyscale } remove_trait = greyscale }
	if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
	if = { limit = { has_trait = wounded_2 } remove_trait = wounded_2 }
	if = { limit = { has_trait = wounded_3 } remove_trait = wounded_3 }
	if = { limit = { has_trait = maimed } remove_trait = maimed }
	if = { limit = { has_trait = blind } remove_trait = blind }
	if = { limit = { has_trait = incapable } remove_trait = incapable }
	if = { limit = { has_trait = one_legged } remove_trait = one_legged }
	if = { limit = { has_trait = disfigured } remove_trait = disfigured }
	if = { limit = { has_trait = early_great_pox } remove_trait = early_great_pox }
	if = { limit = { has_trait = great_pox } remove_trait = great_pox }

	set_character_faith = faith:cold_gods
	set_culture = culture:frostborn

	# Raise as a wight LAST — AFTER the disease/age/wound purge above, so no opposite trait blocks
	# add_trait = wight_trait (trait `opposites` prevent the add while the opposite is still present).
	# Everything that must run exactly ONCE per wight lives in this NOT-has-wight_trait guard, because
	# others_heartbeat re-runs this whole effect on existing wights.
	if = {
		limit = { NOT = { has_trait = wight_trait } }
		add_trait = wight_trait
		# Random gore variant (ln_wight_gore allele, selected by this flag in wight_trait_modifiers.txt).
		random_list = {
			1 = { add_character_flag = wight_v1 }
			1 = { add_character_flag = wight_v2 }
			1 = { add_character_flag = wight_v3 }
		}
		# Random idle pose (read by the idle group in gfx/portraits/portrait_animations/animations.txt) — one of
		# five barbershop poses, fixed per wight by the flag set here.
		random_list = {
			1 = { add_character_flag = wight_pose_wounded }
			1 = { add_character_flag = wight_pose_loss }
			1 = { add_character_flag = wight_pose_aggressive }
			1 = { add_character_flag = wight_pose_yield }
			1 = { add_character_flag = wight_pose_stayback }
		}
		# Dated "first death" memory (timestamped natively in AGOT's calendar). The wight's collapse at the

		create_character_memory = { type = wight_raised_memory }

		# The Long Night+: a raised wight belongs to the dead — sever every alliance it held in life. Otherwise a
		# former ally still shields it (you could not declare war on a realm allied to a wight).
		save_temporary_scope_as = ln_raised_wight
		every_ally = {
			break_alliance = scope:ln_raised_wight
		}
	}
	}
