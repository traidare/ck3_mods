E_kCAFG_try_trigger_cultural_benefits = {
    if = {
        limit = {
            is_landed = yes
            NOT = { has_realm_law_flag = F_kCAFG_intolerant_to_other_cultures }
        }
        #debug_log = "kCAFG - Cultural Benefits - Trying for [THIS.Char.GetNameNoTooltip]"
    
        save_temporary_scope_as = char
        culture = { save_temporary_scope_as = character_culture }
        clear_variable_list = domain_cultures
        every_directly_owned_province = {
            #debug_log = "kCAFG - Cultural Benefits - Province [THIS.Province.GetNameNoTooltip]"
            every_in_list = {
                variable = kCAFG_province_cultures
                limit = {
                    scope:character_culture != this
                    cultural_acceptance = {
                        target = root.culture
                        value >= {
                            value = V_kCAFG_cultural_benefits_minimum_base_cultural_acceptance
                            if = {
                                limit = { scope:char = { knows_language_of_culture = prev } }
                                divide = define:NCharacterOpinion|CULTURE_ACCEPTANCE_MULT_KNOWS_LANGUAGE
                            }
                        }
                    }
                }

                scope:char = {
                    add_to_variable_list = {
                        name = domain_cultures
                        target = prev
                    }
                }
            }
        }
        ordered_in_list = {
            variable = domain_cultures
            max = 10
            check_range_bounds = no
            limit = {
                save_temporary_scope_as = kCAFG_culture
                ## ALL provinces of that culture must be eligible
                NOT = {
                    scope:char = {
                        any_directly_owned_province = {
                            T_kCAFG_does_province_contain_culture = { CULTURE = scope:kCAFG_culture }
                            T_kCAFG_is_province_eligible_for_cultural_benefits = no
                        }
                    }
                }
            }
            order_by = {
                value = 0
                save_temporary_scope_as = kCAFG_culture
                scope:char = { every_directly_owned_province = { add = V_kCAFG_province_culture_amount } }
            }

            save_temporary_scope_as = kCAFG_culture
            #set_global_variable = { name = kCAFG_TMP_TEST_CHANCE value = V_kCAFG_cultural_benefits_trigger_chance }
            #debug_log = "kCAFG - Cultural Benefits - chance for culture [THIS.Culture.GetNameNoTooltip]: [GetGlobalVariable('kCAFG_TMP_TEST_CHANCE').GetValue]"
            random = {
                chance = V_kCAFG_cultural_benefits_trigger_chance
                save_scope_as = culture
                #debug_log = "kCAFG - Cultural Benefits - Triggered for culture [THIS.Culture.GetNameNoTooltip]!"
                scope:char = {
                    if = {
                        limit = { NOT = { has_variable = kCAFG_cultural_benefits_count } }
                        set_variable = { name = kCAFG_cultural_benefits_count value = 0 }
                    }
                    change_variable = { name = kCAFG_cultural_benefits_count add = 1 }
                    random_directly_owned_province = {
                        weight = {
                            base = 0
                            modifier = {
                                add = {
                                    scope:culture = { prev = { value = V_kCAFG_province_prev_culture_amount } }
                                }
                            }
                        }
                        #debug_log = "kCAFG - Cultural Benefits - Selected Province [THIS.Province.GetNameNoTooltip]"
                        save_scope_as = province
                    }
                    E_kCAFG_trigger_cultural_benefits = yes
                }
            }
        }
        clear_variable_list = domain_cultures
    }
}
