                education_martial_prowess_2 = { weight = { base = 20 } }
                education_martial_prowess_3 = { weight = { base = 30 } }
                education_martial_prowess_4 = { weight = { base = 10 } }
            }
            
			culture = scope:culture
            faith = scope:new_character_faith
            location = scope:province
            after_creation = {
                save_scope_as = impressive_character
                save_scope_as = introduced_character

                E_kCAFG_add_grateful_opinion = yes

                E_kei_character_creation_nudge_cultural_traits = yes
                set_interesting_traits_and_modifiers_effect = yes
                add_trait = lifestyle_blademaster
                add_random_tiered_trait_xp_effect = {
                    TRAIT = lifestyle_blademaster
                    LEVEL_1 = yes
                    LEVEL_3 = no
                }
                random = {
                    chance = 15
                    add_trait = lifestyle_hunter
                    add_random_tiered_trait_track_xp_effect = {
                        TRAIT = lifestyle_hunter
                        TRACK = hunter
                        LEVEL_1 = yes
                        LEVEL_3 = no
                    }
                }
                random = {
                    chance = 80
                    add_trait = tourney_participant
                    add_random_tourney_participant_level_effect = yes
                }
                force_character_skill_recalculation = yes
            }
        }
    }
}

# tradition_horse_breeder
E_kCAFG_cultural_boon_tradition_horse_breeder = {
    save_scope_value_as = { name = boon_type value = flag:maa_refill_gift }

    hidden_effect = {
        set_variable = {
            name = cavalry_gift
            value = {
                value = { 25 75 }
                add = {
                    value = 0
                    every_directly_owned_province = {
                        limit = { T_kCAFG_does_province_contain_culture = { CULTURE = scope:culture } }
                        scope:culture = { prev = { add = V_kCAFG_province_prev_culture_amount } }
                    }
                    multiply = 100
                }
                max = 1000
            }
        }
        save_scope_value_as = { name = gifted_amount value = var:cavalry_gift }
    }

    if = {
        limit = {
            var:cavalry_gift > 0
            any_maa_regiment = {
                maa_current_troops_count < maa_max_troops_count
                OR = {
                    is_unit_type = light_cavalry
                    is_unit_type = heavy_cavalry
                }
            }
        }
        show_as_tooltip = { custom_tooltip = E_kCAFG_cultural_boon_tradition_maa_cavalry_refill_tt }
    }

    hidden_effect = {
        while = {
            limit = {
                var:cavalry_gift > 0
                any_maa_regiment = {
                    maa_current_troops_count < maa_max_troops_count
                    OR = {
                        is_unit_type = light_cavalry
                        is_unit_type = heavy_cavalry
                    }
                }
            }
        
            random_maa_regiment = {
                limit = {
                    maa_current_troops_count < maa_max_troops_count
                    OR = {
                        is_unit_type = light_cavalry
                        is_unit_type = heavy_cavalry
                    }
                }
                set_local_variable = {
                    name = refilled_amount
                    value = {
                        value = 100
                        max = root.var:cavalry_gift
                        max = {
                            value = maa_max_troops_count
                            subtract = maa_current_troops_count
                        }
                    }
                }
                change_maa_troops_count = {
                    value = maa_current_troops_count
                    add = local_var:refilled_amount
                    max = maa_max_troops_count
                }
                root = {
                    change_variable = {
                        name = cavalry_gift
                        subtract = local_var:refilled_amount
                    }
                }
            }
        }
    }

    if = {
        limit = { var:cavalry_gift > 0 }

        save_scope_value_as = { name = maa_refill_gift_name value = flag:light_horsemen }
        show_as_tooltip = { custom_tooltip = E_kCAFG_cultural_boon_tradition_maa_gift_tt }
        
        hidden_effect = {
            spawn_army = {
                men_at_arms = {
                    type = light_horsemen
                    stacks = {
                        value = root.var:cavalry_gift
                        divide = 100
                        floor = yes
                        add = 1
                    }
                }
                inheritable = no
                location = scope:province
                save_scope_as = new_army
                name = kCAFG_horse_breeders_gift_name
            }

            while = {
                limit = { var:cavalry_gift > 100 }
                
                change_variable = {
                    name = cavalry_gift
                    subtract = 100
                }
            }

            # Not set if not at war, unfortunately
            scope:new_army ?= {
                random_army_maa_regiment = {
                    change_maa_troops_count = root.var:cavalry_gift
                }
            }
        }
    }
}

# tradition_esteemed_hospitality
E_kCAFG_cultural_boon_tradition_esteemed_hospitality = {
    hidden_effect = {
        save_scope_value_as = { name = boon_type value = flag:esteemed_hospitality }

        E_kCAFG_cultural_boon_tradition_esteemed_hospitality_create_character = yes

        scope:created_character = { save_scope_as = introduced_character }

        E_kCAFG_cultural_boon_tradition_esteemed_hospitality_create_character = yes

        scope:created_character = { save_scope_as = introduced_character_2 }

        E_kCAFG_cultural_boon_tradition_esteemed_hospitality_create_character = yes

        scope:created_character = { save_scope_as = introduced_character_3 }

        random_list = {
            1 = { scope:introduced_character = { E_kCAFG_cultural_boon_tradition_esteemed_hospitality_boost_character = yes } }
            1 = { scope:introduced_character_2 = { E_kCAFG_cultural_boon_tradition_esteemed_hospitality_boost_character = yes } }
            1 = { scope:introduced_character_3 = { E_kCAFG_cultural_boon_tradition_esteemed_hospitality_boost_character = yes } }
        }
    }
}

E_kCAFG_cultural_boon_tradition_esteemed_hospitality_create_character = {
    E_kCAFG_get_generated_character_random_province_faith = yes
    create_character = {
        template = kCAFG_cultural_benefits_character_base
        dynasty = none
        random_traits_list = {
            education_diplomacy_3 = { weight = { base = 20 } }
            education_diplomacy_4 = { weight = { base = 10 } }
            education_martial_3 = { weight = { base = 20 } }
            education_martial_4 = { weight = { base = 10 } }
            education_stewardship_3 = { weight = { base = 20 } }
            education_stewardship_4 = { weight = { base = 10 } }
            education_intrigue_3 = { weight = { base = 20 } }
            education_intrigue_4 = { weight = { base = 10 } }
            education_learning_3 = { weight = { base = 20 } }
            education_learning_4 = { weight = { base = 10 } }
        }
        
        employer = root
        after_creation = {
            save_scope_as = created_character
            E_kCAFG_add_grateful_opinion = yes
            E_kei_character_creation_nudge_cultural_traits = yes
            set_interesting_traits_and_modifiers_effect = yes
            if = {
                limit = { has_trait = shy }
                remove_trait = shy
                add_trait = gregarious
            }
            if = {
                limit = { has_trait = paranoid }
                remove_trait = paranoid
                add_trait = trusting
            }
            if = {
                limit = { has_trait = lazy }
                remove_trait = lazy
                add_trait = diligent
            }
            if = {
                limit = { has_trait = craven }
                remove_trait = craven
                add_trait = brave
            }
            add_gold = { 1 32 }
            force_character_skill_recalculation = yes
        }
    }
}

E_kCAFG_generic_cultural_boon_MAA_gift = {
    save_scope_value_as = { name = boon_type value = flag:maa_refill_gift }
	save_scope_value_as = { name = maa_refill_gift_name value = flag:$MAA_TYPE$ }

    hidden_effect = {
        set_variable = {
            name = maa_gift
            value = {
                value = { 25 75 }
                add = {
                    value = 0
                    every_directly_owned_province = {
                        limit = { T_kCAFG_does_province_contain_culture = { CULTURE = scope:culture } }
                        scope:culture = { prev = { add = V_kCAFG_province_prev_culture_amount } }
                        multiply = 0.8
                        add = 0.2
                        multiply = building_levies
                        divide = 2
                        multiply = {
                            value = county.development_level
                            divide = max_development_level
                            multiply = 0.75
                            add = 0.25
                        }
                    }
                }
                max = 1000
                multiply = $SCALE$
                min = 25
            }
        }
        save_scope_value_as = { name = gifted_amount value = var:maa_gift }
    }

    if = {
        limit = {
            var:maa_gift > 0
            any_maa_regiment = {
                maa_current_troops_count < maa_max_troops_count
                is_maa_type = $MAA_TYPE$
            }
        }
        save_scope_value_as = { name = maa_refilled value = flag:yes }
        show_as_tooltip = { custom_tooltip = E_kCAFG_cultural_boon_tradition_maa_refill_tt }
    }

    hidden_effect = {
        while = {
            limit = {
                var:maa_gift > 0
                any_maa_regiment = {
                    maa_current_troops_count < maa_max_troops_count
                    is_maa_type = $MAA_TYPE$
                }
            }
        
            random_maa_regiment = {
                limit = {
                    maa_current_troops_count < maa_max_troops_count
                    is_maa_type = $MAA_TYPE$
                }
                set_local_variable = {
                    name = refilled_amount
                    value = {
                        value = 100
                        max = root.var:maa_gift
                        max = {
                            value = maa_max_troops_count
                            subtract = maa_current_troops_count
                        }
                    }
                }
                change_maa_troops_count = {
                    value = maa_current_troops_count
                    add = local_var:refilled_amount
                    max = maa_max_troops_count
                }
                root = {
                    change_variable = {
                        name = maa_gift
                        subtract = local_var:refilled_amount
                    }
                }
            }
        }
    }

    if = {
        limit = { var:maa_gift > 0 }

        save_scope_value_as = { name = maa_gifted value = flag:yes }
        show_as_tooltip = { custom_tooltip = E_kCAFG_cultural_boon_tradition_maa_gift_tt }
        
        hidden_effect = {
            spawn_army = {
                men_at_arms = {
                    type = $MAA_TYPE$
                    stacks = {
                        value = root.var:maa_gift
                        divide = 100
                        floor = yes
                        add = 1
                    }
                }
                inheritable = no
                location = scope:province
                save_scope_as = new_army
                name = kCAFG_cultural_maa_gift_name
            }

            while = {
                limit = { var:maa_gift > 100 }
                
                change_variable = {
                    name = maa_gift
                    subtract = 100
                }
            }

            # Not set if not at war, unfortunately
            scope:new_army ?= {
                random_army_maa_regiment = {
                    change_maa_troops_count = root.var:maa_gift
                }
            }
        }
    }
}

E_kCAFG_add_cultural_tradition_commander_trait = {
    if = {
        limit = {
            culture = { has_cultural_parameter = winter_soldier_trait_more_common }
            NOT = { has_trait = winter_soldier }
        }
        add_trait = winter_soldier
    }
    if = {
        limit = {
            culture = { has_cultural_parameter = forest_fighter_trait_more_common }
            NOT = { has_trait = forest_fighter }
        }
        add_trait = forest_fighter
    }
    if = {
        limit = {
            culture = { has_cultural_parameter = rough_terrain_expert_trait_more_common }
            NOT = { has_trait = rough_terrain_expert }
        }
        add_trait = rough_terrain_expert
    }
    if = {
        limit = {
            culture = { has_cultural_parameter = desert_warrior_trait_more_common }
            NOT = { has_trait = desert_warrior }
        }
        add_trait = desert_warrior
    }
    if = {
        limit = {
            culture = { has_cultural_parameter = jungle_stalker_trait_more_common }
            NOT = { has_trait = jungle_stalker }
        }
        add_trait = jungle_stalker
    }
}
