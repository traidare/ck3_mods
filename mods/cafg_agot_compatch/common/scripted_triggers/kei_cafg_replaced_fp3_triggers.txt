favour_skilled_outsiders_decision_valid_character_to_take_trigger = {
    NOT = { has_realm_law = kCAFG_culture_intolerant_law }
	OR = {
		has_realm_law = kCAFG_culture_tolerant_law
		has_perk = meritocracy_perk
		government_has_flag = government_is_clan
		culture = { has_cultural_pillar = ethos_egalitarian }
		calc_true_if = {
			amount >= 2
			has_trait = calm
			has_trait = humble
			has_trait = ambitious
		}
	}
}


favour_skilled_outsiders_decision_valid_county_to_generate_trigger = {
	any_in_list = {
		variable = kCAFG_county_cultures
		kCAFG_favour_skilled_outsiders_decision_valid_culture_to_generate_trigger = yes
	}
}
